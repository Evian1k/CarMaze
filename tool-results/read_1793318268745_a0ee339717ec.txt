    54→model Conversation {
    55→  id                 String   @id @default(cuid())
    56→  orgId              String
    57→  org                Org      @relation(fields: [orgId], references: [id], onDelete: Cascade)
    58→  customerId         String
    59→  customer           User     @relation("Conversations", fields: [customerId], references: [id])
    60→  subject            String? // e.g. "Accident repair — KDJ 345K"
    61→  vehicleId          String? // optional vehicle listing reference
    62→  vehicle            Vehicle? @relation(fields: [vehicleId], references: [id])
    63→  serviceId          String? // optional garage service reference
    64→  kind               String   @default("GENERAL") // GENERAL|QUOTE|BOOKING|VEHICLE_ENQUIRY|SUPPORT
    65→  status             String   @default("OPEN") // OPEN|CLOSED
    66→  lastMessageAt      DateTime @default(now())
    67→  lastMessagePreview String?
    68→  unreadForOrg       Boolean  @default(false)
    69→  unreadForCustomer  Boolean  @default(false)
    70→  orgArchived        Boolean  @default(false)
    71→  customerArchived   Boolean  @default(false)
    72→  createdAt          DateTime @default(now())
    73→  updatedAt          DateTime @updatedAt
    74→
    75→  messages ConversationMessage[]
    76→
    77→  @@index([orgId, lastMessageAt])
    78→  @@index([customerId, lastMessageAt])
    79→}
