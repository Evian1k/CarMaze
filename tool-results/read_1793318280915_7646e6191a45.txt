    60→/** KES major-unit input → integer minor units (×100). Returns null when invalid. */
    61→function kesToMinor(input: string): number | null {
    62→  const v = input.trim().replace(/,/g, '')
    63→  if (!/^\d+(\.\d{1,2})?$/.test(v)) return null
    64→  const minor = Math.round(parseFloat(v) * 100)
    65→  return Number.isFinite(minor) && minor > 0 ? minor : null
    66→}

    76→export default function GarageDashboardView() {
    77→  const { user, loading } = useAuth()
    78→  const route = useHashRoute()
    79→  const [phase, setPhase] = useState<'loading' | 'wizard' | 'application' | 'dashboard'>('loading')
    80→  const [garages, setGarages] = useState<MyGarage[]>([])
    81→  const [activeOrgId, setActiveOrgId] = useState<string>('')
    82→  const [application, setApplication] = useState<GarageApplication | null>(null)
    83→
    84→  const activeGarage = garages.find((g) => g.org.id === activeOrgId) ?? garages[0] ?? null
    85→
    86→  // ── Bootstrap: which phase are we in? ──
    87→  const bootstrap = useCallback(() => {
    88→    if (!user) return
    89→    let appsFetched: GarageApplication[] | null = null
    90→    let mineFetched: MyGarage[] | null = null
    91→    const decide = () => {
    92→      if (appsFetched === null || mineFetched === null) return
    93→      if (mineFetched.length > 0) {
    94→        setGarages(mineFetched)
    95→        setActiveOrgId(mineFetched[0].org.id)
    96→        setPhase('dashboard')
    97→      } else {
    98→        const app = appsFetched.find((a) => a.type === 'GARAGE') ?? null
    99→        setApplication(app)
   100→        setPhase(app ? 'application' : 'wizard')
   101→      }
   102→    }
   103→    apiGet<{ applications: (GarageApplication & { type: string })[] }>('/api/applications')
   104→      .then((d) => { appsFetched = d.applications; decide() })
   105→      .catch(() => { appsFetched = []; decide() })
   106→    apiGet<{ garages: MyGarage[] }>('/api/garages?mine=1')
   107→      .then((d) => { mineFetched = d.garages; decide() })
   108→      .catch(() => { mineFetched = []; decide() })
   109→  }, [user])
   110→
   111→  useEffect(() => {
   112→    if (!loading && user) bootstrap()
   113→  }, [loading, user, bootstrap])
   114→
   115→  // ─────────────────────────── Gate: signed out ───────────────────────────
   116→  if (!loading && !user) {
   117→    return (
   118→      <div className="mx-auto max-w-md px-4 py-20 text-center">
   119→        <div className="mx-auto flex h-14 w-14 items-center justify-center rounded-full bg-[#315e4b]/10 text-[#315e4b]"><Wrench className="h-7 w-7" /></div>
   120→        <h1 className="mt-4 text-2xl font-bold">Garage Dashboard</h1>
   121→        <p className="mt-2 text-sm text-muted-foreground">Sign in with your garage account to manage services, quotes, bookings and reviews.</p>
   122→        <Button className="mt-6" onClick={() => navigate('signin')}>Sign in</Button>
   123→      </div>
   124→    )
   125→  }
   126→
   127→  if (loading || phase === 'loading') {
   128→    return (
   129→      <div className="mx-auto max-w-5xl space-y-4 p-6">
   130→        <Skeleton className="h-8 w-64" />
   131→        <Skeleton className="h-24 rounded-xl" />
   132→        <Skeleton className="h-64 rounded-xl" />
   133→      </div>
   134→    )
   135→  }
   136→
   137→  // ─────────────────────────── Gate: application wizard ───────────────────────────
   138→  if (phase === 'wizard') {
   139→    return <ApplicationWizard user={user!} onDone={bootstrap} />
   140→  }
   141→
   142→  // ─────────────────────────── Gate: application under review ───────────────────────────
   143→  if (phase === 'application' && application) {
   144→    return (
   145→      <ApplicationStatus
   146→        application={application}
   147→        onResubmitted={bootstrap}
   148→        onNewApplication={() => { setApplication(null); setPhase('wizard') }}
   149→      />
   150→    )
   151→  }
   152→
   153→  // ─────────────────────────── Dashboard ───────────────────────────
   154→  return <Dashboard garages={garages} activeGarage={activeGarage!} activeOrgId={activeOrgId} onOrgChange={setActiveOrgId} initialTab={route.query.get('tab') ?? 'overview'} onRefresh={bootstrap} />
   155→}

   292→function ServicesTab({ orgId, services, onChanged }: { orgId: string; services: ServiceRow[]; onChanged: () => void }) {
   293→  const [dialogOpen, setDialogOpen] = useState(false)
   294→  const [editing, setEditing] = useState<ServiceRow | null>(null)
   295→  const [category, setCategory] = useState<string>('GENERAL_SERVICE')
   296→  const [title, setTitle] = useState('')
   297→  const [description, setDescription] = useState('')
   298→  const [price, setPrice] = useState('')
   299→  const [saving, setSaving] = useState(false)
   300→
   301→  function openCreate() {
   302→    setEditing(null)
   303→    setCategory('GENERAL_SERVICE')
   304→    setTitle('')
   305→    setDescription('')
   306→    setPrice('')
   307→    setDialogOpen(true)
   308→  }
   309→
   310→  function openEdit(s: ServiceRow) {
   311→    setEditing(s)
   312→    setCategory(s.category)
   313→    setTitle(s.title)
   314→    setDescription(s.description ?? '')
   315→    setPrice(s.priceFromMinor ? String(s.priceFromMinor / 100) : '')
   316→    setDialogOpen(true)
   317→  }
   318→
   319→  const priceMinor = price.trim() ? kesToMinor(price) : null
   320→  const priceInvalid = price.trim() !== '' && priceMinor === null
   321→
   322→  function save() {
   323→    if (title.trim().length < 3) { toast.error('Give the service a clear title'); return }
   324→    if (priceInvalid) { toast.error('Price must be a KES amount like 3500 or 3500.50'); return }
   325→    setSaving(true)
   326→    const payload = {
   327→      category,
   328→      title: title.trim(),
   329→      description: description.trim() || null,
   330→      priceFromMinor: priceMinor,
   331→    }
   332→    const call = editing
   333→      ? apiSend(`/api/garage-services?id=${editing.id}`, 'PATCH', payload)
   334→      : apiSend('/api/garage-services', 'POST', { ...payload, orgId })
   335→    call
   336→      .then(() => {
   337→        toast.success(editing ? 'Service updated' : 'Service added')
   338→        setDialogOpen(false)
   339→        onChanged()
   340→      })
   341→      .catch((e) => toast.error(e instanceof Error ? e.message : 'Could not save service'))
   342→      .finally(() => setSaving(false))
   343→  }
   344→
   345→  function toggleActive(s: ServiceRow) {
   346→    apiSend(`/api/garage-services?id=${s.id}`, 'PATCH', { active: !s.active })
   347→      .then(() => { toast.success(s.active ? 'Service deactivated' : 'Service activated'); onChanged() })
   348→      .catch((e) => toast.error(e instanceof Error ? e.message : 'Could not update service'))
   349→  }
   350→
   351→  return (
   352→    <div className="space-y-4">
   353→      <div className="flex flex-wrap items-center justify-between gap-2">
   354→        <p className="text-sm text-muted-foreground">Active services decide which customer requests you see. Deactivate instead of deleting to keep history.</p>
   355→        <Button onClick={openCreate} className="gap-1.5"><Plus className="h-4 w-4" /> Add service</Button>
   356→      </div>
   357→
   358→      {services.length === 0 ? (
   359→        <EmptyBlock icon={<Wrench className="h-6 w-6" />} title="No services yet" desc="Add the categories you cover — e.g. general servicing, tyres, or modifications — so matching requests reach you." />
   360→      ) : (
   361→        <div className="grid gap-3 md:grid-cols-2">
   362→          {services.map((s) => (
   363→            <Card key={s.id} className={`p-4 ${!s.active ? 'opacity-60' : ''}`}>
   364→              <div className="flex items-start justify-between gap-2">
   365→                <div className="min-w-0">
   366→                  <div className="flex flex-wrap items-center gap-2">
   367→                    <h4 className="font-semibold">{s.title}</h4>
   368→                    <Badge variant={s.active ? 'secondary' : 'outline'}>{s.active ? 'Active' : 'Inactive'}</Badge>
   369→                  </div>
   370→                  <p className="mt-0.5 text-xs text-muted-foreground">{SERVICE_CATEGORY_LABELS[s.category] ?? s.category}</p>
   371→                  {s.description && <p className="mt-2 line-clamp-2 text-sm text-muted-foreground">{s.description}</p>}
   372→                  <p className="mt-2 text-sm font-medium">{s.priceFromMinor ? `from ${money(s.priceFromMinor)}` : 'Price on quote'}</p>
   373→                </div>
   374→                <div className="flex shrink-0 flex-col gap-1.5">
   375→                  <Button size="sm" variant="outline" className="gap-1" onClick={() => openEdit(s)}><Pencil className="h-3.5 w-3.5" /> Edit</Button>
   376→                  <Button size="sm" variant="ghost" onClick={() => toggleActive(s)}>{s.active ? 'Deactivate' : 'Activate'}</Button>
   377→                </div>
   378→              </div>
   379→            </Card>
   380→          ))}
   381→        </div>
   382→      )}
   383→
   384→      <Dialog open={dialogOpen} onOpenChange={setDialogOpen}>
   385→        <DialogContent className="max-h-[85vh] overflow-y-auto sm:max-w-lg">
   386→          <DialogHeader>
   387→            <DialogTitle>{editing ? 'Edit service' : 'Add a service'}</DialogTitle>
   388→            <DialogDescription>Shown on your public garage profile. &quot;From&quot; pricing sets customer expectations.</DialogDescription>
   389→          </DialogHeader>
   390→          <div className="space-y-4">
   391→            <div className="space-y-1.5">
   392→              <Label htmlFor="svc-category">Category</Label>
   393→              <Select value={category} onValueChange={setCategory}>
   394→                <SelectTrigger id="svc-category"><SelectValue /></SelectTrigger>
   395→                <SelectContent>
   396→                  {SERVICE_CATEGORIES.map((c) => <SelectItem key={c} value={c}>{SERVICE_CATEGORY_LABELS[c]}</SelectItem>)}
   397→                </SelectContent>
   398→              </Select>
   399→            </div>
   400→            <div className="space-y-1.5">
   401→              <Label htmlFor="svc-title">Title</Label>
   402→              <Input id="svc-title" value={title} onChange={(e) => setTitle(e.target.value)} placeholder="e.g. Full engine service" />
   403→            </div>
   404→            <div className="space-y-1.5">
   405→              <Label htmlFor="svc-desc">Description (optional)</Label>
   406→              <Textarea id="svc-desc" rows={3} value={description} onChange={(e) => setDescription(e.target.value)} placeholder="What's included, warranties, turnaround…" />
   407→            </div>
   408→            <div className="space-y-1.5">
   409→              <Label htmlFor="svc-price">Price from (KES, optional)</Label>
   410→              <div className="flex gap-2">
   411→                <Input id="svc-price" inputMode="decimal" value={price} onChange={(e) => setPrice(e.target.value)} placeholder="e.g. 6500" className="max-w-40" />
   412→                {priceMinor !== null && <Badge variant="secondary" className="self-center">{money(priceMinor)}</Badge>}
   413→              </div>
   414→              {priceInvalid && <p className="text-xs text-destructive">Enter a valid KES amount, e.g. 6500 or 6500.50</p>}

   429→function useOrgRequests(orgId: string) {
   430→  const [feed, setFeed] = useState<{ orgId: string; requests: LeadRequest[]; tick: number } | null>(null)
   431→  const [tick, setTick] = useState(0)
   432→  useEffect(() => {

   434→    apiGet<{ requests: LeadRequest[] }>(buildQuery('/api/service-requests', { org: 1, orgId }))
   435→      .then((d) => { if (!cancelled) setFeed({ orgId, requests: d.requests, tick }) })
   436→      .catch(() => { if (!cancelled) setFeed({ orgId, requests: [], tick }) })

   442→  return { requests, refresh }
