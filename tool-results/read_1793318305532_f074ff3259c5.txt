   794→function PendingApplication({ app, onUpdated }: { app: DealerApplication; onUpdated: () => void }) {
   795→  const [docs, setDocs] = useState<DocItem[]>([])
   796→  const [uploading, setUploading] = useState(false)
   797→  const [notes, setNotes] = useState('')
   798→  const [submitting, setSubmitting] = useState(false)
   799→
   800→  const needsInfo = app.status === 'MORE_INFO_REQUIRED'
   801→  const infoEvent = [...app.events].reverse().find((e) => e.type === 'INFO_REQUESTED')
   802→
   803→  async function handleFiles(files: FileList | null) {
   804→    if (!files || files.length === 0) return
   805→    setUploading(true)
   806→    try {
   807→      for (const file of Array.from(files)) {
   808→        const fd = new FormData()
   809→        fd.append('file', file)
   810→        fd.append('kind', 'document')
   811→        const res = await apiUpload<{ url: string; name: string }>('/api/upload', fd)
   812→        setDocs((prev) => [...prev, { name: res.name || file.name, url: res.url }])
   813→      }
   814→      toast.success('Document uploaded')
   815→    } catch (e) {
   816→      toast.error(e instanceof Error ? e.message : 'Upload failed — please try again.')
   817→    } finally {
   818→      setUploading(false)
   819→    }
   820→  }
   821→
   822→  async function resubmit() {
   823→    if (docs.length === 0 && notes.trim().length < 5) {
   824→      toast.error('Add a note or attach the requested documents before resubmitting.')
   825→      return
   826→    }
   827→    setSubmitting(true)
   828→    try {
   829→      await apiSend(`/api/applications/${app.id}`, 'PATCH', {
   830→        action: 'resubmit',
   831→        documents: docs.length > 0 ? docs : undefined,
   832→        notes: notes.trim() || undefined,
   833→      })
   834→      toast.success('Information resubmitted — the review team has been notified.')
   835→      onUpdated()
   836→    } catch (e) {
   837→      toast.error(e instanceof Error ? e.message : 'Could not resubmit.')
   838→    } finally {
   839→      setSubmitting(false)
   840→    }
   841→  }
   842→
   843→  return (
   844→    <div className="mx-auto w-full max-w-3xl px-3 py-8 sm:px-6 sm:py-12">
   845→      <Card>
   846→        <CardHeader>
   847→          <div className="flex flex-wrap items-center justify-between gap-2">
   848→            <CardTitle className="flex items-center gap-2 text-xl">
   849→              <Building2 className="h-5 w-5 text-primary" /> {app.orgName}
   850→            </CardTitle>
   851→            <StatusBadge status={app.status} labels={APPLICATION_STATUS_LABELS} classes={APPLICATION_BADGE_CLASS} />
   852→          </div>
   853→          <CardDescription className="flex flex-wrap items-center gap-x-3 gap-y-1">
   854→            <span className="font-mono text-xs">{app.refCode}</span>
   855→            <span aria-hidden>·</span>
   856→            <span>Submitted {app.submittedAt ? dateLabel(app.submittedAt) : '—'}</span>
   857→            <span aria-hidden>·</span>
   858→            <span className="flex items-center gap-1"><CalendarClock className="h-3.5 w-3.5" /> Initial review target: 24 business hours</span>
   859→          </CardDescription>
   860→        </CardHeader>
   861→        <CardContent className="space-y-5">
   862→          {app.escalated && (
   863→            <Alert className="border-accent/60 bg-accent/10">
   864→              <AlertTriangle className="h-4 w-4" />
   865→              <AlertTitle>Review is taking longer than our target</AlertTitle>
   866→              <AlertDescription>
   867→                Your application has been escalated to senior staff — no action is needed from you. We are sorry for the wait.
   868→              </AlertDescription>
   869→            </Alert>
   870→          )}
   871→
   872→          {needsInfo && (
   873→            <div className="space-y-3 rounded-lg border border-accent/60 bg-accent/10 p-4">
   874→              <div className="flex items-start gap-2">
   875→                <AlertTriangle className="mt-0.5 h-4 w-4 shrink-0" />
   876→                <div className="text-sm">
   877→                  <p className="font-semibold">The review team requested more information</p>
   878→                  {infoEvent?.note && <p className="mt-1 text-muted-foreground">{infoEvent.note}</p>}
   879→                  {app.correctionsDueAt && (
   880→                    <p className="mt-1 text-muted-foreground">Please respond before <strong>{dateLabel(app.correctionsDueAt)}</strong> to keep your application in the queue.</p>
   881→                  )}
   882→                </div>
   883→              </div>
   884→              <Separator />
   885→              <div className="space-y-3">
   886→                <DocUploader
   887→                  idPrefix="rs"
   888→                  docs={docs}
   889→                  uploading={uploading}
   890→                  onAdd={handleFiles}
   891→                  onRemove={(url) => setDocs((prev) => prev.filter((d) => d.url !== url))}
   892→                />
   893→                <div className="grid gap-1.5">
   894→                  <Label htmlFor="rs-notes">Note for the review team</Label>
   895→                  <Textarea id="rs-notes" rows={3} value={notes} onChange={(e) => setNotes(e.target.value)} placeholder="Explain what you corrected or attached…" maxLength={2000} />
   896→                </div>
   897→                <Button onClick={resubmit} disabled={submitting} className="min-h-11">
   898→                  {submitting ? <Loader2 className="h-4 w-4 animate-spin" /> : <Send className="h-4 w-4" />} Resubmit application
   899→                </Button>
   900→              </div>
   901→            </div>
   902→          )}
   903→
   904→          {!needsInfo && app.status !== 'APPROVED' && (
   905→            <p className="text-sm leading-relaxed text-muted-foreground">
   906→              Your application is in the review queue. You will receive an in-app notification as soon as the review team
   907→              updates it — this page updates automatically when you revisit.
   908→            </p>
   909→          )}
   910→          {app.status === 'APPROVED' && (
   911→            <Alert>
   912→              <BadgeCheck className="h-4 w-4 text-primary" />
   913→              <AlertTitle>Application approved</AlertTitle>
   914→              <AlertDescription>
   915→                Your dealership has been verified. If your DealerOS workspace does not appear shortly, sign out and back in
   916→                once — or contact support with your reference {app.refCode}.
   917→              </AlertDescription>
   918→            </Alert>
   919→          )}
   920→
   921→          {/* Event timeline */}
   922→          <div>
   923→            <h3 className="text-sm font-semibold">Application timeline</h3>
   924→            <ol className="mt-3 space-y-0">
   925→              {app.events.map((e, i) => (
   926→                <li key={e.id} className="relative flex gap-3 pb-5 last:pb-0">
   927→                  {i < app.events.length - 1 && <span aria-hidden className="absolute left-[7px] top-4 h-full w-px bg-border" />}
   928→                  <span className={cn('mt-1 h-3.5 w-3.5 shrink-0 rounded-full border-2', ['APPROVED'].includes(e.type) ? 'border-primary bg-primary' : ['INFO_REQUESTED', 'REJECTED', 'ESCALATED'].includes(e.type) ? 'border-accent bg-accent' : 'border-primary/40 bg-background')} />
   929→                  <div className="min-w-0">
   930→                    <p className="text-sm font-medium">{EVENT_LABELS[e.type] ?? e.type}</p>
   931→                    {e.note && <p className="mt-0.5 text-xs leading-relaxed text-muted-foreground">{e.note}</p>}
   932→                    <p className="mt-0.5 text-[11px] text-muted-foreground">
   933→                      {timeAgo(e.createdAt)}{e.actorName ? ` · ${e.actorName}` : ''}
   934→                    </p>
   935→                  </div>
   936→                </li>
   937→              ))}
   938→            </ol>
   939→          </div>
   940→        </CardContent>
   941→      </Card>
   942→    </div>
   943→  )
   944→}
