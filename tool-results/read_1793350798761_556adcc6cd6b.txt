    78→function draftFromQuery(query: URLSearchParams): FilterDraft {
    79→  const minor = (key: string) => {
    80→    const raw = query.get(key)
    81→    if (!raw) return ''
    82→    const n = Number(raw)
    83→    return Number.isFinite(n) && n > 0 ? String(Math.round(n / 100)) : ''
    84→  }
    85→  return {
    86→    q: query.get('q') ?? '',
    87→    make: query.get('make') ?? '',
    88→    model: query.get('model') ?? '',
    89→    bodyType: query.get('bodyType') ?? '',
    90→    county: query.get('county') ?? '',
    91→    fuel: query.get('fuel') ?? '',
    92→    transmission: query.get('transmission') ?? '',
    93→    condition: query.get('condition') ?? '',
    94→    sellerType: query.get('sellerType') ?? '',
    95→    verified: query.get('verified') === '1',
    96→    minPrice: minor('minPrice'),
    97→    maxPrice: minor('maxPrice'),
    98→    minYear: query.get('minYear') ?? '',
    99→    maxYear: query.get('maxYear') ?? '',
   100→    maxMileage: query.get('maxMileage') ?? '',
   101→    sort: query.get('sort') || 'newest',
   102→    page: query.get('page') || '1',
   103→    org: query.get('org') ?? '',
   104→  }
   105→}
   106→
   107→/** Convert the editable draft into URL query params (prices back to minor units). */
   108→function draftToParams(d: FilterDraft): Record<string, string | number | undefined> {
   109→  const num = (s: string) => {
   110→    const n = Number(s)
   111→    return s !== '' && Number.isFinite(n) && n > 0 ? n : undefined
   112→  }
   113→  return {
   114→    q: d.q.trim() || undefined,
   115→    make: d.make || undefined,
   116→    model: d.model.trim() || undefined,
   117→    bodyType: d.bodyType || undefined,
   118→    county: d.county || undefined,
   119→    fuel: d.fuel || undefined,
   120→    transmission: d.transmission || undefined,
   121→    condition: d.condition || undefined,
   122→    sellerType: d.sellerType || undefined,
   123→    verified: d.verified ? '1' : undefined,
   124→    minPrice: num(d.minPrice) ? Math.round(Number(d.minPrice) * 100) : undefined,
   125→    maxPrice: num(d.maxPrice) ? Math.round(Number(d.maxPrice) * 100) : undefined,
   126→    minYear: num(d.minYear),
   127→    maxYear: num(d.maxYear),
   128→    maxMileage: num(d.maxMileage),
   129→    sort: d.sort && d.sort !== 'newest' ? d.sort : undefined,
   130→    org: d.org || undefined,
   131→    // page intentionally omitted → resets to 1 whenever filters change
   132→  }
   133→}

   462→              onClick={() => { setAiResults(null); setAiMessage('') }}
   463→            >
   464→              Clear AI results
   465→            </Button>
   466→          </div>
   467→          {aiResults.length === 0 ? (
   468→            <p className="text-sm text-muted-foreground">No matches found for that description — try being more specific.</p>
   469→          ) : (
   470→            <div className="grid grid-cols-1 gap-4 sm:grid-cols-2 xl:grid-cols-4">
   471→              {aiResults.map((v) => (
   472→                <VehicleCard key={v.id} v={v} saved={savedIds.has(v.id)} onToggleSave={toggleSaved} />
   473→              ))}
   474→            </div>
   475→          )}
   476→        </section>
   477→      )}
   478→
   479→      <div className="flex gap-6">
   480→        {/* Desktop filter sidebar */}
   481→        <aside className="hidden w-72 shrink-0 lg:block" aria-label="Search filters">
   482→          <Card>
   483→            <CardContent className="p-6">
   484→              <div className="mb-4 flex items-center justify-between">
   485→                <h2 className="font-semibold">Filters</h2>
   486→                {activeFilterCount > 0 && <Badge variant="secondary">{activeFilterCount}</Badge>}
   487→              </div>
   488→              {filterFields}
   489→            </CardContent>
   490→          </Card>
   491→        </aside>
   492→
   493→        <div className="min-w-0 flex-1">
   494→          {/* Toolbar */}
   495→          <div className="mb-4 flex flex-wrap items-center justify-between gap-3">
   496→            <div className="flex items-center gap-2">
   497→              <Sheet open={filtersOpen} onOpenChange={setFiltersOpen}>
   498→                <SheetTrigger asChild>
   499→                  <Button variant="outline" className="h-11 px-4 lg:hidden">
   500→                    <SlidersHorizontal className="h-4 w-4" /> Filters
   501→                    {activeFilterCount > 0 && <Badge variant="secondary" className="ml-1">{activeFilterCount}</Badge>}
   502→                  </Button>
   503→                </SheetTrigger>
   504→                <SheetContent side="left" className="w-80 overflow-y-auto scrollbar-thin">
   505→                  <SheetHeader>
   506→                    <SheetTitle>Filters</SheetTitle>
   507→                  </SheetHeader>
   508→                  <div className="px-4 pb-6">{filterFields}</div>
   509→                </SheetContent>
   510→              </Sheet>
   511→              <span className="text-sm text-muted-foreground">
   512→                {loading ? 'Searching…' : `${total.toLocaleString()} results`}
   513→              </span>
   514→            </div>
   515→            <div className="flex items-center gap-2">
   516→              <Button
   517→                variant="outline"
   518→                className="h-11 px-4"
   519→                onClick={() => {
   520→                  if (!user) {
   521→                    toast.info('Sign in to save this search')
   522→                    navigate('signin')
   523→                    return
   524→                  }
   525→                  setSaveSearchOpen(true)
   526→                }}
   527→              >
   528→                <BookmarkPlus className="h-4 w-4" /> <span className="hidden sm:inline">Save search</span>
   529→              </Button>
   530→              <Select value={draft.sort} onValueChange={(v) => patchDraft({ sort: v })}>
   531→                <SelectTrigger className="h-11 w-[170px]" aria-label="Sort results">
   532→                  <SelectValue />
   533→                </SelectTrigger>
   534→                <SelectContent>
   535→                  {SORT_OPTIONS.map((o) => <SelectItem key={o.value} value={o.value}>{o.label}</SelectItem>)}
   536→                </SelectContent>
   537→              </Select>
   538→            </div>
   539→          </div>
   540→
   541→          {/* Results */}
   542→          {loading ? (
   543→            <div className="grid grid-cols-1 gap-4 sm:grid-cols-2 xl:grid-cols-4" aria-busy="true">
   544→              {Array.from({ length: 8 }).map((_, i) => (
   545→                <Card key={i} className="overflow-hidden">
   546→                  <Skeleton className="aspect-[16/10] w-full rounded-none" />
   547→                  <CardContent className="space-y-2 p-4">
   548→                    <Skeleton className="h-4 w-3/4" />
   549→                    <Skeleton className="h-5 w-1/2" />
   550→                    <Skeleton className="h-3 w-2/3" />
   551→                  </CardContent>
   552→                </Card>
   553→              ))}
   554→            </div>
   555→          ) : error ? (
   556→            <Card>
   557→              <CardContent className="flex flex-col items-center gap-3 p-10 text-center">
   558→                <SearchX className="h-10 w-10 text-muted-foreground" />
   559→                <p className="font-medium">Could not load listings</p>
   560→                <p className="text-sm text-muted-foreground">{error}</p>
   561→                <Button className="h-11 px-5" onClick={() => navigate('cars')}>Retry</Button>
   562→              </CardContent>
   563→            </Card>
   564→          ) : vehicles.length === 0 ? (
   565→            <Card>
   566→              <CardContent className="flex flex-col items-center gap-3 p-10 text-center">
   567→                <SearchX className="h-10 w-10 text-muted-foreground" />
   568→                <p className="font-medium">No vehicles match your search</p>
   569→                <p className="max-w-sm text-sm text-muted-foreground">
   570→                  Try widening your price range, removing a filter, or searching a different make.
   571→                </p>
   572→                <Button className="h-11 px-5" onClick={clearFilters}>Clear all filters</Button>
   573→              </CardContent>
   574→            </Card>
   575→          ) : (
   576→            <div className="grid grid-cols-1 gap-4 sm:grid-cols-2 xl:grid-cols-4">
   577→              {vehicles.map((v) => (
   578→                <div key={v.id} className={`relative rounded-xl ${compareIds.includes(v.id) ? 'ring-2 ring-accent ring-offset-2 ring-offset-background' : ''}`}>
   579→                  <VehicleCard v={v} saved={savedIds.has(v.id)} onToggleSave={toggleSaved} />
   580→                  <button
   581→                    onClick={() => toggleCompare(v)}
   582→                    aria-pressed={compareIds.includes(v.id)}
   583→                    aria-label={compareIds.includes(v.id) ? `Remove ${v.make} ${v.model} from compare` : `Add ${v.make} ${v.model} to compare`}
   584→                    className={`absolute right-12 top-2 z-10 flex h-9 w-9 items-center justify-center rounded-full backdrop-blur-sm transition-colors ${
   585→                      compareIds.includes(v.id) ? 'bg-accent text-accent-foreground' : 'bg-black/35 text-white hover:bg-black/55'
   586→                    }`}
   587→                  >
   588→                    <Scale className="h-4 w-4" />
   589→                  </button>
   590→                </div>
   591→              ))}
   592→            </div>
   593→          )}
   594→
   595→          {/* Pagination */}
   596→          {!loading && !error && totalPages > 1 && (
   597→            <nav className="mt-8 flex flex-wrap items-center justify-center gap-2" aria-label="Pagination">
   598→              <Button
   599→                variant="outline"
   600→                className="h-11 w-11 p-0"
   601→                disabled={page <= 1}
   602→                onClick={() => goToPage(page - 1)}
   603→                aria-label="Previous page"
   604→              >
   605→                ‹
   606→              </Button>
   607→              {pageNumbers.map((p, i) =>
   608→                p === '…' ? (
   609→                  <span key={`gap-${i}`} className="px-1 text-muted-foreground">…</span>
   610→                ) : (
   611→                  <Button
   612→                    key={p}
   613→                    variant={p === page ? 'default' : 'outline'}
   614→                    className="h-11 w-11 p-0"
   615→                    onClick={() => goToPage(p)}
   616→                    aria-current={p === page ? 'page' : undefined}
   617→                  >
   618→                    {p}
   619→                  </Button>
   620→                )
   621→              )}
   622→              <Button
   623→                variant="outline"
   624→                className="h-11 w-11 p-0"
   625→                disabled={page >= totalPages}
   626→                onClick={() => goToPage(page + 1)}
   627→                aria-label="Next page"
   628→              >
   629→                ›
   630→              </Button>
   631→            </nav>
   632→          )}
   633→        </div>
   634→      </div>
   635→
   636→      {/* Compare bar */}
   637→      {compareIds.length > 0 && (
   638→        <div className="fixed inset-x-0 bottom-0 z-40 border-t border-border bg-card/95 px-4 py-3 shadow-[0_-4px_20px_rgba(0,0,0,0.08)] backdrop-blur">
   639→          <div className="mx-auto flex max-w-7xl items-center justify-between gap-3">
   640→            <div className="flex items-center gap-2 overflow-x-auto scrollbar-thin">
   641→              <Scale className="h-5 w-5 shrink-0 text-primary" />
   642→              <span className="shrink-0 text-sm font-medium">
   643→                {compareIds.length} of 3 selected
   644→              </span>
   645→              <div className="hidden items-center gap-1 sm:flex">
   646→                {compareVehicles.map((v) => (
   647→                  <Badge key={v.id} variant="secondary" className="max-w-44 gap-1 truncate">
   648→                    <span className="truncate">{v.make} {v.model}</span>
   649→                    <button onClick={() => toggleCompare(v)} aria-label={`Remove ${v.make} ${v.model} from compare`} className="rounded-full p-0.5 hover:bg-foreground/10">
   650→                      <X className="h-3 w-3" />
   651→                    </button>
   652→                  </Badge>
   653→                ))}
   654→              </div>
   655→            </div>
   656→            <div className="flex shrink-0 items-center gap-2">
   657→              <Button variant="ghost" className="h-11 px-3" onClick={() => setCompareIds([])}>Clear</Button>
   658→              <Button className="h-11 px-5" disabled={compareVehicles.length < 2} onClick={() => setCompareOpen(true)}>
   659→                Compare {compareVehicles.length >= 2 ? `(${compareVehicles.length})` : ''}
   660→              </Button>
   661→            </div>
   662→          </div>
   663→        </div>
   664→      )}
   665→
   666→      {/* Compare dialog */}
   667→      <Dialog open={compareOpen} onOpenChange={setCompareOpen}>
   668→        <DialogContent className="max-h-[85vh] max-w-3xl overflow-y-auto scrollbar-thin">
   669→          <DialogHeader>
   670→            <DialogTitle>Compare vehicles</DialogTitle>
   671→            <DialogDescription>Side-by-side specification comparison.</DialogDescription>
   672→          </DialogHeader>
   673→          {compareVehicles.length >= 2 ? (
   674→            <div className="overflow-x-auto">
   675→              <table className="w-full min-w-[480px] text-sm">
   676→                <thead>
   677→                  <tr>
   678→                    <th className="w-28 p-2 text-left font-medium text-muted-foreground">Spec</th>
   679→                    {compareVehicles.map((v) => (
   680→                      <th key={v.id} className="p-2 text-left font-semibold">
   681→                        {v.make} {v.model}
   682→                        <div className="text-xs font-normal text-muted-foreground">{v.year}</div>
   683→                      </th>
   684→                    ))}
   685→                  </tr>
   686→                </thead>
   687→                <tbody>
   688→                  {COMPARE_ROWS.map((row) => (
   689→                    <tr key={row.label} className="border-t border-border">
   690→                      <td className="p-2 align-top text-muted-foreground">{row.label}</td>
   691→                      {compareVehicles.map((v) => (
   692→                        <td key={v.id} className="p-2 align-top font-medium">{row.get(v)}</td>
   693→                      ))}
   694→                    </tr>
   695→                  ))}
   696→                </tbody>
   697→              </table>
   698→            </div>
   699→          ) : (
   700→            <p className="text-sm text-muted-foreground">Select at least 2 vehicles to compare.</p>
   701→          )}
   702→        </DialogContent>
   703→      </Dialog>
   704→
   705→      {/* Save search dialog */}
   706→      <Dialog open={saveSearchOpen} onOpenChange={setSaveSearchOpen}>
   707→        <DialogContent className="max-w-md">
   708→          <DialogHeader>
   709→            <DialogTitle>Save this search</DialogTitle>
   710→            <DialogDescription>
   711→              We&apos;ll keep these filters so you can rerun them anytime from your account.
   712→            </DialogDescription>
   713→          </DialogHeader>
   714→          <div className="space-y-1.5">
   715→            <Label htmlFor="search-name">Search name</Label>
   716→            <Input
   717→              id="search-name"
   718→              placeholder="e.g. SUVs under 4M in Nairobi"
   719→              value={searchName}
   720→              onChange={(e) => setSearchName(e.target.value)}
   721→              onKeyDown={(e) => { if (e.key === 'Enter') saveSearch() }}
   722→            />
   723→            {route.query.toString() && (
   724→              <p className="truncate text-xs text-muted-foreground">Filters: {route.query.toString()}</p>
   725→            )}
   726→          </div>
   727→          <DialogFooter>
   728→            <Button variant="ghost" className="h-11" onClick={() => setSaveSearchOpen(false)}>Cancel</Button>
   729→            <Button className="h-11 px-5" onClick={saveSearch}>Save search</Button>
   730→          </DialogFooter>
   731→        </DialogContent>
   732→      </Dialog>
   733→    </div>
   734→  )
