    94→type Tone = 'green' | 'gold' | 'red' | 'neutral'
    95→
    96→function toneFor(status: string): Tone {
    97→  const green = ['PUBLISHED', 'APPROVED', 'ACTIVE', 'PAID', 'RESOLVED', 'SUCCESS', 'SENT', 'CONNECTED', 'CONFIGURED', 'DONE']
    98→  const gold = ['PENDING_REVIEW', 'PENDING', 'PENDING_PAYMENT', 'OPEN', 'IN_PROGRESS', 'SUBMITTED', 'UNDER_REVIEW', 'MORE_INFO_REQUIRED', 'RESUBMITTED', 'INITIATED', 'QUEUED', 'RESERVED', 'PAST_DUE']
    99→  const red = ['REJECTED', 'SUSPENDED', 'FAILED', 'OVERDUE', 'REVOKED', 'ERROR']
   100→  if (green.includes(status)) return 'green'
   101→  if (gold.includes(status)) return 'gold'
   102→  if (red.includes(status)) return 'red'
   103→  return 'neutral'
   104→}
   105→

   339→export default function View() {
   340→  const user = useAuth((s) => s.user)
   341→  const authLoading = useAuth((s) => s.loading)
   342→  const route = useHashRoute()
   343→  const [health, setHealth] = useState<Health | null>(null)
   344→  const [sweeping, setSweeping] = useState(false)
   345→
   346→  const tabParam = route.query.get('tab')
   347→  const tab: TabKey = (TABS as readonly string[]).includes(tabParam ?? '') ? (tabParam as TabKey) : 'overview'
   348→  const isStaff = hasAnyRole(user, ...STAFF_ROLES_CLIENT)
   349→
   350→  useEffect(() => {
   351→    let alive = true
   352→    apiGet<Health>('/api/health')
   353→      .then((d) => {
   354→        if (alive) setHealth(d)
   355→      })
   356→      .catch(() => {})
   357→    return () => {
   358→      alive = false
   359→    }
   360→  }, [])
   361→
   362→  function runSweep() {
   363→    setSweeping(true)
   364→    apiSend<{ reviewed: number; reviewerReminders: number; applicantReminders: number; escalations: number }>('/api/applications/sweep', 'POST')
   365→      .then((r) => {
   366→        toast.success(
   367→          `Sweep complete — ${r.reviewed} active applications checked, ${r.reviewerReminders} reviewer reminders, ${r.applicantReminders} applicant reminders, ${r.escalations} escalations.`
   368→        )
   369→      })
   370→      .catch((e: Error) => toast.error(e.message))
   371→      .finally(() => setSweeping(false))
   372→  }
   373→
   374→  if (authLoading) return <Loading />
   375→  if (!user || !isStaff) return <Restricted />
   376→
   377→  return (
   378→    <div className="min-h-screen bg-[#f7f6f2] dark:bg-background">
   379→      <main className="mx-auto max-w-7xl px-4 pb-16 pt-6 sm:px-6">
   380→        <header>
   381→          <div className="flex flex-wrap items-start justify-between gap-3">
   382→            <div>
   383→              <h1 className="flex items-center gap-2 text-xl font-semibold tracking-tight sm:text-2xl">
   384→                CARMAZE Control Center
   385→                <ShieldAlert className="h-5 w-5 text-primary" aria-hidden />
   386→              </h1>
   387→              <p className="mt-0.5 text-sm text-muted-foreground">
   388→                Platform operations, verification and billing — live aggregates, honest payment states.
   389→              </p>
   390→            </div>
   391→            <div className="flex flex-wrap items-center gap-2">
   392→              <HealthChips health={health} />
   393→              <Button size="sm" variant="outline" onClick={runSweep} disabled={sweeping}>
   394→                <RefreshCw className={`mr-1.5 h-4 w-4 ${sweeping ? 'animate-spin' : ''}`} aria-hidden />
   395→                {sweeping ? 'Sweeping…' : 'Run follow-up sweep'}
   396→              </Button>
   397→            </div>
   398→          </div>
   399→
   400→          <nav aria-label="Control Center sections" className="mt-5 flex gap-1 overflow-x-auto pb-1 [scrollbar-width:thin]">
   401→            {TABS.map((t) => (
   402→              <button
   403→                key={t}
   404→                type="button"
   405→                onClick={() => navigate(`admin?tab=${t}`)}
   406→                aria-current={tab === t ? 'page' : undefined}
   407→                className={`whitespace-nowrap rounded-full px-3.5 py-1.5 text-sm font-medium transition-colors ${
   408→                  tab === t ? 'bg-primary text-primary-foreground' : 'text-muted-foreground hover:bg-black/5 dark:hover:bg-white/10'
   409→                }`}
   410→              >
   411→                {TAB_LABELS[t]}
   412→              </button>
   413→            ))}
   414→          </nav>
   415→        </header>
   416→
   417→        <div className="mt-5">
   418→          {tab === 'overview' && <OverviewTab />}
   419→          {tab === 'applications' && <ApplicationsTab user={user} />}
   420→          {tab === 'listings' && <ListingsTab />}
   421→          {tab === 'users' && <UsersTab user={user} />}
   422→          {tab === 'reports' && <ReportsTab />}
   423→          {tab === 'billing' && <BillingTab user={user} />}
   424→          {tab === 'campaigns' && <CampaignsTab />}
   425→          {tab === 'support' && <SupportTab />}
   426→          {tab === 'notifications' && <NotificationsTab />}
   427→          {tab === 'settings' && <SettingsTab user={user} />}
   428→          {tab === 'audit' && <AuditTab />}
   429→        </div>
   430→      </main>
   431→    </div>
   432→  )
   433→}

   490→function OverviewTab() {
   491→  const [data, setData] = useState<OverviewData | null>(null)
   492→  const [error, setError] = useState<string | null>(null)
   493→  const [reload, setReload] = useState(0)
   494→
   495→  useEffect(() => {
   496→    let alive = true
   497→    apiGet<OverviewData>('/api/admin/overview')
   498→      .then((d) => {
   499→        if (alive) setData(d)
   500→      })
   501→      .catch((e: Error) => {
   502→        if (alive) setError(e.message)
   503→      })
   504→    return () => {
   505→      alive = false
   506→    }
   507→  }, [reload])
   508→
   509→  if (error) return <ErrorNote message={error} onRetry={() => setReload((r) => r + 1)} />
   510→  if (!data) return <LoadingCards n={9} />
   511→
   512→  const rev = data.revenue
   513→  const revenueChartData = rev.byType.map((t) => ({
   514→    name: t.type === 'SUBSCRIPTION' ? 'Subscriptions' : t.type === 'CAMPAIGN' ? 'Campaigns' : 'Lead fees',
   515→    Paid: t.paidMinor / 100,
   516→    Pending: t.pendingMinor / 100,
   517→  }))
   518→  const hasRevenue = rev.paidTotalMinor > 0 || rev.pendingTotalMinor > 0
   519→  const pieData = data.chart.listingsByStatus.map((s) => ({
   520→    name: VEHICLE_STATUS_LABELS[s.status] ?? s.status,
   521→    value: s.count,
   522→  }))
   523→  const hasListings = pieData.some((p) => p.value > 0)
   524→
