    58→model Conversation {
    59→  id                 String   @id @default(cuid())
    60→  orgId              String
    61→  org                Org      @relation(fields: [orgId], references: [id], onDelete: Cascade)
    62→  customerId         String
    63→  customer           User     @relation("Conversations", fields: [customerId], references: [id])
    64→  subject            String? // e.g. "Accident repair — KDJ 345K"
    65→  vehicleId          String? // optional vehicle listing reference
    66→  vehicle            Vehicle? @relation(fields: [vehicleId], references: [id])
    67→  serviceId          String? // optional garage service reference
    68→  kind               String   @default("GENERAL") // GENERAL|QUOTE|BOOKING|VEHICLE_ENQUIRY|SUPPORT
    69→  status             String   @default("OPEN") // OPEN|CLOSED
    70→  lastMessageAt      DateTime @default(now())
    71→  lastMessagePreview String?
    72→  unreadForOrg       Boolean  @default(false)
    73→  unreadForCustomer  Boolean  @default(false)
    74→  orgArchived        Boolean  @default(false)
    75→  customerArchived   Boolean  @default(false)
    76→  createdAt          DateTime @default(now())
    77→  updatedAt          DateTime @updatedAt
    78→
    79→  messages ConversationMessage[]
    80→
    81→  @@index([orgId, lastMessageAt])
    82→  @@index([customerId, lastMessageAt])
    83→}
