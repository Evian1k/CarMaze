    66→/** KES major-unit input → integer minor units (×100). Returns null when invalid. */
    67→function kesToMinor(input: string): number | null {
    68→  const v = input.trim().replace(/,/g, '')
    69→  if (!/^\d+(\.\d{1,2})?$/.test(v)) return null
    70→  const minor = Math.round(parseFloat(v) * 100)
    71→  return Number.isFinite(minor) && minor > 0 ? minor : null
    72→}

    82→export default function GarageDashboardView() {
    83→  const { user, loading } = useAuth()
    84→  const route = useHashRoute()
    85→  const [phase, setPhase] = useState<'loading' | 'wizard' | 'application' | 'dashboard'>('loading')
    86→  const [garages, setGarages] = useState<MyGarage[]>([])
    87→  const [activeOrgId, setActiveOrgId] = useState<string>('')
    88→  const [application, setApplication] = useState<GarageApplication | null>(null)
    89→
    90→  const activeGarage = garages.find((g) => g.org.id === activeOrgId) ?? garages[0] ?? null
    91→
    92→  // ── Bootstrap: which phase are we in? ──
    93→  const bootstrap = useCallback(() => {
    94→    if (!user) return
    95→    let appsFetched: GarageApplication[] | null = null
    96→    let mineFetched: MyGarage[] | null = null
    97→    const decide = () => {
    98→      if (appsFetched === null || mineFetched === null) return
    99→      if (mineFetched.length > 0) {
   100→        setGarages(mineFetched)
   101→        setActiveOrgId(mineFetched[0].org.id)
   102→        setPhase('dashboard')
   103→      } else {
   104→        const app = appsFetched.find((a) => a.type === 'GARAGE') ?? null
   105→        setApplication(app)
   106→        setPhase(app ? 'application' : 'wizard')
   107→      }
   108→    }
   109→    apiGet<{ applications: (GarageApplication & { type: string })[] }>('/api/applications')
   110→      .then((d) => { appsFetched = d.applications; decide() })
   111→      .catch(() => { appsFetched = []; decide() })
   112→    apiGet<{ garages: MyGarage[] }>('/api/garages?mine=1')
   113→      .then((d) => { mineFetched = d.garages; decide() })
   114→      .catch(() => { mineFetched = []; decide() })
   115→  }, [user])
   116→
   117→  useEffect(() => {
   118→    if (!loading && user) bootstrap()
   119→  }, [loading, user, bootstrap])
   120→
   121→  // ─────────────────────────── Gate: signed out ───────────────────────────
   122→  if (!loading && !user) {
   123→    return (
   124→      <div className="mx-auto max-w-md px-4 py-20 text-center">
   125→        <div className="mx-auto flex h-14 w-14 items-center justify-center rounded-full bg-primary/10 text-primary"><Wrench className="h-7 w-7" /></div>
   126→        <h1 className="mt-4 text-2xl font-bold">Garage Dashboard</h1>
   127→        <p className="mt-2 text-sm text-muted-foreground">Sign in with your garage account to manage services, quotes, bookings and reviews.</p>
   128→        <Button className="mt-6" onClick={() => navigate('signin')}>Sign in</Button>
   129→      </div>
   130→    )
   131→  }
   132→
   133→  if (loading || phase === 'loading') {
   134→    return (
   135→      <div className="mx-auto max-w-5xl space-y-4 p-6">
   136→        <Skeleton className="h-8 w-64" />
   137→        <Skeleton className="h-24 rounded-xl" />
   138→        <Skeleton className="h-64 rounded-xl" />
   139→      </div>
   140→    )
   141→  }
   142→
   143→  // ─────────────────────────── Gate: application wizard ───────────────────────────
   144→  if (phase === 'wizard') {
   145→    return <ApplicationWizard user={user!} onDone={bootstrap} />
   146→  }
   147→
   148→  // ─────────────────────────── Gate: application under review ───────────────────────────
   149→  if (phase === 'application' && application) {
   150→    return (
   151→      <ApplicationStatus
   152→        application={application}
   153→        onResubmitted={bootstrap}
   154→        onNewApplication={() => { setApplication(null); setPhase('wizard') }}
   155→      />
   156→    )
   157→  }
   158→
   159→  // ─────────────────────────── Dashboard ───────────────────────────
   160→  return <Dashboard garages={garages} activeGarage={activeGarage!} activeOrgId={activeOrgId} onOrgChange={setActiveOrgId} initialTab={route.query.get('tab') ?? 'overview'} onRefresh={bootstrap} />
   161→}

   316→function ServicesTab({ orgId, services, onChanged }: { orgId: string; services: ServiceRow[]; onChanged: () => void }) {
   317→  const [dialogOpen, setDialogOpen] = useState(false)
   318→  const [editing, setEditing] = useState<ServiceRow | null>(null)
   319→  const [category, setCategory] = useState<string>('GENERAL_SERVICE')
   320→  const [title, setTitle] = useState('')
   321→  const [description, setDescription] = useState('')
   322→  const [price, setPrice] = useState('')
   323→  const [saving, setSaving] = useState(false)
   324→
   325→  function openCreate() {
   326→    setEditing(null)
   327→    setCategory('GENERAL_SERVICE')
   328→    setTitle('')
   329→    setDescription('')
   330→    setPrice('')
   331→    setDialogOpen(true)
   332→  }
   333→
   334→  function openEdit(s: ServiceRow) {
   335→    setEditing(s)
   336→    setCategory(s.category)
   337→    setTitle(s.title)
   338→    setDescription(s.description ?? '')
   339→    setPrice(s.priceFromMinor ? String(s.priceFromMinor / 100) : '')
   340→    setDialogOpen(true)
   341→  }
   342→
   343→  const priceMinor = price.trim() ? kesToMinor(price) : null
   344→  const priceInvalid = price.trim() !== '' && priceMinor === null
   345→
   346→  function save() {
   347→    if (title.trim().length < 3) { toast.error('Give the service a clear title'); return }
   348→    if (priceInvalid) { toast.error('Price must be a KES amount like 3500 or 3500.50'); return }
   349→    setSaving(true)
   350→    const payload = {
   351→      category,
   352→      title: title.trim(),
   353→      description: description.trim() || null,
   354→      priceFromMinor: priceMinor,
   355→    }
   356→    const call = editing
   357→      ? apiSend(`/api/garage-services?id=${editing.id}`, 'PATCH', payload)
   358→      : apiSend('/api/garage-services', 'POST', { ...payload, orgId })
   359→    call
   360→      .then(() => {
   361→        toast.success(editing ? 'Service updated' : 'Service added')
   362→        setDialogOpen(false)
   363→        onChanged()
   364→      })
   365→      .catch((e) => toast.error(e instanceof Error ? e.message : 'Could not save service'))
   366→      .finally(() => setSaving(false))
   367→  }
   368→
   369→  function toggleActive(s: ServiceRow) {
   370→    apiSend(`/api/garage-services?id=${s.id}`, 'PATCH', { active: !s.active })
   371→      .then(() => { toast.success(s.active ? 'Service deactivated' : 'Service activated'); onChanged() })
   372→      .catch((e) => toast.error(e instanceof Error ? e.message : 'Could not update service'))
   373→  }
   374→
   375→  return (
   376→    <div className="space-y-4">
   377→      <div className="flex flex-wrap items-center justify-between gap-2">
   378→        <p className="text-sm text-muted-foreground">Active services decide which customer requests you see. Deactivate instead of deleting to keep history.</p>
   379→        <Button onClick={openCreate} className="gap-1.5"><Plus className="h-4 w-4" /> Add service</Button>
   380→      </div>
   381→
   382→      {services.length === 0 ? (
   383→        <EmptyBlock icon={<Wrench className="h-6 w-6" />} title="No services yet" desc="Add the categories you cover — e.g. general servicing, tyres, or modifications — so matching requests reach you." />
   384→      ) : (
   385→        <div className="grid gap-3 md:grid-cols-2">
   386→          {services.map((s) => (
   387→            <Card key={s.id} className={`p-4 ${!s.active ? 'opacity-60' : ''}`}>
   388→              <div className="flex items-start justify-between gap-2">
   389→                <div className="min-w-0">
   390→                  <div className="flex flex-wrap items-center gap-2">
   391→                    <h4 className="font-semibold">{s.title}</h4>
   392→                    <Badge variant={s.active ? 'secondary' : 'outline'}>{s.active ? 'Active' : 'Inactive'}</Badge>
   393→                  </div>
   394→                  <p className="mt-0.5 text-xs text-muted-foreground">{SERVICE_CATEGORY_LABELS[s.category] ?? s.category}</p>
   395→                  {s.description && <p className="mt-2 line-clamp-2 text-sm text-muted-foreground">{s.description}</p>}
   396→                  <p className="mt-2 text-sm font-medium">{s.priceFromMinor ? `from ${money(s.priceFromMinor)}` : 'Price on quote'}</p>
   397→                </div>
   398→                <div className="flex shrink-0 flex-col gap-1.5">
   399→                  <Button size="sm" variant="outline" className="gap-1" onClick={() => openEdit(s)}><Pencil className="h-3.5 w-3.5" /> Edit</Button>
   400→                  <Button size="sm" variant="ghost" onClick={() => toggleActive(s)}>{s.active ? 'Deactivate' : 'Activate'}</Button>
   401→                </div>
   402→              </div>
   403→            </Card>
   404→          ))}
   405→        </div>
   406→      )}
   407→
   408→      <Dialog open={dialogOpen} onOpenChange={setDialogOpen}>
   409→        <DialogContent className="max-h-[85vh] overflow-y-auto sm:max-w-lg">
   410→          <DialogHeader>
   411→            <DialogTitle>{editing ? 'Edit service' : 'Add a service'}</DialogTitle>
   412→            <DialogDescription>Shown on your public garage profile. &quot;From&quot; pricing sets customer expectations.</DialogDescription>
   413→          </DialogHeader>
   414→          <div className="space-y-4">

   453→function useOrgRequests(orgId: string) {
   454→  const [feed, setFeed] = useState<{ orgId: string; requests: LeadRequest[]; tick: number } | null>(null)
