   133→export function ProductDetail({ productId }: { productId: string }) {
   134→  const { user } = useAuth()
   135→  const hydrateSaved = useSavedSupply((s) => s.hydrate)
   136→  const [state, setState] = useState<{ key: string; data: ProductResponse } | null>(null)
   137→  const [errState, setErrState] = useState<{ key: string; message: string } | null>(null)
   138→  const [reloadKey, setReloadKey] = useState(0)
   139→  // Derived loading/error keyed to the current request (no sync setState in effects)
   140→  const fetchKey = `${productId}|${reloadKey}`
   141→  const loading = state?.key !== fetchKey
   142→  const error = errState?.key === fetchKey ? errState.message : null
   143→  const data = state?.data ?? null
   144→
   145→  // Deep links skip the landing — make sure saved bookmarks are hydrated.
   146→  useEffect(() => {
   147→    if (user) void hydrateSaved()
   148→  }, [user, hydrateSaved])
   149→
   150→  useEffect(() => {
   151→    let active = true
   152→    const key = `${productId}|${reloadKey}`
   153→    apiGet<ProductResponse>(`/api/supply/products/${encodeURIComponent(productId)}`)
   154→      .then((d) => { if (active) { setState({ key, data: d }); setErrState(null) } })
   155→      .catch((e: Error) => { if (active) setErrState({ key, message: e.message }) })
   156→    return () => { active = false }
   157→  }, [productId, reloadKey])
   158→
   159→  if (loading) return <ProductDetailSkeleton />
   160→  if (error || !data) {
   161→    return (
   162→      <div className="mx-auto w-full max-w-2xl px-4 py-20 text-center">
   163→        <Package className="mx-auto h-10 w-10 text-muted-foreground" aria-hidden="true" />
   164→        <p className="mt-4 font-medium">Could not load this product</p>
   165→        <p className="mt-1 text-sm text-muted-foreground">{error ?? 'It may no longer be listed.'}</p>
   166→        <div className="mt-6 flex justify-center gap-2">
   167→          <Button variant="outline" className="h-11 px-6" onClick={() => setReloadKey((k) => k + 1)}>Retry</Button>
   168→          <Button className="h-11 px-6" onClick={() => navigate('/supply')}>Back to catalogue</Button>
   169→        </div>
   170→      </div>
   171→    )
   172→  }
   173→
   174→  return <ProductDetailBody key={`${data.product.id}|${reloadKey}`} data={data} isOwner={data.isOwner} signedIn={Boolean(user)} onReviewed={() => setReloadKey((k) => k + 1)} />
   175→}
   176→
   177→function ProductDetailBody({ data, isOwner, signedIn, onReviewed }: { data: ProductResponse; isOwner: boolean; signedIn: boolean; onReviewed: () => void }) {
   178→  const p = data.product
   179→  const toggleSaved = useSavedSupply((s) => s.toggle)
   180→  const savedProducts = useSavedSupply((s) => s.products)
   181→  const saved = Boolean(savedProducts[p.id])
   182→
   183→  const [imgIdx, setImgIdx] = useState(0)
   184→  const [imgBroken, setImgBroken] = useState<Record<number, boolean>>({})
   185→  const [qty, setQty] = useState(1)
   186→  const hasWholesale = Boolean(p.wholesalePriceMinor && p.wholesaleMinQty > 0)
   187→  const wholesaleUnlocked = hasWholesale && qty >= (p.wholesaleMinQty ?? 0)
   188→  const maxQty = p.availableQty > 0 ? p.availableQty : 0
   189→
   190→  function addToCart() {
   191→    if (maxQty <= 0) return
   192→    const capped = Math.min(qty, maxQty)
   193→    useCart.getState().add(
   194→      {
   195→        productId: p.id,
   196→        name: p.name,
   197→        sku: p.sku,
   198→        unitPriceMinor: p.unitPriceMinor,
   199→        wholesalePriceMinor: p.wholesalePriceMinor,
   200→        wholesaleMinQty: p.wholesaleMinQty,
   201→        supplierId: p.supplier.id,
   202→        supplierName: p.supplier.name,
   203→        image: p.images[0] ?? null,
   204→        availableQty: p.availableQty,
   205→      },
   206→      capped
   207→    )
   208→    toast.success(`Added ${capped} × ${p.name} to your procurement cart`)
   209→  }
   210→
   211→  function onSave() {
   212→    if (!signedIn) {
   213→      toast.info('Sign in to save parts for later')
   214→      navigate('signin')
   215→      return
   216→    }
   217→    void toggleSaved('PRODUCT', p.id, p.name)
   218→  }
   219→
   220→  const activeImage = p.images[imgIdx] ?? null
   221→
   222→  return (
   223→    <div className="mx-auto w-full max-w-7xl px-4 py-6 sm:px-6">
   224→      {/* Back */}
   225→      <Button variant="ghost" className="mb-4 h-11 gap-1.5 px-2" onClick={() => navigate('/supply')}>
   226→        <ArrowLeft className="h-4 w-4" aria-hidden="true" /> All parts
   227→      </Button>
   228→
   229→      {isOwner && (
   230→        <Alert className="mb-4">
   231→          <Info className="h-4 w-4" aria-hidden="true" />
   232→          <AlertTitle>This is your listing</AlertTitle>
   233→          <AlertDescription>You manage this product from your supplier dashboard.</AlertDescription>
   234→        </Alert>
   235→      )}
   236→
   237→      {/* Top: gallery + purchase panel */}
   238→      <div className="grid gap-6 lg:grid-cols-2">
   239→        {/* Gallery */}
   240→        <div>
   241→          <div className="relative aspect-[4/3] w-full overflow-hidden rounded-xl border border-border bg-muted">
   242→            {activeImage && !imgBroken[imgIdx] ? (
   243→              <img src={activeImage} alt={p.name} className="h-full w-full object-cover" onError={() => setImgBroken((m) => ({ ...m, [imgIdx]: true }))} />
   244→            ) : (
   245→              <span className="flex h-full w-full items-center justify-center text-muted-foreground">
   246→                <Package className="h-12 w-12" aria-hidden="true" />
   247→              </span>
   248→            )}
   249→            {p.featured && <Badge className="absolute left-3 top-3 bg-gold text-gold-foreground">Featured</Badge>}
   250→          </div>
   251→          {p.images.length > 1 && (
   252→            <div className="scrollbar-thin mt-3 flex gap-2 overflow-x-auto pb-1">
   253→              {p.images.map((img, i) => (
   254→                <button
   255→                  key={`${img}-${i}`}
   256→                  onClick={() => setImgIdx(i)}
   257→                  className={`h-16 w-20 shrink-0 overflow-hidden rounded-lg border-2 transition-colors ${i === imgIdx ? 'border-primary' : 'border-transparent hover:border-border'}`}
   258→                  aria-label={`View image ${i + 1} of ${p.name}`}
   259→                  aria-current={i === imgIdx}
   260→                >
   261→                  {img && !imgBroken[i] ? (
   262→                    <img src={img} alt="" className="h-full w-full object-cover" onError={() => setImgBroken((m) => ({ ...m, [i]: true }))} />
   263→                  ) : (
   264→                    <span className="flex h-full items-center justify-center bg-muted text-muted-foreground">
   265→                      <Package className="h-4 w-4" aria-hidden="true" />
   266→                    </span>
   267→                  )}
   268→                </button>
   269→              ))}
   270→            </div>
   271→          )}
   272→        </div>
   273→
   274→        {/* Purchase panel */}
   275→        <div>
   276→          <div className="flex flex-wrap items-center gap-1.5">
   277→            <Badge variant="secondary">{SUPPLY_CONDITION_LABELS[p.condition] ?? p.condition}</Badge>
   278→            {p.category && <Badge variant="outline">{p.category.name}</Badge>}
   279→            {p.authenticityVerified && (
   280→              <Badge variant="outline" className="gap-1 border-primary/40 text-primary">
   281→                <BadgeCheck className="h-3 w-3" aria-hidden="true" /> Authenticity verified
   282→              </Badge>
   283→            )}
   284→          </div>
   285→
   286→          <h1 className="mt-3 text-2xl font-bold tracking-tight sm:text-3xl">{p.name}</h1>
   287→          <p className="mt-1 text-sm text-muted-foreground">
   288→            {p.brand ? `${p.brand} · ` : ''}SKU {p.sku}
   289→            {p.mpn ? ` · MPN ${p.mpn}` : ''}
   290→            {' '}· Ref {p.refCode}
   291→          </p>
   292→
   293→          {(p.ratingCount > 0 && p.ratingAvg != null) ? (
   294→            <div className="mt-2 flex items-center gap-1.5">
   295→              <Stars avg={p.ratingAvg} size="md" />
   296→              <span className="text-sm text-muted-foreground">{p.ratingAvg.toFixed(1)} · {p.ratingCount} review{p.ratingCount === 1 ? '' : 's'}</span>
   297→            </div>
   298→          ) : (
   299→            <p className="mt-2 flex items-center gap-1 text-sm text-muted-foreground">
   300→              <Star className="h-3.5 w-3.5" aria-hidden="true" /> No reviews yet
   301→            </p>
   302→          )}
   303→
   304→          {/* Price */}
   305→          <div className="mt-4 rounded-xl border border-border bg-card p-4">
   306→            <p className="text-3xl font-bold tracking-tight">{money(p.unitPriceMinor, p.currency)}</p>
   307→            {p.taxRatePct > 0 && (
   308→              <p className="mt-0.5 text-xs text-muted-foreground">Tax rate {p.taxRatePct}% where applicable</p>
   309→            )}
   310→            {hasWholesale && (
   311→              <p className="mt-1 text-sm text-muted-foreground">
   312→                Wholesale <span className="font-semibold text-foreground">{money(p.wholesalePriceMinor!, p.currency)}</span>{' '}
   313→                per unit at {p.wholesaleMinQty}+ units
   314→              </p>
   315→            )}
   316→            <div className="mt-2"><StockStatus qty={p.availableQty} /></div>
   317→
   318→            {/* Qty stepper */}
   319→            <div className="mt-4 flex flex-wrap items-center gap-3">
   320→              <div className="flex items-center rounded-lg border border-border" role="group" aria-label="Quantity">
   321→                <button
   322→                  className="flex h-11 w-11 items-center justify-center rounded-l-lg hover:bg-muted disabled:opacity-40"
   323→                  onClick={() => setQty((q) => Math.max(1, q - 1))}
   324→                  disabled={qty <= 1}
   325→                  aria-label="Decrease quantity"
   326→                >
   327→                  <Minus className="h-4 w-4" aria-hidden="true" />
   328→                </button>
   329→                <Input
   330→                  className="h-11 w-14 rounded-none border-x border-y-0 text-center [appearance:textfield] [&::-webkit-inner-spin-button]:appearance-none [&::-webkit-outer-spin-button]:appearance-none"
   331→                  type="number"
   332→                  min={1}
   333→                  max={maxQty || undefined}
   334→                  value={qty}
   335→                  onChange={(e) => {
   336→                    const n = Number(e.target.value)
   337→                    if (Number.isNaN(n)) return
   338→                    setQty(Math.max(1, maxQty > 0 ? Math.min(n, maxQty) : n))
   339→                  }}
   340→                  aria-label="Quantity"
   341→                />
   342→                <button
   343→                  className="flex h-11 w-11 items-center justify-center rounded-r-lg hover:bg-muted disabled:opacity-40"
   344→                  onClick={() => setQty((q) => (maxQty > 0 ? Math.min(maxQty, q + 1) : q + 1))}
   345→                  disabled={maxQty > 0 && qty >= maxQty}
   346→                  aria-label="Increase quantity"
   347→                >
   348→                  <Plus className="h-4 w-4" aria-hidden="true" />
   349→                </button>
   350→              </div>
   351→              <Button className="h-11 flex-1 gap-2" onClick={addToCart} disabled={maxQty <= 0}>
   352→                <ShoppingCart className="h-4 w-4" aria-hidden="true" /> Add to cart
   353→              </Button>
   354→              <Button
   355→                variant="outline"
   356→                size="icon"
   357→                className="h-11 w-11"
   358→                onClick={onSave}
   359→                aria-label={saved ? 'Remove from saved' : 'Save product'}
   360→              >
   361→                {saved ? <BookmarkCheck className="h-4 w-4 text-primary" aria-hidden="true" /> : <Bookmark className="h-4 w-4" aria-hidden="true" />}
   362→              </Button>
   363→            </div>
   364→            {hasWholesale && (
   365→              <p className={`mt-2 text-xs ${wholesaleUnlocked ? 'font-medium text-primary' : 'text-muted-foreground'}`}>
   366→                {wholesaleUnlocked
   367→                  ? `Wholesale price unlocked at qty ${p.wholesaleMinQty} — you save ${money((p.unitPriceMinor - (p.wholesalePriceMinor ?? 0)) * qty, p.currency)} on this line`
   368→                  : `Add ${p.wholesaleMinQty - qty} more to unlock wholesale pricing (min qty ${p.wholesaleMinQty})`}
   369→              </p>
   370→            )}
   371→          </div>
   372→
   373→          {/* Supplier card */}
   374→          <Card className="mt-4">
   375→            <CardContent className="flex items-start gap-3 p-4">
   376→              <SupplierAvatar name={p.supplier.name} logoUrl={p.supplier.logoUrl} className="h-12 w-12" />
   377→              <div className="min-w-0 flex-1">
   378→                <div className="flex flex-wrap items-center gap-2">
   379→                  <button className="truncate font-semibold hover:underline" onClick={() => navigate(`/supply?supplier=${p.supplier.id}`)}>
   380→                    {p.supplier.name}
   381→                  </button>
   382→                  <VerifiedBadge verified={p.supplier.verified} />
   383→                </div>
   384→                <p className="mt-0.5 flex items-center gap-1 text-xs text-muted-foreground">
   385→                  <MapPin className="h-3.5 w-3.5 shrink-0" aria-hidden="true" />
   386→                  {[p.supplier.town, p.supplier.county].filter(Boolean).join(', ') || p.supplier.county}
   387→                </p>
   388→                {/* Contact details are only shown for verified suppliers */}
   389→                {p.supplier.verified && (p.supplier.phone || p.supplier.email) && (
   390→                  <p className="mt-1 text-xs text-muted-foreground">
   391→                    {[p.supplier.phone, p.supplier.email].filter(Boolean).join(' · ')}
   392→                  </p>
   393→                )}
   394→                {p.supplier.supplierRating != null && p.supplier.supplierRating > 0 && (
   395→                  <div className="mt-1 flex items-center gap-1.5">
   396→                    <Stars avg={p.supplier.supplierRating} />
   397→                    <span className="text-xs text-muted-foreground">{p.supplier.supplierRating.toFixed(1)} supplier rating</span>
   398→                  </div>
   399→                )}
   400→              </div>
   401→              <Button variant="outline" size="sm" className="h-10 shrink-0" onClick={() => navigate(`/supply?supplier=${p.supplier.id}`)}>
   402→                Storefront
   403→              </Button>
   404→            </CardContent>
   405→          </Card>
   406→
   407→          {/* Per-warehouse availability */}
   408→          <Card className="mt-4">
   409→            <CardContent className="p-4">
   410→              <h2 className="flex items-center gap-1.5 text-sm font-semibold">
   411→                <Warehouse className="h-4 w-4 text-muted-foreground" aria-hidden="true" /> Availability by warehouse
   412→              </h2>
   413→              {p.stocks.length === 0 ? (
   414→                <p className="mt-2 text-sm text-muted-foreground">No stock is currently held in any warehouse.</p>
   415→              ) : (
   416→                <ul className="scrollbar-thin mt-2 max-h-44 space-y-1.5 overflow-y-auto pr-1">
   417→                  {p.stocks.map((s) => (
   418→                    <li key={s.branchId} className="flex items-center justify-between gap-2 rounded-md bg-muted/60 px-2.5 py-2 text-sm">
   419→                      <span className="min-w-0">
   420→                        <span className="block truncate font-medium">{s.branchName}</span>
   421→                        <span className="text-xs text-muted-foreground">
   422→                          {s.kind === 'WAREHOUSE' ? 'Warehouse' : 'Branch'} · {[s.town, s.county].filter(Boolean).join(', ')}
   423→                        </span>
   424→                      </span>
   425→                      <span className={`shrink-0 text-sm font-semibold ${s.available > 0 ? 'text-primary' : 'text-destructive'}`}>
   426→                        {s.available} {s.available === 1 ? 'unit' : 'units'}
   427→                      </span>
   428→                    </li>
   429→                  ))}
   430→                </ul>
   431→              )}
   432→            </CardContent>
   433→          </Card>
   434→        </div>
   435→      </div>
   436→
   437→      {/* Description */}
   438→      <section className="mt-8" aria-label="Product description">
   439→        <h2 className="text-lg font-bold">About this part</h2>
   440→        <p className="mt-2 whitespace-pre-line text-sm leading-relaxed text-foreground/90">{p.description}</p>
   441→        {p.tags.length > 0 && (
   442→          <div className="mt-3 flex flex-wrap gap-1.5">
   443→            {p.tags.map((t) => <Badge key={t} variant="secondary">{t}</Badge>)}
   444→          </div>
   445→        )}
   446→      </section>
   447→
   448→      {/* Provenance */}
   449→      {p.provenance?.provenanceNote && (
   450→        <Alert className="mt-6">
   451→          <ShieldCheck className="h-4 w-4" aria-hidden="true" />
   452→          <AlertTitle>Authenticity &amp; provenance</AlertTitle>
   453→          <AlertDescription>{p.provenance.provenanceNote}</AlertDescription>
   454→        </Alert>
   455→      )}
   456→
   457→      {/* Spec table */}
   458→      <section className="mt-8" aria-label="Specifications">
   459→        <h2 className="text-lg font-bold">Specifications</h2>
   460→        <div className="mt-3 overflow-hidden rounded-xl border border-border">
   461→          <Table>
   462→            <TableHeader>
   463→              <TableRow>
   464→                <TableHead className="w-1/3">Attribute</TableHead>
   465→                <TableHead>Value</TableHead>
   466→              </TableRow>
   467→            </TableHeader>
   468→            <TableBody>
   469→              <SpecRow label="SKU" value={p.sku} mono />
   470→              <SpecRow label="MPN (manufacturer part no.)" value={p.mpn ?? '—'} mono />
   471→              <SpecRow label="OEM reference" value={p.oemRef ?? '—'} mono />
   472→              <SpecRow label="Brand" value={p.brand ?? '—'} />
   473→              <SpecRow label="Condition" value={SUPPLY_CONDITION_LABELS[p.condition] ?? p.condition} />
   474→              <SpecRow label="Warranty" value={p.warrantyMonths > 0 ? `${p.warrantyMonths} month${p.warrantyMonths === 1 ? '' : 's'}` : 'None offered'} />

   477→              <SpecRow label="Serial tracked" value={p.serialTracked ? 'Yes' : 'No'} />

   480→            </TableBody>

   487→        <h2 className="text-lg font-bold">Vehicle compatibility</h2>

   512→                    <TableCell className="font-medium">{row.make}</TableCell>
   513→                    <TableCell>{row.model ?? 'Any'}</TableCell>

   534→        {p.vehicleMakes.length > 0 && (

   536→            <p className="text-xs text-muted-foreground">
   537→              Brand-level quick filter (indicative only — the table above is authoritative):

   549→          <History className="h-4 w-4 text-muted-foreground" aria-hidden="true" /> Price history

   554→          <ul className="scrollbar-thin mt-3 max-h-96 space-y-1.5 overflow-y-auto pr-1">

   585→function ReviewsSection({
   586→  productId,
   587→  reviews,
   588→  ratingAvg,
   589→  ratingCount,
   590→  signedIn,
   591→  onReviewed,
   592→}: {
   593→  productId: string
   594→  reviews: DetailReview[]
   595→  ratingAvg: number | null
   596→  ratingCount: number
   597→  signedIn: boolean
   598→  onReviewed: () => void
   599→}) {
   600→  const [rating, setRating] = useState(0)
   601→  const [comment, setComment] = useState('')
   602→  const [busy, setBusy] = useState(false)
   603→
   604→  function submit() {
   605→    if (rating < 1) {
   606→      toast.error('Pick a star rating first')
   607→      return
   608→    }
   609→    setBusy(true)
   610→    apiSend<{ message: string; verifiedPurchase: boolean }>('/api/supply/reviews', 'POST', {
   611→      productId,
   612→      rating,
   613→      comment: comment.trim() || undefined,
   614→    })
   615→      .then((d) => {
   616→        toast.success(d.verifiedPurchase ? 'Review recorded — marked as a verified purchase' : 'Review recorded. Thank you!')
   617→        setRating(0)
   618→        setComment('')
   619→        onReviewed()
   620→      })
   621→      .catch((e) => {
   622→        const status = (e as { status?: number }).status
   623→        if (status === 409) toast.info('You already reviewed this product')
   624→        else toast.error(e instanceof Error ? e.message : 'Could not record review')
   625→      })
   626→      .finally(() => setBusy(false))
   627→  }
   628→
   629→  return (
   630→    <section aria-label="Product reviews">
   631→      <div className="flex flex-wrap items-center gap-2">
   632→        <h2 className="text-lg font-bold">Reviews</h2>
   633→        {ratingCount > 0 && ratingAvg != null && (
   634→          <span className="flex items-center gap-1.5 text-sm text-muted-foreground">
   635→            <Stars avg={ratingAvg} /> {ratingAvg.toFixed(1)} from {ratingCount} review{ratingCount === 1 ? '' : 's'}
   636→          </span>
   637→        )}
   638→      </div>
   639→
   640→      {reviews.length === 0 ? (
   641→        <p className="mt-3 text-sm text-muted-foreground">No reviews yet — be the first to rate this part.</p>
   642→      ) : (
   643→        <ul className="mt-3 space-y-3">
   644→          {reviews.map((r) => (
   645→            <li key={r.id} className="rounded-xl border border-border p-4">
   646→              <div className="flex flex-wrap items-center gap-2">
   647→                <Stars avg={r.rating} />
   648→                <span className="text-sm font-medium">{r.author}</span>
   649→                <span className="text-xs text-muted-foreground">{timeAgo(r.createdAt)}</span>
   650→                {r.verifiedPurchase && (
   651→                  <Badge variant="outline" className="gap-1 border-primary/40 text-primary">
   652→                    <BadgeCheck className="h-3 w-3" aria-hidden="true" /> Verified purchase
   653→                  </Badge>
   654→                )}
   655→              </div>
   656→              {r.comment && <p className="mt-2 text-sm text-foreground/90">{r.comment}</p>}
   657→            </li>
   658→          ))}
   659→        </ul>
   660→      )}
   661→
   662→      {signedIn ? (
   663→        <div className="mt-5 rounded-xl border border-border p-4">
   664→          <p className="text-sm font-semibold">Write a review</p>
   665→          <div className="mt-2 flex items-center gap-1" role="radiogroup" aria-label="Your star rating">
   666→            {[1, 2, 3, 4, 5].map((i) => (
   667→              <button
   668→                key={i}
   669→                role="radio"
   670→                aria-checked={rating === i}
   671→                aria-label={`${i} star${i === 1 ? '' : 's'}`}
   672→                className="rounded p-1 hover:bg-muted"
   673→                onClick={() => setRating(i)}
   674→              >
   675→                <Star className={`h-6 w-6 ${i <= rating ? 'fill-gold text-gold' : 'text-muted-foreground/50'}`} aria-hidden="true" />
   676→              </button>
   677→            ))}
   678→          </div>
   679→          <Textarea
   680→            className="mt-3"
   681→            rows={3}
   682→            value={comment}
   683→            onChange={(e) => setComment(e.target.value)}
   684→            placeholder="How was the part, packaging, delivery? (optional)"
   685→            maxLength={1000}
   686→          />
   687→          <Button className="mt-3 h-11" onClick={submit} disabled={busy || rating < 1}>
   688→            {busy ? 'Submitting…' : 'Submit review'}
   689→          </Button>
   690→        </div>
   691→      ) : (
   692→        <div className="mt-5 flex flex-wrap items-center justify-between gap-3 rounded-xl border border-dashed p-4">
   693→          <p className="text-sm text-muted-foreground">Bought this part? Sign in to leave a review.</p>
   694→          <Button variant="outline" className="h-11" onClick={() => navigate('signin')}>Sign in to review</Button>
   695→        </div>
   696→      )}
   697→    </section>
   698→  )
   699→}
