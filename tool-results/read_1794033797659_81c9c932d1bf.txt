   120→   120→/** Resolve the demo-ecosystem subject of a supply-console action from the
   121→   121→ * database (owner user or owning org). Returns null when the record does not
   122→   122→ * exist: demo sessions are then blocked rather than trusted. */
   123→   123→async function supplyActionSubject(body: { action: string; orgId?: string; applicationId?: string; productId?: string; disputeId?: string; orderId?: string; refundId?: string }): Promise<{ ownerId?: string | null; orgId?: string | null } | null> {
   124→   124→  switch (body.action) {
   125→   125→    case 'verify-supplier':
   126→   126→    case 'unverify-supplier':
   127→   127→    case 'suspend-supplier':
   128→   128→    case 'restore-supplier':
   129→   129→    case 'risk-score':
   130→   130→      return { orgId: body.orgId ?? null }
   131→   131→    case 'approve-application':
   132→   132→    case 'reject-application': {
   133→   133→      const app = body.applicationId ? await db.application.findUnique({ where: { id: body.applicationId }, select: { userId: true } }) : null
   134→   134→      return app ? { ownerId: app.userId } : null
   135→   135→    }
   136→   136→    case 'verify-authenticity':
   137→   137→    case 'revoke-authenticity':
   138→   138→    case 'suspend-product':
   139→   139→    case 'republish-product': {
   140→   140→      const product = body.productId ? await db.product.findUnique({ where: { id: body.productId }, select: { supplierOrgId: true } }) : null
   141→   141→      return product ? { orgId: product.supplierOrgId } : null
   142→   142→    }
   143→   143→    case 'dispute-note':
   144→   144→    case 'dispute-status':
   145→   145→    case 'resolve-dispute': {
   146→   146→      const dispute = body.disputeId ? await db.dispute.findUnique({ where: { id: body.disputeId }, select: { orderId: true, customerOrderId: true, order: { select: { garageOrgId: true, supplierOrgId: true } }, customerOrder: { select: { buyerUserId: true, supplierOrgId: true } } } }) : null
   147→   147→      if (!dispute) return null
   148→   148→      if (dispute.order) return { orgId: dispute.order.garageOrgId || dispute.order.supplierOrgId }
   149→   149→      if (dispute.customerOrder) return { ownerId: dispute.customerOrder.buyerUserId, orgId: dispute.customerOrder.supplierOrgId }
   150→   150→      return null
   151→   151→    }
   152→   152→    case 'reconcile-payment': {
   153→   153→      const order = body.orderId ? await db.purchaseOrder.findUnique({ where: { id: body.orderId }, select: { garageOrgId: true, supplierOrgId: true } }) : null
   154→   154→      return order ? { orgId: order.garageOrgId || order.supplierOrgId } : null
   155→   155→    }
   156→   156→    case 'process-refund': {
   157→   157→      const refund = body.refundId ? await db.refund.findUnique({ where: { id: body.refundId }, include: { order: { select: { garageOrgId: true, supplierOrgId: true } } } }) : null
   158→   158→      return refund?.order ? { orgId: refund.order.garageOrgId || refund.order.supplierOrgId } : null
   159→   159→    }
   160→   160→    default:
   161→   161→      return null
   162→   162→  }
   163→   163→}

   165→   165→/** POST: all sensitive actions are logged to the audit trail. */
   166→   166→export const POST = route(async (req: NextRequest) => {
   167→   167→  const user = await requireRole(ROLE.SUPPORT, ROLE.REVIEWER, ROLE.MODERATOR, ROLE.FINANCE, ROLE.ADMIN, ROLE.FOUNDER)
   168→   168→  const body = await parseBody(req, actionSchema)
   169→   169→  const staffSide = 'ADMIN' as const
   170→   170→
   171→   171→  // FINANCE SPLIT: reconciling a payment marks an invoice PAID and processing
   172→   172→  // a refund moves money state: restricted to FINANCE/ADMIN/FOUNDER instead
   173→   173→  // of every staff role.
   174→   174→  if ((body.action === 'reconcile-payment' || body.action === 'process-refund') && !hasRole(user, ROLE.FINANCE, ROLE.ADMIN, ROLE.FOUNDER)) {
   175→   175→    return fail(403, 'FINANCE_ONLY', 'Payment reconciliation and refunds are restricted to the Finance department.')
   176→   176→  }
   177→   177→
   178→   178→  // DEMO SANDBOX: demo staff may only run console actions whose subject
   179→   179→  // belongs to the demo ecosystem. The subject is resolved from the database,
   180→   180→  // never from client-supplied ownership claims.
   181→   181→  if (user.isDemo) {
   182→   182→    const subject = await supplyActionSubject(body)
   183→   183→    if (!subject) return fail(403, 'DEMO_SANDBOXED', 'Demo mode: the record for this action could not be resolved, so it has been blocked.')
   184→   184→    await assertDemoResourceGuard(user, subject)
   185→   185→  }
   186→   186→
   187→   187→  switch (body.action) {
   188→   188→    case 'verify-supplier': {
   189→   189→      const org = await db.org.findFirst({ where: { id: body.orgId, type: 'SUPPLIER' } })
   190→   190→      if (!org) return fail(404, 'NOT_FOUND', 'Supplier not found.')
   191→   191→      if (org.verified) return fail(409, 'ALREADY_VERIFIED', 'Supplier is already verified.')
   192→   192→      await db.org.update({ where: { id: org.id }, data: { verified: true } })
   193→   193→      const members = await db.orgMember.findMany({ where: { orgId: org.id }, select: { userId: true } })
   194→   194→      await notifyMany(members.map((m) => m.userId), { type: 'SUPPLIER_VERIFIED', title: 'Supplier verification approved', body: 'Your business is now verified on the Automotive Supply Network.', link: '/supplier' })
   195→   195→      await audit({ actorId: user.id, actorName: user.name, action: 'SUPPLIER_VERIFIED', targetType: 'ORG', targetId: org.id, metadata: { note: body.note } })
   196→   196→      return ok({ message: `${org.name} is now verified.` })
   197→   197→    }
   198→   198→    case 'unverify-supplier': {
   199→   199→      const org = await db.org.findFirst({ where: { id: body.orgId, type: 'SUPPLIER' } })
   200→   200→      if (!org) return fail(404, 'NOT_FOUND', 'Supplier not found.')
   201→   201→      await db.org.update({ where: { id: org.id }, data: { verified: false } })
   202→   202→      await audit({ actorId: user.id, actorName: user.name, action: 'SUPPLIER_VERIFICATION_REVOKED', targetType: 'ORG', targetId: org.id, metadata: { note: body.note } })
   203→   203→      return ok({ message: 'Verification revoked.' })
   204→   204→    }
   205→   205→    case 'suspend-supplier': {
   206→   206→      const org = await db.org.findFirst({ where: { id: body.orgId, type: 'SUPPLIER' } })
   207→   207→      if (!org) return fail(404, 'NOT_FOUND', 'Supplier not found.')
   208→   208→      await db.org.update({ where: { id: org.id }, data: { status: 'SUSPENDED', verified: false } })
   209→   209→      await db.product.updateMany({ where: { supplierOrgId: org.id, status: 'PUBLISHED' }, data: { status: 'SUSPENDED' } })
   210→   210→      await audit({ actorId: user.id, actorName: user.name, action: 'SUPPLIER_SUSPENDED', targetType: 'ORG', targetId: org.id, metadata: { note: body.note } })
   211→   211→      return ok({ message: `${org.name} suspended and delisted.` })
   212→   212→    }
   213→   213→    case 'restore-supplier': {
   214→   214→      const org = await db.org.findFirst({ where: { id: body.orgId, type: 'SUPPLIER' } })
   215→   215→      if (!org) return fail(404, 'NOT_FOUND', 'Supplier not found.')
   216→   216→      await db.org.update({ where: { id: org.id }, data: { status: 'ACTIVE' } })
   217→   217→      await audit({ actorId: user.id, actorName: user.name, action: 'SUPPLIER_RESTORED', targetType: 'ORG', targetId: org.id, metadata: { note: body.note } })
   218→   218→      return ok({ message: 'Supplier restored (products require manual republish review).' })
   219→   219→    }
   220→   220→    case 'risk-score': {
   221→   221→      await db.org.update({ where: { id: body.orgId }, data: { supplierRiskScore: body.score } })
   222→   222→      await audit({ actorId: user.id, actorName: user.name, action: 'SUPPLIER_RISK_UPDATED', targetType: 'ORG', targetId: body.orgId, metadata: { score: body.score, note: body.note } })
   223→   223→      return ok({ message: `Risk score set to ${body.score}.` })
   224→   224→    }
   225→   225→    case 'approve-application': {
   226→   226→      const app = await db.application.findUnique({ where: { id: body.applicationId } })
   227→   227→      if (!app || app.type !== 'SUPPLIER') return fail(404, 'NOT_FOUND', 'Application not found.')
   228→   228→      if (app.orgId) return fail(409, 'ALREADY_APPROVED', 'Application already approved.')
   229→   229→      let org = await db.org.findFirst({ where: { name: app.orgName, type: 'SUPPLIER' } })
   230→   230→      if (!org) {
   231→   231→        org = await db.org.create({
   232→   232→          data: {
   233→   233→            type: 'SUPPLIER', name: app.orgName, regNo: app.regNo, county: app.county, town: app.town,
   234→   234→            location: app.physicalAddress, phone: app.phone, email: app.email,
   235→   235→            description: app.notes, verified: false,
   236→   236→            specialities: app.categories,
   237→   237→          },
   238→   238→        })
   239→   239→      }
   240→   240→      await db.orgMember.upsert({
   241→   241→        where: { orgId_userId: { orgId: org.id, userId: app.userId } },
   242→   242→        create: { orgId: org.id, userId: app.userId, role: 'OWNER' },
   243→   243→        update: {},
   244→   244→      })
   245→   245→      const buyer = await db.user.findUnique({ where: { id: app.userId } })
   246→   246→      if (buyer && !buyer.roles.split(',').includes('SUPPLIER_OWNER')) {
   247→   247→        await db.user.update({ where: { id: buyer.id }, data: { roles: `${buyer.roles},SUPPLIER_OWNER` } })
   248→   248→      }
   249→   249→      await db.application.update({
   250→   250→        where: { id: app.id },
   251→   251→        data: { status: 'APPROVED', orgId: org.id, decisionNote: body.note ?? null, events: { create: { type: 'APPROVED', actorId: user.id, actorName: user.name, note: body.note ?? 'Documents reviewed. Business account created: verification (listing) follows separately.' } } },
   252→   252→      })
   253→   253→      await notify({ userId: app.userId, type: 'SUPPLIER_APPROVED', title: 'Supplier application approved', body: `${app.orgName} is registered. Add your warehouses and products in the Supplier Portal. Verification status shows once platform review completes.`, link: '/supplier' })
   254→   254→      await audit({ actorId: user.id, actorName: user.name, action: 'SUPPLIER_APPLICATION_APPROVED', targetType: 'APPLICATION', targetId: app.id, metadata: { orgId: org.id } })
   255→   255→      return ok({ message: 'Application approved. Supplier business account created (unverified until listing review).' })
   256→   256→    }
   257→   257→    case 'reject-application': {
   258→   258→      const app = await db.application.findUnique({ where: { id: body.applicationId } })
   259→   259→      if (!app || app.type !== 'SUPPLIER') return fail(404, 'NOT_FOUND', 'Application not found.')
   260→   260→      await db.application.update({
   261→   261→        where: { id: app.id },
   262→   262→        data: { status: 'REJECTED', decisionNote: body.note, events: { create: { type: 'REJECTED', actorId: user.id, actorName: user.name, note: body.note } } },
   263→   263→      })
   264→   264→      await notify({ userId: app.userId, type: 'SUPPLIER_REJECTED', title: `Application ${app.refCode} rejected`, body: body.note, link: '/supply' })
   265→   265→      await audit({ actorId: user.id, actorName: user.name, action: 'SUPPLIER_APPLICATION_REJECTED', targetType: 'APPLICATION', targetId: app.id, metadata: { note: body.note } })
   266→   266→      return ok({ message: 'Application rejected.' })
   267→   267→    }
   268→   268→    case 'verify-authenticity': {
   269→   269→      const product = await db.product.findUnique({ where: { id: body.productId } })

   271→   271→      await db.product.update({
   272→   272→        where: { id: product.id },
   273→   273→        data: {
   274→   274→          authenticityVerified: true,
   275→   275→          provenance: JSON.stringify({ ...(jsonParse<Record<string, unknown>>(product.provenance, {})), authenticator: 'ADMIN', verifiedAt: new Date().toISOString(), note: body.note }),
   276→   276→        },
   277→   277→      })
   278→   278→      await audit({ actorId: user.id, actorName: user.name, action: 'PRODUCT_AUTHENTICITY_VERIFIED', targetType: 'PRODUCT', targetId: product.id, metadata: { note: body.note } })
   279→   279→      return ok({ message: 'Product authenticity independently verified by CARMAZE.' })
   280→   280→    }
   281→   281→    case 'revoke-authenticity': {
   282→   282→      const product = await db.product.findUnique({ where: { id: body.productId } })
   283→   283→      if (!product) return fail(404, 'NOT_FOUND', 'Product not found.')
   284→   284→      await db.product.update({ where: { id: product.id }, data: { authenticityVerified: false } })
   285→   285→      await audit({ actorId: user.id, actorName: user.name, action: 'PRODUCT_AUTHENTICITY_REVOKED', targetType: 'PRODUCT', targetId: product.id, metadata: { note: body.note } })
   286→   286→      return ok({ message: 'Authenticity mark removed.' })
   287→   287→    }
   288→   288→    case 'suspend-product': {

   290→   290→      await audit({ actorId: user.id, actorName: user.name, action: 'PRODUCT_SUSPENDED', targetType: 'PRODUCT', targetId: body.productId, metadata: { note: body.note } })

   292→   292→    }

   299→   299→      const dispute = await db.dispute.findUnique({ where: { id: body.disputeId } })
   300→   300→      if (!dispute) return fail(404, 'NOT_FOUND', 'Dispute not found.')
   301→   301→      await db.disputeEvent.create({ data: { disputeId: dispute.id, type: 'NOTE', actorId: user.id, actorName: user.name, note: body.note } })
   302→   302→      await audit({ actorId: user.id, actorName: user.name, action: 'DISPUTE_NOTE', targetType: 'DISPUTE', targetId: dispute.id })
   303→   303→      return ok({ message: 'Note added to the dispute file.' })
   304→   304→    }
   305→   305→    case 'dispute-status': {
   306→   306→      await db.dispute.update({ where: { id: body.disputeId }, data: { status: body.status } })
   307→   307→      await db.disputeEvent.create({ data: { disputeId: body.disputeId, type: 'STATUS_CHANGE', actorId: user.id, actorName: user.name, note: `Status set to ${body.status}` } })
   308→   308→      await audit({ actorId: user.id, actorName: user.name, action: 'DISPUTE_STATUS_CHANGED', targetType: 'DISPUTE', targetId: body.disputeId, metadata: { status: body.status } })
   309→   309→      return ok({ message: 'Dispute status updated.' })
   310→   310→    }
   311→   311→    case 'resolve-dispute': {
   312→   312→      const dispute = await db.dispute.findUnique({ where: { id: body.disputeId }, include: { order: true, customerOrder: { include: { supplier: { select: { id: true, name: true } } } } } })
   313→   313→      if (!dispute) return fail(404, 'NOT_FOUND', 'Dispute not found.')
   314→   314→      if (dispute.status === 'RESOLVED' || dispute.status === 'CLOSED') return fail(409, 'ALREADY_RESOLVED', 'Dispute already resolved.')
   315→   315→
   316→   316→      // ── Consumer (B2C) parts-order dispute ──
   317→   317→      if (dispute.customerOrderId) {
   318→   318→        const order = dispute.customerOrder
   319→   319→        if (!order) return fail(404, 'NOT_FOUND', 'Dispute has no linked consumer order.')
   320→   320→        await db.dispute.update({
   321→   321→          where: { id: dispute.id },
   322→   322→          data: { status: 'RESOLVED', resolution: body.resolution, resolutionNote: body.note, resolvedBy: user.name, resolvedAt: new Date() },
   323→   323→        })
   324→   324→        await db.disputeEvent.create({ data: { disputeId: dispute.id, type: 'RESOLUTION', actorId: user.id, actorName: user.name, note: `${body.resolution}: ${body.note}` } })
   325→   325→        if (body.resolution === 'REJECTED') {
   326→   326→          // Complaint not upheld: the delivered order stands as completed.
   327→   327→          await db.customerOrder.update({ where: { id: order.id }, data: { status: 'COMPLETED', completedAt: order.completedAt ?? new Date() } })
   328→   328→          await db.customerOrderEvent.create({ data: { orderId: order.id, type: 'NOTE', actorSide: 'SYSTEM', actorName: user.name, note: `Dispute ${dispute.refCode} not upheld: ${body.note}` } })
   329→   329→        } else if (body.resolution === 'REPLACEMENT') {
   330→   330→          // Re-enters fulfilment: the supplier re-dispatches, the buyer confirms receipt again.
   331→   331→          await db.customerOrder.update({ where: { id: order.id }, data: { status: 'CONFIRMED' } })
   332→   332→          await db.customerOrderEvent.create({ data: { orderId: order.id, type: 'NOTE', actorSide: 'SYSTEM', actorName: user.name, note: `Dispute ${dispute.refCode} resolved by replacement. The supplier re-dispatches; confirm receipt when it arrives.` } })
   333→   333→        }
   334→   334→        // REFUND_FULL / REFUND_PARTIAL: the decision is recorded; the money
   335→   335→        // movement itself happens when FINANCE refunds the linked invoice via
   336→   336→        // the invoice refund endpoint (which completes the order). We never
   337→   337→        // mark a refund as done here.
   338→   338→        await notify({
   339→   339→          userId: order.buyerUserId,
   340→   340→          type: 'DISPUTE_RESOLVED',
   341→   341→          title: `Dispute ${dispute.refCode} resolved: ${body.resolution.replaceAll('_', ' ').toLowerCase()}`,
   342→   342→          body: body.note,
   343→   343→          link: '/account?tab=orders',
   344→   344→        })
   345→   345→        await notifyMany((await db.orgMember.findMany({ where: { orgId: order.supplierOrgId }, select: { userId: true } })).map((m) => m.userId), {
   346→   346→          type: 'DISPUTE_RESOLVED',
   347→   347→          title: `Dispute ${dispute.refCode} resolved`,
   348→   348→          body: `${body.resolution}: ${body.note}`,
   349→   349→          link: '/supplier?tab=consumer-orders',
   350→   350→        })
   351→   351→        await audit({ actorId: user.id, actorName: user.name, action: 'DISPUTE_RESOLVED', targetType: 'DISPUTE', targetId: dispute.id, metadata: { resolution: body.resolution, note: body.note, customerOrder: order.refCode } })
   352→   352→        return ok({ message: `Dispute resolved (${body.resolution}).${body.resolution.includes('REFUND') ? ' Finance will process the refund on the order invoice.' : ''}` })
   353→   353→      }
   354→   354→
   355→   355→      // ── Garage↔Supplier purchase-order dispute ──
   356→   356→      if (!dispute.order) return fail(404, 'NOT_FOUND', 'Dispute has no linked order.')
   357→   357→      await db.dispute.update({
   358→   358→        where: { id: dispute.id },
   359→   359→        data: { status: 'RESOLVED', resolution: body.resolution, resolutionNote: body.note, resolvedBy: user.name, resolvedAt: new Date() },
   360→   360→      })
   361→   361→      await db.disputeEvent.create({ data: { disputeId: dispute.id, type: 'RESOLUTION', actorId: user.id, actorName: user.name, note: `${body.resolution}: ${body.note}` } })
   362→   362→      if (body.resolution === 'REFUND_FULL' || body.resolution === 'REFUND_PARTIAL') {
   363→   363→        const amount = body.resolution === 'REFUND_FULL' ? dispute.order.totalMinor : (body.refundAmountMinor ?? 0)
   364→   364→        if (amount > 0) {
   365→   365→          await db.refund.create({
   366→   366→            data: { refCode: refCode('RFD', 6), orderId: dispute.orderId!, paymentAuthId: (await db.paymentAuthorization.findUnique({ where: { orderId: dispute.orderId! } }))?.id, amountMinor: amount, kind: body.resolution === 'REFUND_FULL' ? 'FULL' : 'PARTIAL', reason: `Dispute ${dispute.refCode}: ${body.note.slice(0, 200)}`, status: 'REQUESTED', requestedById: user.id },
   367→   367→          })
   368→   368→        }
   369→   369→        if (dispute.order.paymentStatus === 'PROTECTED') {
   370→   370→          await db.purchaseOrder.update({ where: { id: dispute.orderId! }, data: { paymentStatus: body.resolution === 'REFUND_FULL' ? 'REFUNDED' : 'PARTIALLY_RELEASED' } })
   371→   371→        }
   372→   372→      }
   373→   373→      await db.purchaseOrder.update({ where: { id: dispute.orderId! }, data: { status: body.resolution === 'REJECTED' ? 'COMPLETED' : 'INSPECTED' } })
   374→   374→      await audit({ actorId: user.id, actorName: user.name, action: 'DISPUTE_RESOLVED', targetType: 'DISPUTE', targetId: dispute.id, metadata: { resolution: body.resolution, note: body.note } })
   375→   375→      await notifyMany(await staffUserIds('ADMIN', 'SUPPORT'), { type: 'DISPUTE_RESOLVED', title: `Dispute ${dispute.refCode} resolved`, body: body.resolution, link: '/admin?tab=disputes' })
   376→   376→      return ok({ message: `Dispute resolved (${body.resolution}).${body.resolution.includes('REFUND') ? ' Refund queued for finance processing.' : ''}` })
   377→   377→    }
   378→   378→    case 'reconcile-payment': {
   379→   379→      const order = await db.purchaseOrder.findUnique({ where: { id: body.orderId }, include: { payment: true } })
   380→   380→      if (!order) return fail(404, 'NOT_FOUND', 'Order not found.')
   381→   381→      if (order.paymentStatus === 'PROTECTED') return fail(409, 'ALREADY_PROTECTED', 'Order payment already protected.')
   382→   382→      const fees = await getFees()
   383→   383→      await db.paymentAuthorization.upsert({
   384→   384→        where: { orderId: order.id },
   385→   385→        create: { refCode: refCode('PAY', 6), orderId: order.id, provider: 'MANUAL_RECONCILED', status: 'PROTECTED', amountMinor: order.totalMinor, checkoutRef: body.providerRef, protectedAt: new Date(), resultDesc: body.note ?? 'Reconciled against provider statement' },
   386→   386→        update: { provider: 'MANUAL_RECONCILED', status: 'PROTECTED', checkoutRef: body.providerRef, protectedAt: new Date(), resultDesc: body.note ?? 'Reconciled against provider statement' },
   387→   387→      })
   388→   388→      await db.purchaseOrder.update({ where: { id: order.id }, data: { paymentStatus: 'PROTECTED' } })
   389→   389→      if (order.invoiceId) await db.invoice.update({ where: { id: order.invoiceId }, data: { status: 'PAID', paidAt: new Date(), paidVia: 'MANUAL' } })
   390→   390→      await poEvent(order.id, { type: 'PAYMENT', actorId: user.id, actorName: user.name, actorSide: staffSide, note: `Payment reconciled (provider ref ${body.providerRef}). Funds protected pending inspection.` })
   391→   391→      await audit({ actorId: user.id, actorName: user.name, action: 'ORDER_PAYMENT_RECONCILED', targetType: 'PURCHASE_ORDER', targetId: order.id, metadata: { providerRef: body.providerRef } })
   392→   392→      await notifyMany((await db.orgMember.findMany({ where: { orgId: order.garageOrgId }, select: { userId: true } })).map((m) => m.userId), { type: 'ORDER_PAYMENT', title: `Payment protected: ${order.refCode}`, body: 'Finance confirmed the payment. The supplier can now dispatch; funds release after your inspection.', link: '/procurement?tab=orders' })
   393→   393→      return ok({ message: 'Payment reconciled and protected. Supplier can dispatch.' })
   394→   394→    }
   395→   395→    case 'process-refund': {
   396→   396→      const refund = await db.refund.findUnique({ where: { id: body.refundId }, include: { order: true } })
   397→   397→      if (!refund) return fail(404, 'NOT_FOUND', 'Refund not found.')
   398→   398→      if (refund.status === 'PROCESSED') return fail(409, 'ALREADY_PROCESSED', 'Refund already processed.')
   399→   399→      await db.refund.update({
   400→   400→        where: { id: refund.id },
   401→   401→        data: { status: body.decision, processedBy: user.name, processedAt: new Date(), providerRef: body.providerRef ?? null },
   402→   402→      })
   403→   403→      if (body.decision === 'PROCESSED') {
   404→   404→        const payment = await db.paymentAuthorization.findUnique({ where: { orderId: refund.orderId } })
   405→   405→        if (payment) {
   406→   406→          const totalRefunded = payment.refundedAmountMinor + refund.amountMinor
   407→   407→          await db.paymentAuthorization.update({
   408→   408→            where: { id: payment.id },
   409→   409→            data: { refundedAmountMinor: totalRefunded, status: totalRefunded >= payment.amountMinor ? 'REFUNDED' : payment.status },
   410→   410→          })
   411→   411→        }
   412→   412→        await audit({ actorId: user.id, actorName: user.name, action: 'REFUND_PROCESSED', targetType: 'PURCHASE_ORDER', targetId: refund.orderId, metadata: { amount: refund.amountMinor, providerRef: body.providerRef } })
   413→   413→      } else {
   414→   414→        await audit({ actorId: user.id, actorName: user.name, action: `REFUND_${body.decision}`, targetType: 'PURCHASE_ORDER', targetId: refund.orderId, metadata: { note: body.note } })
   415→   415→      }
   416→   416→      return ok({ message: `Refund ${body.decision.toLowerCase()}.` })
   417→   417→    }
   418→   418→  }
   419→   419→})
