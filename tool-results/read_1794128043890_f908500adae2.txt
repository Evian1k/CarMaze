   423→  async function doCollectSelf(id: string | null) {
   424→    if (!id) return
   425→    try {
   426→      await apiSend(`/api/custody/${id}`, 'PATCH', { action: 'collect-self' })
   427→      toast.success('Confirmed: the garage has been notified to release your vehicle to you.')
   428→      setCollectSelfId(null)
   429→      refreshDetail(id)
   430→      loadCustodies()
   431→    } catch (e) {
   432→      toast.error(e instanceof Error ? e.message : 'Could not confirm collection.')
   433→    }
   434→  }

   440→  const meta = CUSTODY_STATUS_META[row.status] ?? { label: row.status, badge: 'bg-muted text-muted-foreground' }
   441→  return (
   442→    <button
   443→      onClick={onOpen}
   444→      className="block w-full rounded-xl border bg-card p-4 text-left shadow-sm transition-all hover:shadow-md"
   445→      aria-label={`Open custody job ${row.refCode} for ${row.vehicleDesc}`}
   446→    >
   447→      <div className="flex flex-wrap items-start justify-between gap-2">
   448→        <div className="min-w-0">
   449→          <h3 className="flex items-center gap-2 font-semibold leading-tight">
   450→            <CarFront className="h-4 w-4 shrink-0 text-primary" aria-hidden="true" />
   451→            <span className="truncate">{row.vehicleDesc}</span>
   452→            {row.plate && <Badge variant="secondary" className="shrink-0 font-mono text-[11px]">{row.plate}</Badge>}
   453→          </h3>
   454→          <p className="mt-1 text-xs text-muted-foreground">
   455→            {row.orgName ?? 'Garage'} · <span className="font-mono">{row.refCode}</span> · updated {timeAgo(row.updatedAt)}
   456→          </p>
   457→        </div>
   458→        <Badge variant="outline" className={cn('shrink-0', meta.badge)}>{meta.label}</Badge>
   459→      </div>
   460→      {row.requestedWork && <p className="mt-2 line-clamp-1 text-sm text-muted-foreground">{row.requestedWork}</p>}
   461→      <div className="mt-3">
   462→        <StatusStepper status={row.status} compact />
   463→      </div>
   464→    </button>
   465→  )

   468→function StatusStepper({ status, compact = false }: { status: string; compact?: boolean }) {
   469→  const reduce = useReducedMotion()
   470→  const idx = STEP_INDEX[status]
   471→
   472→  if (idx === undefined) {
   473→    return (

   777→function eventLabel(ev: CustodyEvent): string {
   778→  const labels: Record<string, string> = {
   779→    CREATED: 'Custody requested',
   780→    RECEIVE: 'Vehicle received',
   781→    STATUS_CHANGED: 'Status updated',
   782→    NOTE: 'Note added',
   783→    ROAD_TEST_REQUESTED: 'Road test requested',
   784→    ROAD_TEST_AUTHORISED: 'Road test authorised',
   785→    ROAD_TEST_DECLINED: 'Road test declined',
   786→    ROAD_TEST_STARTED: 'Road test started',
   787→    ROAD_TEST_ENDED: 'Road test ended',
   788→    COMPLETION_RECORDED: 'Work completed',
   789→    COLLECTION_AUTH_CREATED: 'Collection authorised for a third party',
   790→    COLLECTION_AUTH_REVOKED: 'Collection authorisation revoked',
   791→    COLLECT_SELF: 'Owner confirmed personal collection',
   792→    COLLECTED: 'Vehicle collected',
   793→    DISPUTE: 'Dispute opened',
   794→    INCIDENT: 'Incident reported',
   795→  }
   796→  if (labels[ev.type]) return labels[ev.type]
   797→  if (ev.toStatus) {
   798→    const meta = CUSTODY_STATUS_META[ev.toStatus]
   799→    return `Moved to ${meta?.label.toLowerCase() ?? ev.toStatus.toLowerCase()}`
   800→  }
   801→  return ev.type.replaceAll('_', ' ').toLowerCase()
   802→}

   806→function CollectionAuthDialog({ custodyId, onClose, onDone }: {
   807→  custodyId: string | null
   808→  onClose: () => void
   809→  onDone: () => void
   810→}) {
   811→  const [collectorName, setCollectorName] = useState('')
   812→  const [collectorPhone, setCollectorPhone] = useState('')
   813→  const [busy, setBusy] = useState(false)
   814→  const [code, setCode] = useState<string | null>(null)
   815→
   816→  useEffect(() => {
   817→    if (custodyId) { setCode(null); setCollectorName(''); setCollectorPhone('') }
   818→  }, [custodyId])
   819→
   820→  async function create() {
   821→    if (!custodyId) return
   822→    if (collectorName.trim().length < 3) { toast.error('Enter the full name of the person collecting.'); return }
   823→    if (!/^(?:\+?254|0)(?:7|1)\d{8}$/.test(collectorPhone.replace(/\s+/g, ''))) {
   824→      toast.error('Enter a valid Safaricom phone number for the collector, e.g. 0712 345 678.')
   825→      return
   826→    }
   827→    setBusy(true)
   828→    try {
   829→      const d = await apiSend<{ code: string }>(`/api/custody/${custodyId}/collection-auth`, 'POST', {
   830→        collectorName: collectorName.trim(),
   831→        collectorPhone: collectorPhone.replace(/\s+/g, ''),
   832→      })
   833→      setCode(d.code)
   834→      onDone()
   835→    } catch (e) {
   836→      toast.error(e instanceof Error ? e.message : 'Could not create the authorisation.')
   837→    } finally {
   838→      setBusy(false)
   839→    }
   840→  }

   842→  return (
   843→    <Dialog open={custodyId !== null} onOpenChange={(open) => { if (!open) onClose() }}>

   859→                onClick={async () => {
   860→                  try {
   861→                    await navigator.clipboard.writeText(code)
   862→                    toast.success('Code copied')
   863→                  } catch {
   864→                    toast.error('Copy failed: write it down now.')
   865→                  }
   866→                }}
   867→              >
   868→                <Copy className="h-3.5 w-3.5" /> Copy code

   872→                the vehicle will not be handed over without it.
   873→              </p>
   874→            </div>
   875→            <DialogFooter>
   876→              <Button className="min-h-11" onClick={onClose}>Done</Button>

   885→                valid 48 hours.
   886→              </DialogDescription>
   887→            </DialogHeader>
   888→            <div className="space-y-3">
   889→              <div className="space-y-1.5">
   890→                <Label htmlFor="ca-name">Collector&apos;s full name *</Label>

   893→              <div className="space-y-1.5">
   894→                <Label htmlFor="ca-phone">Collector&apos;s phone *</Label>
   895→                <Input id="ca-phone" type="tel" inputMode="tel" value={collectorPhone} onChange={(e) => setCollectorPhone(e.target.value)} placeholder="07xx xxx xxx" />
   896→              </div>
   897→            </div>
   898→            <DialogFooter>
   899→              <Button variant="ghost" onClick={onClose}>Cancel</Button>
   900→              <Button onClick={create} disabled={busy} className="min-h-11 gap-2">
   901→                {busy && <Loader2 className="h-4 w-4 animate-spin" />} Generate code
   902→              </Button>
   903→            </DialogFooter>
   904→          </>
   905→        )}
   906→      </DialogContent>
   907→    </Dialog>

   989→function ReportIncidentDialog({ open, onClose, onReported }: {
   990→  open: boolean
   991→  onClose: () => void
   992→  onReported: (safetyNote: string) => void
   993→}) {
   994→  const [category, setCategory] = useState<string>('OTHER')
   995→  const [title, setTitle] = useState('')
   996→  const [details, setDetails] = useState('')
   997→  const [vehicleDesc, setVehicleDesc] = useState('')
   998→  const [evidence, setEvidence] = useState<{ url: string; name: string }[]>([])
   999→  const [uploading, setUploading] = useState(false)
  1000→  const [busy, setBusy] = useState(false)
  1001→  const valid = title.trim().length >= 4 && details.trim().length >= 20
  1002→
  1003→  useEffect(() => {
  1004→    if (open) { setCategory('OTHER'); setTitle(''); setDetails(''); setVehicleDesc(''); setEvidence([]) }
  1005→  }, [open])
  1006→
  1007→  async function uploadEvidence(file: File) {
  1008→    if (evidence.length >= 3) { toast.error('Up to 3 evidence files.'); return }
  1009→    setUploading(true)
  1010→    try {
  1011→      const fd = new FormData()
  1012→      fd.append('file', file)
  1013→      const d = await apiUpload<{ url: string; name: string }>('/api/upload', fd)
  1014→      setEvidence((prev) => [...prev, { url: d.url, name: d.name || file.name }])
  1015→    } catch (e) {
  1016→      toast.error(e instanceof Error ? e.message : 'Upload failed.')
  1017→    } finally {
  1018→      setUploading(false)
  1019→    }
  1020→  }
  1021→
  1022→  async function submit() {
  1023→    if (!valid) return
  1024→    setBusy(true)
  1025→    try {
  1026→      const d = await apiSend<{ incident: { refCode: string }; safetyNote: string }>('/api/incidents', 'POST', {
  1027→        category,
  1028→        title: title.trim(),
  1029→        details: details.trim(),
  1030→        vehicleDesc: vehicleDesc.trim() || undefined,
  1031→        evidence: evidence.length > 0 ? evidence.map((f) => f.url) : undefined,
  1032→      })
  1033→      toast.success(`Incident ${d.incident.refCode} reported.`)
  1034→      onReported(d.safetyNote ?? SAFETY_NOTE_FALLBACK)
  1035→      onClose()
  1036→    } catch (e) {
  1037→      toast.error(e instanceof Error ? e.message : 'Could not report the incident.')
  1038→    } finally {
  1039→      setBusy(false)
  1040→    }
  1041→  }
  1042→
  1043→  return (
  1044→    <Dialog open={open} onOpenChange={(o) => { if (!o) onClose() }}>
  1045→      <DialogContent className="max-h-[92vh] overflow-y-auto sm:max-w-md">
  1046→        <DialogHeader>
  1047→          <DialogTitle>Report a security concern</DialogTitle>
  1048→          <DialogDescription>
  1049→            Every report is reviewed by CARMAZE security. False reports waste everyone&apos;s time: be precise and honest.
  1050→          </DialogDescription>
  1051→        </DialogHeader>
  1052→        <div className="space-y-3">
  1053→          <div className="space-y-1.5">
  1054→            <Label>Category *</Label>
  1055→            <Select value={category} onValueChange={setCategory}>
  1056→              <SelectTrigger aria-label="Incident category"><SelectValue /></SelectTrigger>
  1057→              <SelectContent>
  1058→                {INCIDENT_CATEGORIES.map((c) => <SelectItem key={c.value} value={c.value}>{c.label}</SelectItem>)}
  1059→              </SelectContent>
  1060→            </Select>
  1061→          </div>
  1062→          <div className="space-y-1.5">
  1063→            <Label htmlFor="ri-title">Short title *</Label>
  1064→            <Input id="ri-title" value={title} onChange={(e) => setTitle(e.target.value)} placeholder="e.g. Car missing from yard after service" maxLength={120} />
  1065→          </div>
  1066→          <div className="space-y-1.5">
  1067→            <Label htmlFor="ri-details">Full details * <span className="font-normal text-muted-foreground">(min 20 characters)</span></Label>
  1068→            <Textarea id="ri-details" rows={4} value={details} onChange={(e) => setDetails(e.target.value)} placeholder="What happened, when, who was involved, what you have already tried…" />
  1069→            <p className={cn('text-xs', details.trim().length >= 20 ? 'text-muted-foreground' : 'text-destructive')}>
  1070→              {details.trim().length}/20 characters minimum
