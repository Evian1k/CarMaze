   207→   207→    let cancelled = false

   267→   267→      )}

   269→   269→      {/* ── Active garage jobs ── */}

   271→   271→        <div className="flex flex-wrap items-center justify-between gap-2">

   274→   274→              <Wrench className="h-5 w-5 text-primary" aria-hidden="true" /> Vehicles with the garage
   275→   275→            </h2>
   276→   276→            <p className="mt-0.5 text-sm text-muted-foreground">
   277→   277→              Live custody tracking: approve road tests, authorise collection and dispute anything that looks wrong.
   278→   278→            </p>

   281→   281→            <ShieldAlert className="h-4 w-4" /> Report a security concern
   282→   282→          </Button>
   283→   283→        </div>
   284→   284→
   285→   285→        {custodies === null ? (
   286→   286→          <div className="mt-4 space-y-3">
   287→   287→            <Skeleton className="h-28 rounded-xl" />
   288→   288→            <Skeleton className="h-28 rounded-xl" />
   289→   289→          </div>
   290→   290→        ) : custodies.length === 0 ? (
   291→   291→          <Card className="mt-4 border-dashed">
   292→   292→            <CardContent className="flex flex-col items-center py-12 text-center">
   293→   293→              <div className="flex h-14 w-14 items-center justify-center rounded-full bg-primary/10 text-primary">
   294→   294→                <ShieldCheck className="h-7 w-7" aria-hidden="true" />

   313→   313→                <div className="space-y-3 px-3 pb-3">
   314→   314→                  {pastJobs.map((c) => <JobCard key={c.id} row={c} onOpen={() => openDetail(c.id)} />)}
   315→   315→                </div>
   316→   316→              </details>
   317→   317→            )}
   318→   318→          </>

   322→   322→      {/* ── Security alerts ── */}

   376→   376→            </AlertDialogDescription>

   387→   387→        </AlertDialogContent>
