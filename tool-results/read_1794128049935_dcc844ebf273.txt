    63→    63→/** Next 14 days (local YYYY-MM-DD) + hourly 09:00–17:00 slots. */
    64→    64→export function slotOptions() {
    65→    65→  const days: { value: string; label: string }[] = []
    66→    66→  const fmt = new Intl.DateTimeFormat('en-KE', { weekday: 'short', day: 'numeric', month: 'short' })
    67→    67→  for (let i = 0; i < 14; i++) {
    68→    68→    const d = new Date()
    69→    69→    d.setDate(d.getDate() + i)
    70→    70→    const iso = `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, '0')}-${String(d.getDate()).padStart(2, '0')}`
    71→    71→    days.push({ value: iso, label: i === 0 ? `Today · ${fmt.format(d)}` : i === 1 ? `Tomorrow · ${fmt.format(d)}` : fmt.format(d) })
    72→    72→  }
    73→    73→  const times: string[] = []
    74→    74→  for (let h = 9; h <= 17; h++) times.push(`${String(h).padStart(2, '0')}:00`)
    75→    75→  return { days, times }
    76→    76→}

    78→    78→function localIsoDate(d: Date): string {
    79→    79→  return `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, '0')}-${String(d.getDate()).padStart(2, '0')}`
    80→    80→}

    82→    82→/** Handles the auth-gate uniformly for all storefront dialogs. */
    83→    83→function useSubmitGuard() {
    84→    84→  const user = useAuth((s) => s.user)
    85→    85→  return function guard(): boolean {
    86→    86→    if (!user) {
    87→    87→      toast.info('Sign in to continue', { description: 'Create a free CARMAZE account first: it takes a minute.' })
    88→    88→      navigate('signin')
    89→    89→      return false
    90→    90→    }
    91→    91→    return true
    92→    92→  }
    93→    93→}

    95→    95→function isAuthError(e: unknown): boolean {
    96→    96→  return typeof e === 'object' && e !== null && 'status' in e && (e as { status?: number }).status === 401
    97→    97→}

    99→    99→function errMessage(e: unknown, fallback: string): string {
   100→   100→  return e instanceof Error && e.message ? e.message : fallback
   101→   101→}

   103→   103→/** Time input slots that are still bookable for the chosen date. */
   104→   104→function useValidTimes(date: string, bufferMinutes: number) {
   105→   105→  const { times } = useMemo(() => slotOptions(), [])
   106→   106→  const todayIso = localIsoDate(new Date())
   107→   107→  return times.filter((t) => {
   108→   108→    if (date !== todayIso) return true
   109→   109→    const [h, m] = t.split(':').map(Number)
   110→   110→    const slot = new Date()
   111→   111→    slot.setHours(h, m, 0, 0)
   112→   112→    return slot.getTime() >= Date.now() + bufferMinutes * 60 * 1000
   113→   113→  })
   114→   114→}

   653→   653→function ReviewForm({
