    79→function draftFromQuery(query: URLSearchParams): FilterDraft {
    80→  const minor = (key: string) => {
    81→    const raw = query.get(key)
    82→    if (!raw) return ''
    83→    const n = Number(raw)
    84→    return Number.isFinite(n) && n > 0 ? String(Math.round(n / 100)) : ''
    85→  }
    86→  return {
    87→    q: query.get('q') ?? '',
    88→    make: query.get('make') ?? '',
    89→    model: query.get('model') ?? '',
    90→    bodyType: query.get('bodyType') ?? '',
    91→    county: query.get('county') ?? '',
    92→    fuel: query.get('fuel') ?? '',
    93→    transmission: query.get('transmission') ?? '',
    94→    condition: query.get('condition') ?? '',
    95→    sellerType: query.get('sellerType') ?? '',
    96→    verified: query.get('verified') === '1',
    97→    minPrice: minor('minPrice'),
    98→    maxPrice: minor('maxPrice'),
    99→    minYear: query.get('minYear') ?? '',
   100→    maxYear: query.get('maxYear') ?? '',
   101→    maxMileage: query.get('maxMileage') ?? '',
   102→    sort: query.get('sort') || 'newest',
   103→    page: query.get('page') || '1',
   104→    org: query.get('org') ?? '',
   105→  }
   106→}
   107→
   108→/** Convert the editable draft into URL query params (prices back to minor units). */
   109→function draftToParams(d: FilterDraft): Record<string, string | number | undefined> {
   110→  const num = (s: string) => {
   111→    const n = Number(s)
   112→    return s !== '' && Number.isFinite(n) && n > 0 ? n : undefined
   113→  }
   114→  return {
   115→    q: d.q.trim() || undefined,
   116→    make: d.make || undefined,
   117→    model: d.model.trim() || undefined,
   118→    bodyType: d.bodyType || undefined,
   119→    county: d.county || undefined,
   120→    fuel: d.fuel || undefined,
   121→    transmission: d.transmission || undefined,
   122→    condition: d.condition || undefined,
   123→    sellerType: d.sellerType || undefined,
   124→    verified: d.verified ? '1' : undefined,
   125→    minPrice: num(d.minPrice) ? Math.round(Number(d.minPrice) * 100) : undefined,
   126→    maxPrice: num(d.maxPrice) ? Math.round(Number(d.maxPrice) * 100) : undefined,
   127→    minYear: num(d.minYear),
   128→    maxYear: num(d.maxYear),
   129→    maxMileage: num(d.maxMileage),
   130→    sort: d.sort && d.sort !== 'newest' ? d.sort : undefined,
   131→    org: d.org || undefined,
   132→    // page intentionally omitted → resets to 1 whenever filters change
   133→  }
   134→}

   462→              variant="ghost"
   463→              size="sm"
   464→              className="h-11 px-4"
   465→              onClick={() => { setAiResults(null); setAiMessage('') }}

   467→              Clear AI results

   469→          </div>
   470→          {aiResults.length === 0 ? (
   471→            <p className="text-sm text-muted-foreground">No matches found for that description; try being more specific.</p>
   472→          ) : (
   473→            <Stagger key={aiResults.map((r) => r.id).join(',')} className="grid grid-cols-1 gap-4 sm:grid-cols-2 xl:grid-cols-4">
   474→              {aiResults.map((v) => (
   475→                <StaggerItem key={v.id}>
   476→                  <VehicleCard v={v} saved={savedIds.has(v.id)} onToggleSave={toggleSaved} />
   477→                </StaggerItem>
   478→              ))}
   479→            </Stagger>
   480→          )}
   481→        </section>
   482→      )}
   483→
   484→      <div className="flex gap-6">
   485→        {/* Desktop filter sidebar */}
   486→        <aside className="hidden w-72 shrink-0 lg:block" aria-label="Search filters">
   487→          <Card>
   488→            <CardContent className="p-6">
   489→              <div className="mb-4 flex items-center justify-between">
   490→                <h2 className="font-semibold">Filters</h2>
   491→                {activeFilterCount > 0 && <Badge variant="secondary">{activeFilterCount}</Badge>}
   492→              </div>
   493→              {filterFields}
   494→            </CardContent>
   495→          </Card>
   496→        </aside>
   497→
   498→        <div className="min-w-0 flex-1">
   499→          {/* Toolbar */}
   500→          <div className="mb-4 flex flex-wrap items-center justify-between gap-3">
   501→            <div className="flex items-center gap-2">
   502→              <Sheet open={filtersOpen} onOpenChange={setFiltersOpen}>

   509→                <SheetContent side="left" className="w-80 overflow-y-auto scrollbar-thin">
   510→                  <SheetHeader>
   511→                    <SheetTitle>Filters</SheetTitle>
   512→                  </SheetHeader>
   513→                  <div className="px-4 pb-6">{filterFields}</div>
   514→                </SheetContent>
   515→              </Sheet>
   516→              <span className="text-sm text-muted-foreground">
   517→                {loading ? 'Searching…' : `${total.toLocaleString()} results`}
   518→              </span>
   519→            </div>
   520→            <div className="flex items-center gap-2">
   521→              <Button
   522→                variant="outline"
   523→                className="h-11 px-4"
   524→                onClick={() => {
   525→                  if (!user) {
   526→                    toast.info('Sign in to save this search')
   527→                    navigate('signin')
   528→                    return
   529→                  }
   530→                  setSaveSearchOpen(true)
   531→                }}
   532→              >
   533→                <BookmarkPlus className="h-4 w-4" /> <span className="hidden sm:inline">Save search</span>
   534→              </Button>
   535→              <Select value={draft.sort} onValueChange={(v) => patchDraft({ sort: v })}>
   536→                <SelectTrigger className="h-11 w-[170px]" aria-label="Sort results">
   537→                  <SelectValue />
   538→                </SelectTrigger>
   539→                <SelectContent>
   540→                  {SORT_OPTIONS.map((o) => <SelectItem key={o.value} value={o.value}>{o.label}</SelectItem>)}
   541→                </SelectContent>
   542→              </Select>
   543→            </div>
   544→          </div>
   545→
   546→          {/* Results */}
   547→          {loading ? (
   548→            <div className="grid grid-cols-1 gap-4 sm:grid-cols-2 xl:grid-cols-4" aria-busy="true">
   549→              {Array.from({ length: 8 }).map((_, i) => (
   550→                <Card key={i} className="overflow-hidden">
   551→                  <Skeleton className="aspect-[16/10] w-full rounded-none" />
   552→                  <CardContent className="space-y-2 p-4">
   553→                    <Skeleton className="h-4 w-3/4" />
   554→                    <Skeleton className="h-5 w-1/2" />
   555→                    <Skeleton className="h-3 w-2/3" />
   556→                  </CardContent>
   557→                </Card>
   558→              ))}
   559→            </div>
   560→          ) : error ? (
   561→            <Card>
   562→              <CardContent className="flex flex-col items-center gap-3 p-10 text-center">
   563→                <SearchX className="h-10 w-10 text-muted-foreground" />
   564→                <p className="font-medium">Could not load listings</p>
   565→                <p className="text-sm text-muted-foreground">{error}</p>
   566→                <Button className="h-11 px-5" onClick={() => navigate('cars')}>Retry</Button>

   569→          ) : vehicles.length === 0 ? (
   570→            <Card>
   571→              <CardContent className="flex flex-col items-center gap-3 p-10 text-center">
   572→                <SearchX className="h-10 w-10 text-muted-foreground" />
   573→                <p className="font-medium">No vehicles match your search</p>
   574→                <p className="max-w-sm text-sm text-muted-foreground">
   575→                  Try widening your price range, removing a filter, or searching a different make.
   576→                </p>
   577→                <Button className="h-11 px-5" onClick={clearFilters}>Clear all filters</Button>
   578→              </CardContent>
   579→            </Card>

   581→            <Stagger className="grid grid-cols-1 gap-4 sm:grid-cols-2 xl:grid-cols-4">
   582→              {vehicles.map((v) => (
   583→                <StaggerItem key={v.id} className={`relative rounded-xl ${compareIds.includes(v.id) ? 'ring-2 ring-accent ring-offset-2 ring-offset-background' : ''}`}>
   584→                  <VehicleCard v={v} saved={savedIds.has(v.id)} onToggleSave={toggleSaved} />
   585→                  <button
   586→                    onClick={() => toggleCompare(v)}
   587→                    aria-pressed={compareIds.includes(v.id)}

   590→                      compareIds.includes(v.id) ? 'bg-accent text-accent-foreground' : 'bg-black/35 text-white hover:bg-black/55'
   591→                    }`}
   592→                  >
   593→                    <Scale className="h-4 w-4" />
   594→                  </button>
   595→                </StaggerItem>
   596→              ))}
   597→            </Stagger>
   598→          )}
   599→
   600→          {/* Pagination */}
   601→          {!loading && !error && totalPages > 1 && (
   602→            <nav className="mt-8 flex flex-wrap items-center justify-center gap-2" aria-label="Pagination">
   603→              <Button
   604→                variant="outline"
   605→                className="h-11 w-11 p-0"
   606→                disabled={page <= 1}
   607→                onClick={() => goToPage(page - 1)}
   608→                aria-label="Previous page"
   609→              >
   610→                ‹
   611→              </Button>

   613→                p === '…' ? (
   614→                  <span key={`gap-${i}`} className="px-1 text-muted-foreground">…</span>
   615→                ) : (
   616→                  <Button
   617→                    key={p}
   618→                    variant={p === page ? 'default' : 'outline'}
   619→                    className="h-11 w-11 p-0"
   620→                    onClick={() => goToPage(p)}

   623→                    {p}

   627→              <Button
   628→                variant="outline"
   629→                className="h-11 w-11 p-0"
   630→                disabled={page >= totalPages}
   631→                onClick={() => goToPage(page + 1)}
   632→                aria-label="Next page"
   633→              >
   634→                ›
   635→              </Button>
   636→            </nav>
   637→          )}
   638→        </div>
   639→      </div>
   640→
   641→      {/* Compare bar */}
   642→      {compareIds.length > 0 && (
   643→        <div className="fixed inset-x-0 bottom-0 z-40 border-t border-border bg-card/95 px-4 py-3 shadow-[0_-4px_20px_rgba(0,0,0,0.08)] backdrop-blur">
   644→          <div className="mx-auto flex max-w-7xl items-center justify-between gap-3">
   645→            <div className="flex items-center gap-2 overflow-x-auto scrollbar-thin">
   646→              <Scale className="h-5 w-5 shrink-0 text-primary" />

   648→                {compareIds.length} of 3 selected
   649→              </span>
   650→              <div className="hidden items-center gap-1 sm:flex">
   651→                {compareVehicles.map((v) => (

   653→                    <span className="truncate">{v.make} {v.model}</span>

   656→                    </button>
   657→                  </Badge>
   658→                ))}
   659→              </div>
   660→            </div>
   661→            <div className="flex shrink-0 items-center gap-2">
   662→              <Button variant="ghost" className="h-11 px-3" onClick={() => setCompareIds([])}>Clear</Button>
   663→              <Button className="h-11 px-5" disabled={compareVehicles.length < 2} onClick={() => setCompareOpen(true)}>
   664→                Compare {compareVehicles.length >= 2 ? `(${compareVehicles.length})` : ''}
   665→              </Button>
   666→            </div>
   667→          </div>
   668→        </div>
   669→      )}
   670→
   671→      {/* Compare dialog */}
   672→      <Dialog open={compareOpen} onOpenChange={setCompareOpen}>
   673→        <DialogContent className="max-h-[85vh] max-w-3xl overflow-y-auto scrollbar-thin">
   674→          <DialogHeader>
   675→            <DialogTitle>Compare vehicles</DialogTitle>
   676→            <DialogDescription>Side-by-side specification comparison.</DialogDescription>
   677→          </DialogHeader>
   678→          {compareVehicles.length >= 2 ? (
   679→            <div className="overflow-x-auto">
   680→              <table className="w-full min-w-[480px] text-sm">
   681→                <thead>
   682→                  <tr>
   683→                    <th className="w-28 p-2 text-left font-medium text-muted-foreground">Spec</th>
   684→                    {compareVehicles.map((v) => (
   685→                      <th key={v.id} className="p-2 text-left font-semibold">
   686→                        {v.make} {v.model}
   687→                        <div className="text-xs font-normal text-muted-foreground">{v.year}</div>
   688→                      </th>
   689→                    ))}
   690→                  </tr>
   691→                </thead>
   692→                <tbody>
   693→                  {COMPARE_ROWS.map((row) => (
   694→                    <tr key={row.label} className="border-t border-border">
   695→                      <td className="p-2 align-top text-muted-foreground">{row.label}</td>
   696→                      {compareVehicles.map((v) => (
   697→                        <td key={v.id} className="p-2 align-top font-medium">{row.get(v)}</td>
   698→                      ))}
   699→                    </tr>
   700→                  ))}
   701→                </tbody>
   702→              </table>
   703→            </div>

   707→        </DialogContent>
   708→      </Dialog>
   709→
   710→      {/* Save search dialog */}
   711→      <Dialog open={saveSearchOpen} onOpenChange={setSaveSearchOpen}>

   713→          <DialogHeader>
   714→            <DialogTitle>Save this search</DialogTitle>

   716→              We&apos;ll keep these filters so you can rerun them anytime from your account.

   719→          <div className="space-y-1.5">
   720→            <Label htmlFor="search-name">Search name</Label>

   722→              id="search-name"

   725→              onChange={(e) => setSearchName(e.target.value)}
   726→              onKeyDown={(e) => { if (e.key === 'Enter') saveSearch() }}

   731→          </div>
   732→          <DialogFooter>
