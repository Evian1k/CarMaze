     1→'use client'

     3→/* CARMAZE: Vehicle detail (#/vehicle-detail/<id>)
     4→ * Gallery, full specification, safety guidance, honest seller card,
     5→ * save / share / report / enquire actions and similar listings.
     6→ */
     7→
     8→import { useEffect, useState } from 'react'
     9→import { motion } from 'framer-motion'
    10→import {
    11→  Heart, Share2, Flag, Mail, BadgeCheck, ShieldCheck, MapPin, Gauge, Fuel, Cog, Car,
    12→  Palette, CalendarDays, Eye, Hash, ShieldAlert, ArrowLeft, MessageSquareText, Phone,
    13→  CalendarCheck, MessageCircle,
    14→} from 'lucide-react'
    15→import { Button } from '@/components/ui/button'
    16→import { Card, CardContent, CardHeader, CardTitle } from '@/components/ui/card'
    17→import { Badge } from '@/components/ui/badge'
    18→import { Input } from '@/components/ui/input'
    19→import { Label } from '@/components/ui/label'
    20→import { Textarea } from '@/components/ui/textarea'
    21→import { Checkbox } from '@/components/ui/checkbox'
    22→import { Skeleton } from '@/components/ui/skeleton'
    23→import { Separator } from '@/components/ui/separator'
    24→import { Alert, AlertDescription, AlertTitle } from '@/components/ui/alert'
    25→import { Dialog, DialogContent, DialogDescription, DialogFooter, DialogHeader, DialogTitle } from '@/components/ui/dialog'
    26→import { Select, SelectContent, SelectItem, SelectTrigger, SelectValue } from '@/components/ui/select'
    27→import { VehicleCard, type VehicleCardData } from '@/components/marketplace/vehicle-card'
    28→import { Reveal, Stagger, StaggerItem } from '@/components/motion/reveal'
    29→import { MessageDialog, TestDriveDialog } from '@/components/storefront/dialogs'
    30→import { useHashRoute } from '@/hooks/use-hash-route'
    31→import { useAuth } from '@/stores/auth'
    32→import { apiGet, apiSend, navigate } from '@/lib/client'
    33→import { money, timeAgo, CONDITION_LABELS, VEHICLE_STATUS_LABELS, vehicleTitle } from '@/lib/format'
    34→import { toast } from 'sonner'
    35→
    36→type VehicleDetail = {
    37→  id: string
    38→  refCode: string
    39→  make: string
    40→  model: string
    41→  year: number
    42→  priceMinor: number
    43→  currency: string
    44→  mileageKm: number
    45→  fuel: string
    46→  transmission: string
    47→  bodyType: string
    48→  engineCc: number | null
    49→  condition: string
    50→  county: string
    51→  town: string | null
    52→  color: string
    53→  description: string
    54→  sellerType: string
    55→  status: string
    56→  featured: boolean
    57→  views: number
    58→  matchWithDealers: boolean
    59→  createdAt: string
    60→  photos: { url: string; order: number }[]
    61→  owner: { name: string; verifiedSeller?: boolean }
    62→  org?: {
    63→    id: string
    64→    name: string
    65→    county: string
    66→    town: string | null
    67→    phone: string | null
    68→    location: string | null
    69→    description: string | null
    70→    status: string
    71→  } | null
    72→}
    73→
    74→type DetailResponse = { vehicle: VehicleDetail; similar: VehicleCardData[] }
    75→
    76→const REPORT_REASONS = [
    77→  'Suspected scam or fraud',
    78→  'Price looks wrong or misleading',
    79→  'Vehicle already sold',
    80→  'Duplicate listing',
    81→  'Prohibited or stolen vehicle',
    82→  'Other',
    83→]
    84→
    85→export default function View() {
    86→  const route = useHashRoute()
    87→  const id = route.segments[1] ?? ''
    88→  const { user } = useAuth()
    89→
    90→  const [vehicle, setVehicle] = useState<VehicleDetail | null>(null)
    91→  const [similar, setSimilar] = useState<VehicleCardData[]>([])
    92→  const [notFound, setNotFound] = useState(false)
    93→  const [error, setError] = useState<string | null>(null)
    94→  const [activePhoto, setActivePhoto] = useState(0)
    95→  const [savedIds, setSavedIds] = useState<Set<string>>(new Set())
    96→
    97→  const [enquiryOpen, setEnquiryOpen] = useState(false)
    98→  const [enquiryMsg, setEnquiryMsg] = useState('')
    99→  const [phoneShare, setPhoneShare] = useState(true)
   100→  const [phone, setPhone] = useState('')
   101→  const [sendingEnquiry, setSendingEnquiry] = useState(false)
   102→
   103→  const [reportOpen, setReportOpen] = useState(false)
   104→  const [reportReason, setReportReason] = useState<string>('')
   105→  const [reportDetails, setReportDetails] = useState('')
   106→  const [sendingReport, setSendingReport] = useState(false)
   107→
   108→  // Storefront actions (dealer listings with an org): test drive / vehicle enquiry
   109→  const [storefrontAction, setStorefrontAction] = useState<'message' | 'test-drive' | null>(null)
   110→
   111→  // ── Load detail (and reset view-local state per vehicle) ──
   112→  useEffect(() => {
   113→    if (!id) return
   114→    let active = true
   115→    Promise.resolve().then(() => {
   116→      if (!active) return
   117→      setVehicle(null)
   118→      setNotFound(false)
   119→      setError(null)
   120→      setActivePhoto(0)
   121→    })
   122→    apiGet<DetailResponse>(`/api/vehicles/${id}`)
   123→      .then((d) => {
   124→        if (!active) return
   125→        setVehicle(d.vehicle)
   126→        setSimilar(d.similar ?? [])
   127→      })
   128→      .catch((e: { status?: number; message: string }) => {
   129→        if (!active) return
   130→        if (e.status === 404) setNotFound(true)
   131→        else setError(e.message)
   132→      })
   133→    return () => { active = false }
   134→  }, [id])
   135→
   136→  // ── Fire-and-forget view ping (server dedupes per IP) ──
   137→  useEffect(() => {
   138→    if (!id) return
   139→    apiSend(`/api/vehicles/${id}/view`, 'POST').catch(() => {})
   140→  }, [id])
   141→
   142→  // ── Saved vehicles (for the heart) ──
   143→  useEffect(() => {
   144→    if (!user) {
   145→      const t = setTimeout(() => setSavedIds(new Set()), 0)
   146→      return () => clearTimeout(t)
   147→    }
   148→    let active = true
   149→    apiGet<{ saved: { vehicle: VehicleCardData }[] }>('/api/saved-vehicles')
   150→      .then((d) => { if (active) setSavedIds(new Set(d.saved.map((s) => s.vehicle.id))) })
   151→      .catch(() => {})
   152→    return () => { active = false }
   153→  }, [user])
   154→
   155→  // ── Document title ──
   156→  useEffect(() => {
   157→    if (vehicle) document.title = `${vehicleTitle(vehicle)} | CARMAZE`
   158→    return () => { document.title = 'CARMAZE' }
   159→  }, [vehicle])
   160→
   161→  // ── Prefill the enquiry phone from the session ──
   162→  useEffect(() => {
   163→    if (user) {
   164→      const t = setTimeout(() => setPhone(user.phone ?? ''), 0)
   165→      return () => clearTimeout(t)
   166→    }
   167→  }, [user])
   168→
   169→  function toggleSaved() {
   170→    if (!vehicle) return
   171→    if (!user) {
   172→      toast.info('Sign in to save vehicles')
   173→      navigate('signin')
   174→      return
   175→    }
   176→    const isSaved = savedIds.has(vehicle.id)
   177→    setSavedIds((prev) => {
   178→      const next = new Set(prev)
   179→      if (isSaved) next.delete(vehicle.id)
   180→      else next.add(vehicle.id)
   181→      return next
   182→    })
   183→    const done = isSaved
   184→      ? apiSend(`/api/saved-vehicles?vehicleId=${vehicle.id}`, 'DELETE')
   185→      : apiSend('/api/saved-vehicles', 'POST', { vehicleId: vehicle.id })
   186→    done
   187→      .then(() => toast.success(isSaved ? 'Removed from your shortlist' : 'Saved to your shortlist'))
   188→      .catch((e: Error) => {
   189→        setSavedIds((prev) => {
   190→          const next = new Set(prev)
   191→          if (isSaved) next.add(vehicle.id)
   192→          else next.delete(vehicle.id)
   193→          return next
   194→        })
   195→        toast.error(e.message)
   196→      })
   197→  }
   198→
   199→  function share() {
   200→    const url = window.location.href
   201→    const done = navigator.clipboard?.writeText(url)
   202→    Promise.resolve(done)
   203→      .then(() => toast.success('Link copied to clipboard'))
   204→      .catch(() => toast.info('Copy this link', { description: url }))
   205→  }
   206→
   207→  function sendEnquiry() {
   208→    if (!vehicle) return
   209→    if (enquiryMsg.trim().length < 10) { toast.error('Please write a message (at least 10 characters)'); return }
   210→    if (phoneShare && phone.trim() && !/^[+\d][\d\s-]{6,}$/.test(phone.trim())) {
   211→      toast.error('Enter a valid phone number')
   212→      return
   213→    }
   214→    setSendingEnquiry(true)
   215→    apiSend<{ enquiry: { id: string; refCode: string } }>('/api/enquiries', 'POST', {
   216→      vehicleId: vehicle.id,
   217→      message: enquiryMsg.trim(),
   218→      phoneShare,
   219→      phone: phoneShare ? (phone.trim() || undefined) : undefined,
   220→    })
   221→      .then((res) => {
   222→        setEnquiryOpen(false)
   223→        setEnquiryMsg('')
   224→        toast.success('Enquiry sent to the seller', {
   225→          description: `Reference ${res.enquiry.refCode}. Track replies under Account → Enquiries.`,
   226→          action: { label: 'View', onClick: () => navigate('account') },
   227→        })
   228→      })
   229→      .catch((e: Error) => toast.error(e.message))
   230→      .finally(() => setSendingEnquiry(false))
   231→  }
   232→
   233→  function sendReport() {
   234→    if (!vehicle) return
   235→    if (!user) {
   236→      toast.info('Sign in to report a listing')
   237→      navigate('signin')
   238→      return
   239→    }
   240→    if (!reportReason) { toast.error('Choose a reason'); return }
   241→    setSendingReport(true)
   242→    apiSend('/api/reports', 'POST', {
   243→      targetType: 'LISTING',
   244→      targetId: vehicle.id,
   245→      reason: reportReason,
   246→      details: reportDetails.trim() || undefined,
   247→    })
   248→      .then(() => {
   249→        setReportOpen(false)
   250→        setReportDetails('')
   251→        setReportReason('')
   252→        toast.success('Report submitted', { description: 'Our moderation team will review it shortly.' })
   253→      })
   254→      .catch((e: Error) => toast.error(e.message))
   255→      .finally(() => setSendingReport(false))
   256→  }
   257→
   258→  if (notFound) {
   259→    return (
   260→      <div className="mx-auto w-full max-w-2xl px-4 py-20 text-center">
   261→        <Car className="mx-auto h-12 w-12 text-muted-foreground" />
   262→        <h1 className="mt-4 text-2xl font-bold">This listing is not available</h1>
   263→        <p className="mt-2 text-sm text-muted-foreground">
   264→          It may have been sold, removed or is still awaiting review.
   265→        </p>
   266→        <Button className="mt-6 h-11 px-6" onClick={() => navigate('cars')}>
   267→          <ArrowLeft className="h-4 w-4" /> Browse all cars
   268→        </Button>
   269→      </div>
   270→    )
   271→  }
   272→
   273→  if (error) {
   274→    return (
   275→      <div className="mx-auto w-full max-w-2xl px-4 py-20 text-center">
   276→        <p className="font-medium">Something went wrong loading this listing.</p>
   277→        <p className="mt-1 text-sm text-muted-foreground">{error}</p>
   278→        <Button variant="outline" className="mt-6 h-11 px-6" onClick={() => navigate('cars')}>Back to marketplace</Button>
   279→      </div>
   280→    )
   281→  }
   282→
   283→  if (!vehicle) {
   284→    return (
   285→      <div className="mx-auto w-full max-w-6xl space-y-6 px-4 py-6 sm:px-6" aria-busy="true">
   286→        <Skeleton className="h-8 w-72" />
   287→        <div className="grid gap-6 lg:grid-cols-3">
   288→          <Skeleton className="aspect-[16/10] w-full rounded-xl lg:col-span-2" />
   289→          <div className="space-y-4">
   290→            <Skeleton className="h-24 w-full rounded-xl" />
   291→            <Skeleton className="h-40 w-full rounded-xl" />
   292→          </div>
   293→        </div>
   294→      </div>
   295→    )
   296→  }
   297→
   298→  const title = vehicleTitle(vehicle)
   299→  const photos = vehicle.photos ?? []
   300→  const location = [vehicle.town, vehicle.county].filter(Boolean).join(', ')
   301→  const isDealer = vehicle.sellerType === 'DEALER'
   302→
   303→  const specs = [
   304→    { icon: Gauge, label: 'Mileage', value: `${vehicle.mileageKm.toLocaleString()} km` },
   305→    { icon: Fuel, label: 'Fuel', value: vehicle.fuel.charAt(0) + vehicle.fuel.slice(1).toLowerCase() },
   306→    { icon: Cog, label: 'Transmission', value: vehicle.transmission === 'AUTOMATIC' ? 'Automatic' : 'Manual' },
   307→    { icon: Car, label: 'Body', value: vehicle.bodyType.charAt(0) + vehicle.bodyType.slice(1).toLowerCase() },
   308→    { icon: Cog, label: 'Engine', value: vehicle.engineCc ? `${vehicle.engineCc.toLocaleString()} cc` : 'Not provided' },
   309→    { icon: Palette, label: 'Colour', value: vehicle.color },
   310→    { icon: MapPin, label: 'Location', value: location || 'Not set' },
   311→    { icon: CalendarDays, label: 'Listed', value: timeAgo(vehicle.createdAt) },
   312→    { icon: Hash, label: 'Ref', value: vehicle.refCode },
   313→    { icon: Eye, label: 'Views', value: vehicle.views.toLocaleString() },
   314→  ]
   315→
   316→  return (
   317→    <div className="mx-auto w-full max-w-6xl px-4 py-6 sm:px-6">
   318→      <Button variant="ghost" className="mb-4 h-11 px-3 -ml-3" onClick={() => navigate('cars')}>
   319→        <ArrowLeft className="h-4 w-4" /> Back to marketplace
   320→      </Button>
   321→
   322→      <div className="grid gap-6 lg:grid-cols-3">
   323→        {/* Gallery */}
   324→        <Reveal className="lg:col-span-2" blur={false}>
   325→          <div className="overflow-hidden rounded-xl border border-border bg-muted">
   326→            {photos.length > 0 ? (
   327→              <motion.img
   328→                key={activePhoto}
   329→                src={photos[activePhoto]?.url ?? photos[0].url}
   330→                alt={`${title}: photo ${activePhoto + 1} of ${photos.length}`}
   331→                className="aspect-[16/10] w-full object-cover"
   332→                initial={{ opacity: 0.35, scale: 1.035 }}
   333→                animate={{ opacity: 1, scale: 1 }}
   334→                transition={{ duration: 0.45, ease: [0.22, 0.61, 0.36, 1] }}
   335→              />
   336→            ) : (
   337→              <div className="flex aspect-[16/10] w-full items-center justify-center text-sm text-muted-foreground">
   338→                No photos provided
   339→              </div>
   340→            )}
   341→          </div>
   342→          {photos.length > 1 && (
   343→            <div className="mt-3 flex gap-2 overflow-x-auto pb-1 scrollbar-thin" role="tablist" aria-label="Vehicle photos">
   344→              {photos.map((p, i) => (
   345→                <motion.button
   346→                  key={p.url}
   347→                  role="tab"
   348→                  aria-selected={i === activePhoto}
   349→                  aria-label={`Show photo ${i + 1}`}
   350→                  onClick={() => setActivePhoto(i)}
   351→                  whileHover={{ scale: 1.05 }}
   352→                  whileTap={{ scale: 0.94 }}
   353→                  transition={{ type: 'spring', stiffness: 400, damping: 22 }}
   354→                  className={`h-16 w-24 shrink-0 overflow-hidden rounded-lg border-2 transition-colors ${
   355→                    i === activePhoto ? 'border-accent' : 'border-transparent hover:border-border'
   356→                  }`}
   357→                >
   358→                  <img src={p.url} alt="" className="h-full w-full object-cover" />
   359→                </motion.button>
   360→              ))}
   361→            </div>
   362→          )}
   363→
   364→          {/* Description */}
   365→          <Card className="mt-6">
   366→            <CardHeader>
   367→              <CardTitle className="text-lg">Description</CardTitle>
   368→            </CardHeader>
   369→            <CardContent>
   370→              <p className="whitespace-pre-line text-sm leading-relaxed text-foreground/90">{vehicle.description}</p>
   371→            </CardContent>
   372→          </Card>
   373→
   374→          <Alert className="mt-4 border-amber-600/30 bg-amber-50/60 text-amber-950 dark:border-amber-400/25 dark:bg-amber-400/10 dark:text-amber-200 [&>svg]:text-amber-700 dark:[&>svg]:text-amber-300">
   375→            <ShieldAlert className="h-4 w-4" />
   376→            <AlertTitle>Stay safe when buying</AlertTitle>
   377→            <AlertDescription>
   378→              Never pay a &ldquo;release fee&rdquo; or deposit to a stranger before seeing the vehicle. Inspect in
   379→              person (or via a trusted agent) before paying. Report suspicious listings using the Report button.
   380→            </AlertDescription>
   381→          </Alert>
   382→        </Reveal>
   383→
   384→        {/* Summary sidebar */}
   385→        <div className="space-y-4">
   386→          <Card>
   387→            <CardContent className="p-6">
   388→              <div className="flex flex-wrap items-center gap-2">
   389→                {vehicle.featured && <Badge className="border-none bg-accent text-accent-foreground">Featured</Badge>}
   390→                <Badge variant="secondary">{CONDITION_LABELS[vehicle.condition] ?? vehicle.condition}</Badge>
   391→                {vehicle.status !== 'PUBLISHED' && (
   392→                  <Badge variant={vehicle.status === 'SOLD' ? 'destructive' : 'outline'}>
   393→                    {VEHICLE_STATUS_LABELS[vehicle.status] ?? vehicle.status}
   394→                  </Badge>
   395→                )}
   396→                {isDealer ? (
   397→                  <Badge variant="outline" className="gap-1"><BadgeCheck className="h-3.5 w-3.5 text-primary" /> Dealer</Badge>
   398→                ) : (
   399→                  <Badge variant="outline">Private seller</Badge>
   400→                )}
   401→              </div>
   402→              <h1 className="mt-3 text-2xl font-bold leading-tight">{title}</h1>
   403→              <p className="mt-1 text-3xl font-extrabold text-primary">{money(vehicle.priceMinor, vehicle.currency)}</p>
   404→              <p className="mt-1 flex items-center gap-1 text-sm text-muted-foreground">
   405→                <MapPin className="h-4 w-4" /> {location || 'Kenya'}
   406→              </p>
   407→
   408→              <div className="mt-4 grid grid-cols-2 gap-2">
   409→                <Button
   410→                  className="h-11 flex-col gap-0.5"
   411→                  onClick={() => setEnquiryOpen(true)}
   412→                  disabled={vehicle.status === 'SOLD'}
   413→                >
   414→                  <Mail className="h-4 w-4" />
   415→                  <span className="text-xs">{vehicle.status === 'SOLD' ? 'Sold' : 'Enquire'}</span>
   416→                </Button>
   417→                <Button
   418→                  variant="outline"
   419→                  className="h-11 flex-col gap-0.5"
   420→                  onClick={toggleSaved}
   421→                  aria-pressed={savedIds.has(vehicle.id)}
   422→                >
   423→                  <Heart className={`h-4 w-4 ${savedIds.has(vehicle.id) ? 'fill-[#D9B878] text-[#D9B878]' : ''}`} />
   424→                  <span className="text-xs">{savedIds.has(vehicle.id) ? 'Saved' : 'Save'}</span>
   425→                </Button>
   426→                <Button variant="outline" className="h-11 flex-col gap-0.5" onClick={share}>
   427→                  <Share2 className="h-4 w-4" /><span className="text-xs">Share</span>
   428→                </Button>
   429→                <Button variant="outline" className="h-11 flex-col gap-0.5" onClick={() => setReportOpen(true)}>
   430→                  <Flag className="h-4 w-4" /><span className="text-xs">Report</span>
   431→                </Button>
   432→              </div>
   433→
   434→              {vehicle.org?.id && (
   435→                <div className="mt-2 grid gap-2 border-t border-border pt-3">
   436→                  <Button
   437→                    variant="outline"
   438→                    className="h-11 justify-start gap-2"
   439→                    onClick={() => setStorefrontAction('test-drive')}
   440→                    disabled={vehicle.status === 'SOLD'}
   441→                  >
   442→                    <CalendarCheck className="h-4 w-4 text-primary" /> Book test drive
   443→                  </Button>
   444→                  <Button
   445→                    variant="outline"
   446→                    className="h-11 justify-start gap-2"
   447→                    onClick={() => setStorefrontAction('message')}
   448→                    disabled={vehicle.status === 'SOLD'}
   449→                  >
   450→                    <MessageCircle className="h-4 w-4 text-primary" /> Ask about this vehicle
   451→                  </Button>
   452→                </div>
   453→              )}
   454→            </CardContent>
   455→          </Card>
   456→
   457→          {/* Seller card: honest about seller type */}
   458→          <Card>
   459→            <CardHeader>
   460→              <CardTitle className="text-base">{isDealer ? 'Sold by' : 'Private seller'}</CardTitle>
   461→            </CardHeader>
   462→            <CardContent className="space-y-2">
   463→              {isDealer && vehicle.org ? (
   464→                <>
   465→                  <p className="flex items-center gap-1.5 font-semibold">
   466→                    {vehicle.org.name}
   467→                    <BadgeCheck className="h-4.5 w-4.5 text-primary" aria-label="Verified dealer" />
   468→                    {(vehicle.org as { isDemo?: boolean }).isDemo && (
   469→                      <span className="inline-flex items-center rounded-full bg-amber-500/15 px-1.5 py-0.5 text-[10px] font-semibold uppercase tracking-wide text-amber-700 dark:text-amber-400" title="Demo storefront: sample data, not a real business">
   470→                        Demo
   471→                      </span>
   472→                    )}
   473→                  </p>
   474→                  <p className="flex items-center gap-1.5 text-sm text-muted-foreground">
   475→                    <MapPin className="h-4 w-4" />
   476→                    {[vehicle.org.location, vehicle.org.town, vehicle.org.county].filter(Boolean).join(', ')}
   477→                  </p>
   478→                  {vehicle.org.description && (
   479→                    <p className="text-sm text-muted-foreground">{vehicle.org.description}</p>
   480→                  )}
   481→                  {vehicle.org.phone && (
   482→                    <p className="flex items-center gap-1.5 pt-1 text-sm font-medium">
   483→                      <Phone className="h-4 w-4 text-primary" />
   484→                      <a href={`tel:${vehicle.org.phone}`} className="hover:underline">{vehicle.org.phone}</a>
   485→                    </p>
   486→                  )}
   487→                  <p className="text-xs text-muted-foreground">Business contact: call during working hours.</p>
   488→                </>
   489→              ) : (
   490→                <>
   491→                  <p className="font-semibold">{vehicle.owner?.name ?? 'Private seller'}</p>
   492→                  {vehicle.owner?.verifiedSeller && (
   493→                    <Badge className="mt-1 gap-1 border-transparent bg-primary/15 text-primary">
   494→                      <ShieldCheck className="h-3 w-3" aria-hidden /> Verified seller
   495→                    </Badge>
   496→                  )}
   497→                  <p className="flex items-center gap-1.5 text-sm text-muted-foreground">
   498→                    <MapPin className="h-4 w-4" /> {location || 'Kenya'}
   499→                  </p>
   500→                  <p className="text-xs text-muted-foreground">
   501→                    Contact details stay private. Send an enquiry and the seller can reply securely.
   502→                  </p>
   503→                </>
   504→              )}
   505→            </CardContent>
   506→          </Card>
   507→
   508→          {/* Specs */}
   509→          <Card>
   510→            <CardHeader>
   511→              <CardTitle className="text-base">Specification</CardTitle>
   512→            </CardHeader>
   513→            <CardContent className="grid grid-cols-2 gap-x-4 gap-y-3">
   514→              {specs.map((s) => (
   515→                <div key={s.label} className="flex items-start gap-2">
   516→                  <s.icon className="mt-0.5 h-4 w-4 shrink-0 text-muted-foreground" />
   517→                  <div className="min-w-0">
   518→                    <p className="text-xs text-muted-foreground">{s.label}</p>
   519→                    <p className="truncate text-sm font-medium" title={s.value}>{s.value}</p>
   520→                  </div>
   521→                </div>
   522→              ))}
   523→            </CardContent>
   524→          </Card>
   525→        </div>
   526→      </div>
   527→
   528→      {/* Similar listings */}
   529→      {similar.length > 0 && (
   530→        <section className="mt-10">
   531→          <Separator className="mb-6" />
   532→          <Reveal>
   533→            <h2 className="mb-4 text-xl font-bold">Similar vehicles</h2>
   534→          </Reveal>
   535→          <Stagger className="grid grid-cols-1 gap-4 sm:grid-cols-2 xl:grid-cols-4">
   536→            {similar.map((v) => (
   537→              <StaggerItem key={v.id}>
   538→                <VehicleCard v={v} />
   539→              </StaggerItem>
   540→            ))}
   541→          </Stagger>
   542→        </section>
   543→      )}
   544→
   545→      {/* Enquiry dialog */}
   546→      <Dialog open={enquiryOpen} onOpenChange={setEnquiryOpen}>
   547→        <DialogContent className="max-w-md">
   548→          <DialogHeader>
   549→            <DialogTitle>Enquire about the {title}</DialogTitle>
   550→            <DialogDescription>
   551→              Your message goes straight to {isDealer && vehicle.org ? vehicle.org.name : 'the seller'}. Ask about
   552→              availability, inspection or service history.
   553→            </DialogDescription>
   554→          </DialogHeader>
   555→          <div className="space-y-4">
   556→            <div className="space-y-1.5">
   557→              <Label htmlFor="enq-msg">Message</Label>
   558→              <Textarea
   559→                id="enq-msg"
   560→                rows={4}
   561→                placeholder="Hi, is this still available? I'd like to book a viewing…"
   562→                value={enquiryMsg}
   563→                onChange={(e) => setEnquiryMsg(e.target.value)}
   564→              />
   565→            </div>
   566→            <label className="flex cursor-pointer items-start gap-2 text-sm">
   567→              <Checkbox
   568→                checked={phoneShare}
   569→                onCheckedChange={(c) => setPhoneShare(c === true)}
   570→                className="mt-0.5"
   571→              />
   572→              <span>Share my phone number with the seller for faster replies</span>
   573→            </label>
   574→            {phoneShare && (
   575→              <div className="space-y-1.5">
   576→                <Label htmlFor="enq-phone">Phone number</Label>
   577→                <Input
   578→                  id="enq-phone"
   579→                  inputMode="tel"
   580→                  placeholder="+254 7XX XXX XXX"
   581→                  value={phone}
   582→                  onChange={(e) => setPhone(e.target.value)}
   583→                />
   584→                <p className="text-xs text-muted-foreground">Only visible to this seller.</p>
   585→              </div>
   586→            )}
   587→          </div>
   588→          <DialogFooter>
   589→            <Button variant="ghost" className="h-11" onClick={() => setEnquiryOpen(false)}>Cancel</Button>
   590→            <Button className="h-11 px-5" onClick={sendEnquiry} disabled={sendingEnquiry}>
   591→              <MessageSquareText className="h-4 w-4" /> {sendingEnquiry ? 'Sending…' : 'Send enquiry'}
   592→            </Button>
   593→          </DialogFooter>
   594→        </DialogContent>
   595→      </Dialog>
   596→
   597→      {/* Storefront dialogs: dealer listings */}
   598→      {vehicle.org?.id && (
   599→        <>
   600→          <MessageDialog
   601→            open={storefrontAction === 'message'}
   602→            onOpenChange={(open) => { if (!open) setStorefrontAction(null) }}
   603→            org={{ id: vehicle.org.id, name: vehicle.org.name }}
   604→            vehicleId={vehicle.id}
   605→            kind="VEHICLE_ENQUIRY"
   606→            subject={`Enquiry: ${title}`}
   607→          />
   608→          <TestDriveDialog
   609→            open={storefrontAction === 'test-drive'}
   610→            onOpenChange={(open) => { if (!open) setStorefrontAction(null) }}
   611→            org={{ id: vehicle.org.id, name: vehicle.org.name }}
   612→            vehicleId={vehicle.id}
   613→            vehicleLabel={title}
   614→          />
   615→        </>
   616→      )}
   617→
   618→      {/* Report dialog */}
   619→      <Dialog open={reportOpen} onOpenChange={setReportOpen}>
   620→        <DialogContent className="max-w-md">
   621→          <DialogHeader>
   622→            <DialogTitle>Report this listing</DialogTitle>
   623→            <DialogDescription>
   624→              Tell us what looks wrong. Reports are reviewed by our moderation team.
   625→            </DialogDescription>
   626→          </DialogHeader>
   627→          <div className="space-y-4">
   628→            <div className="space-y-1.5">
   629→              <Label>Reason</Label>
   630→              <Select value={reportReason} onValueChange={setReportReason}>
   631→                <SelectTrigger className="w-full"><SelectValue placeholder="Choose a reason" /></SelectTrigger>
   632→                <SelectContent>
   633→                  {REPORT_REASONS.map((r) => <SelectItem key={r} value={r}>{r}</SelectItem>)}
   634→                </SelectContent>
   635→              </Select>
   636→            </div>
   637→            <div className="space-y-1.5">
   638→              <Label htmlFor="report-details">Details (optional)</Label>
   639→              <Textarea
   640→                id="report-details"
   641→                rows={3}
   642→                placeholder="Anything that helps us investigate…"
   643→                value={reportDetails}
   644→                onChange={(e) => setReportDetails(e.target.value)}
   645→              />
   646→            </div>
   647→          </div>
   648→          <DialogFooter>
   649→            <Button variant="ghost" className="h-11" onClick={() => setReportOpen(false)}>Cancel</Button>
   650→            <Button variant="destructive" className="h-11 px-5" onClick={sendReport} disabled={sendingReport}>
   651→              {sendingReport ? 'Submitting…' : 'Submit report'}
   652→            </Button>
   653→          </DialogFooter>
   654→        </DialogContent>
   655→      </Dialog>
   656→    </div>
   657→  )
   658→}
