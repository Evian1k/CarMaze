    79→    79→function draftFromQuery(query: URLSearchParams): FilterDraft {
    80→    80→  const minor = (key: string) => {
    81→    81→    const raw = query.get(key)
    82→    82→    if (!raw) return ''
    83→    83→    const n = Number(raw)
    84→    84→    return Number.isFinite(n) && n > 0 ? String(Math.round(n / 100)) : ''
    85→    85→  }
    86→    86→  return {
    87→    87→    q: query.get('q') ?? '',
    88→    88→    make: query.get('make') ?? '',
    89→    89→    model: query.get('model') ?? '',
    90→    90→    bodyType: query.get('bodyType') ?? '',
    91→    91→    county: query.get('county') ?? '',
    92→    92→    fuel: query.get('fuel') ?? '',
    93→    93→    transmission: query.get('transmission') ?? '',
    94→    94→    condition: query.get('condition') ?? '',
    95→    95→    sellerType: query.get('sellerType') ?? '',
    96→    96→    verified: query.get('verified') === '1',
    97→    97→    minPrice: minor('minPrice'),
    98→    98→    maxPrice: minor('maxPrice'),
    99→    99→    minYear: query.get('minYear') ?? '',
   100→   100→    maxYear: query.get('maxYear') ?? '',
   101→   101→    maxMileage: query.get('maxMileage') ?? '',
   102→   102→    sort: query.get('sort') || 'newest',
   103→   103→    page: query.get('page') || '1',
   104→   104→    org: query.get('org') ?? '',
   105→   105→  }
   106→   106→}
   107→   107→
   108→   108→/** Convert the editable draft into URL query params (prices back to minor units). */
   109→   109→function draftToParams(d: FilterDraft): Record<string, string | number | undefined> {
   110→   110→  const num = (s: string) => {
   111→   111→    const n = Number(s)
   112→   112→    return s !== '' && Number.isFinite(n) && n > 0 ? n : undefined
   113→   113→  }
   114→   114→  return {
   115→   115→    q: d.q.trim() || undefined,
   116→   116→    make: d.make || undefined,
   117→   117→    model: d.model.trim() || undefined,
   118→   118→    bodyType: d.bodyType || undefined,
   119→   119→    county: d.county || undefined,
   120→   120→    fuel: d.fuel || undefined,
   121→   121→    transmission: d.transmission || undefined,
   122→   122→    condition: d.condition || undefined,
   123→   123→    sellerType: d.sellerType || undefined,
   124→   124→    verified: d.verified ? '1' : undefined,
   125→   125→    minPrice: num(d.minPrice) ? Math.round(Number(d.minPrice) * 100) : undefined,
   126→   126→    maxPrice: num(d.maxPrice) ? Math.round(Number(d.maxPrice) * 100) : undefined,
   127→   127→    minYear: num(d.minYear),
   128→   128→    maxYear: num(d.maxYear),
   129→   129→    maxMileage: num(d.maxMileage),
   130→   130→    sort: d.sort && d.sort !== 'newest' ? d.sort : undefined,
   131→   131→    org: d.org || undefined,
   132→   132→    // page intentionally omitted → resets to 1 whenever filters change
   133→   133→  }
   134→   134→}
