    98→function parseSellerDocs(raw: string | null | undefined): SellerAppDoc[] {
    99→  if (!raw) return []
   100→  try {
   101→    const parsed: unknown = JSON.parse(raw)
   102→    if (!Array.isArray(parsed)) return []
   103→    return parsed.filter(
   104→      (d): d is SellerAppDoc =>
   105→        typeof d === 'object' && d !== null && typeof (d as SellerAppDoc).url === 'string'
   106→    )
   107→  } catch {
   108→    return []
   109→  }
   110→}

   112→/** Fetches the signed-in user's most recent SELLER application.
   113→ * `enabled` gates the request (only when signed in and not already verified). */
   114→export function useSellerApplication(enabled: boolean) {
   115→  const [app, setApp] = useState<SellerApplication | null>(null)
   116→  const [loading, setLoading] = useState(enabled)
   117→  const [error, setError] = useState<string | null>(null)
   118→  const [tick, setTick] = useState(0)
   119→
   120→  useEffect(() => {
   121→    if (!enabled) return
   122→    let cancelled = false
   123→    const run = async () => {
   124→      try {
   125→        const d = await apiGet<{ applications: SellerApplication[] }>('/api/applications')
   126→        if (cancelled) return
   127→        const sellers = d.applications
   128→          .filter((a) => a.type === 'SELLER')
   129→          .sort((a, b) => new Date(b.createdAt).getTime() - new Date(a.createdAt).getTime())
   130→        setApp(sellers[0] ?? null)
   131→        setError(null)
   132→      } catch (e) {
   133→        if (!cancelled) setError((e as { message?: string })?.message ?? 'Could not load verification status.')
   134→      } finally {
   135→        if (!cancelled) setLoading(false)
   136→      }
   137→    }
   138→    run()
   139→    return () => { cancelled = true }
   140→  }, [enabled, tick])
   141→
   142→  // Event-context refresh: safe to set state here (never called from an effect body).
   143→  const reload = useCallback(() => {
   144→    if (enabled) setLoading(true)
   145→    setTick((t) => t + 1)
   146→  }, [enabled])
   147→
   148→  // Derived reset for the disabled state; no synchronous setState inside effects.
   149→  return { app: enabled ? app : null, loading: enabled && loading, error: enabled ? error : null, reload }
   150→}

   263→async function uploadDocs(files: FileList | null, current: number, onDoc: (d: SellerAppDoc) => void) {
   264→  if (!files || files.length === 0) return
   265→  const slots = MAX_DOCS - current
   266→  const list = Array.from(files).slice(0, slots)
   267→  if (list.length === 0) { toast.warning(`Maximum of ${MAX_DOCS} documents`); return }
   268→  try {
   269→    for (const file of list) {
   270→      const fd = new FormData()
   271→      fd.append('file', file)
   272→      fd.append('kind', 'document')
   273→      const res = await apiUpload<{ url: string; name: string }>('/api/upload', fd)
   274→      onDoc({ name: res.name || file.name, url: res.url })
   275→    }
   276→    toast.success('Document uploaded')
   277→  } catch (e) {
   278→    toast.error(e instanceof Error ? e.message : 'Upload failed: please try again.')
   279→  }
   280→}

   345→function ActionRequired({ app, onDone }: { app: SellerApplication; onDone: () => void }) {
   346→  const infoEvent = [...app.events].reverse().find((e) => e.type === 'INFO_REQUESTED')
   347→  const onFile = parseSellerDocs(app.documents)
   348→  const [docs, setDocs] = useState<SellerAppDoc[]>([])
   349→  const [uploading, setUploading] = useState(false)
   350→  const [notes, setNotes] = useState('')
   351→  const [submitting, setSubmitting] = useState(false)
   352→  const [apiError, setApiError] = useState<string | null>(null)
   353→
   354→  async function handleFiles(files: FileList | null) {
   355→    setUploading(true)
   356→    await uploadDocs(files, docs.length, (d) => setDocs((prev) => (prev.length < MAX_DOCS ? [...prev, d] : prev)))
   357→    setUploading(false)
   358→  }
   359→
   360→  async function resubmit() {
   361→    if (docs.length === 0 && notes.trim().length < 5) {
   362→      toast.error('Add a note or attach the requested documents before resubmitting.')
   363→      return
   364→    }
   365→    setSubmitting(true)
   366→    setApiError(null)
   367→    try {
   368→      await apiSend(`/api/applications/${app.id}`, 'PATCH', {
   369→        action: 'resubmit',
   370→        documents: docs.length > 0 ? docs : undefined,
   371→        notes: notes.trim() || undefined,
   372→      })
   373→      toast.success('Information resubmitted: the review team has been notified.')
   374→      onDone()
   375→    } catch (e) {
   376→      const message = e instanceof Error ? e.message : 'Could not resubmit.'
   377→      setApiError(message)
   378→      toast.error(message)
   379→    } finally {
   380→      setSubmitting(false)
   381→    }
   382→  }
   383→
   384→  return (
   385→    <div className="space-y-5">
   386→      <ApplicationHeader app={app} />
   387→
   388→      {/* The reviewer's request: prominent */}
   389→      <div className="flex items-start gap-3 rounded-xl border border-amber-300 bg-amber-50 p-4 text-amber-900 dark:border-amber-500/40 dark:bg-amber-500/10 dark:text-amber-200">
   390→        <AlertTriangle className="mt-0.5 h-5 w-5 shrink-0" aria-hidden="true" />
   391→        <div className="min-w-0 text-sm">
   392→          <p className="font-semibold">The review team requested more information</p>
   393→          {infoEvent?.note && <p className="mt-1 leading-relaxed">{infoEvent.note}</p>}
   394→          {app.correctionsDueAt && (
   395→            <p className="mt-1 text-xs opacity-90">
   396→              Please respond before {dateLabel(app.correctionsDueAt)} to keep your application in the queue.
   397→            </p>
   398→          )}
   399→        </div>
   400→      </div>
   401→
   402→      <DocUploader
   403→        idPrefix="sva-rs"
   404→        docs={docs}
   405→        uploading={uploading}
   406→        onAdd={handleFiles}
   407→        onRemove={(url) => setDocs((prev) => prev.filter((d) => d.url !== url))}
   408→        title="Attach the requested documents"
   409→        hint={
   410→          onFile.length > 0
   411→            ? `${onFile.length} document${onFile.length === 1 ? ' is' : 's are'} already on file: new uploads replace them, so attach everything the reviewer needs.`
   412→            : 'Attach exactly what the reviewer asked for.'
   413→        }
   414→      />
   415→
   416→      <div className="grid gap-1.5">
   417→        <Label htmlFor="sva-rs-notes">Note for the review team</Label>
   418→        <Textarea
   419→          id="sva-rs-notes"
   420→          rows={3}
   421→          maxLength={2000}
   422→          value={notes}
   423→          onChange={(e) => setNotes(e.target.value)}
   424→          placeholder="Explain what you corrected or attached…"
   425→        />
   426→      </div>
   427→
   428→      {apiError && (
   429→        <Alert className="border-destructive/40 bg-destructive/5 text-destructive [&>svg]:text-destructive">
   430→          <AlertTriangle className="h-4 w-4" />
   431→          <AlertTitle>Couldn&apos;t resubmit</AlertTitle>
   432→          <AlertDescription>{apiError}</AlertDescription>
   433→        </Alert>
   434→      )}
   435→
   436→      <Button onClick={resubmit} disabled={submitting} className="min-h-11 px-6">
   437→        {submitting ? <Loader2 className="h-4 w-4 animate-spin" /> : <RefreshCw className="h-4 w-4" />} Resubmit for review
   438→      </Button>
   439→
   440→      <EventTimeline events={app.events} />
   441→    </div>
   442→  )
   443→}

   477→  async function submit() {
   478→    if (form.orgName.trim().length < 2) { toast.error('Enter your full legal name.'); return }
   479→    if (form.regNo.trim().length < 3) { toast.error('Enter your national ID or passport number.'); return }
   480→    if (!form.county) { toast.error('Select your county.'); return }
   481→    if (form.phone.trim().length < 9) { toast.error('Enter a valid phone number.'); return }
   482→    if (form.representative.trim().length < 2) { toast.error('Enter the authorised representative name.'); return }
   483→    if (docs.length === 0) {
   484→      toast.error('Attach at least one document: your government ID and a logbook/ownership document or authority-to-sell letter.')
   485→      return
   486→    }
   487→    if (!consent) { toast.error('Please confirm your information is true and accept the verification terms and privacy notice.'); return }
   488→    setSubmitting(true)
   489→    setApiError(null)
   490→    try {
   491→      const res = await apiSend<{ application: SellerApplication }>('/api/applications', 'POST', {
   492→        type: 'SELLER',
   493→        status: 'SUBMITTED',
   494→        orgName: form.orgName.trim(),
   495→        regNo: form.regNo.trim(),
   496→        county: form.county,
   497→        town: form.town.trim() || undefined,
   498→        phone: form.phone.trim(),
   499→        email: user.email || undefined,
   500→        representative: form.representative.trim(),
   501→        documents: docs,
   502→        consent: true,
   503→        notes: form.notes.trim() || undefined,
   504→      })
   505→      toast.success(`Seller verification submitted: ${res.application.refCode}. Our review team checks documents manually.`)
   506→      setOpen(false)
   507→      onDone()
   508→    } catch (e) {
   509→      const message = e instanceof Error ? e.message : 'Could not submit the application.'
   510→      setApiError(message)
   511→      toast.error(message)
   512→    } finally {
   513→      setSubmitting(false)
   514→    }
   515→  }
   516→
   517→  return (
   518→    <div className="space-y-4">
   519→      {previousApp?.status === 'DRAFT' && (
   520→        <Alert className="border-accent/50 bg-accent/10">
   521→          <Info className="h-4 w-4" />
   522→          <AlertTitle>Draft {previousApp.refCode} saved</AlertTitle>
   523→          <AlertDescription>
   524→            A draft was saved earlier but was never sent to the review team. Submit a fresh application below.
   525→          </AlertDescription>
   526→        </Alert>
   527→      )}
   528→
   529→      {closed && previousApp && (
   530→        <Alert className="border-destructive/40 bg-destructive/5 text-destructive [&>svg]:text-destructive">
   531→          <AlertTriangle className="h-4 w-4" />
   532→          <AlertTitle>
   533→            {previousApp.refCode} was {APPLICATION_STATUS_LABELS[previousApp.status]?.toLowerCase() ?? previousApp.status.toLowerCase()}
   534→          </AlertTitle>
   535→          <AlertDescription>
   536→            {previousApp.decisionNote
   537→              ? `Reviewer note: ${previousApp.decisionNote}`
   538→              : 'You can correct the details and submit a fresh application.'}
   539→          </AlertDescription>
   540→        </Alert>
   541→      )}
   542→
   543→      <div className="flex flex-wrap items-center justify-between gap-3">
   544→        <p className="max-w-md text-sm text-muted-foreground">
   545→          {closed
   546→            ? 'Correct the issues noted above, then submit a fresh application for review.'
   547→            : 'Verify your identity and authority to sell; approved accounts carry the verified trust marker on their listings.'}
   548→        </p>
   549→        <Button
   550→          variant={closed ? 'default' : 'outline'}
   551→          className="h-11 px-5 sm:h-9"
   552→          onClick={() => setOpen((o) => !o)}
   553→          aria-expanded={open}
   554→        >
   555→          {closed ? 'Correct and resubmit' : open ? 'Close form' : 'Start verification'}
   556→          <ChevronDown className={cn('h-4 w-4 transition-transform', open && 'rotate-180')} />
   557→        </Button>
   558→      </div>
   559→
   560→      {open && (
   561→        <div className="space-y-4 rounded-xl border border-border p-4">
   562→          <div className="grid gap-4 sm:grid-cols-2">
   563→            <div className="space-y-1.5">
   564→              <Label htmlFor="sva-name">Full legal name *</Label>
   565→              <Input id="sva-name" placeholder="As it appears on your ID" value={form.orgName} onChange={(e) => set('orgName', e.target.value)} />
   566→            </div>
   567→            <div className="space-y-1.5">
   568→              <Label htmlFor="sva-reg">National ID / passport number *</Label>
   569→              <Input id="sva-reg" placeholder="e.g. 12345678" value={form.regNo} onChange={(e) => set('regNo', e.target.value)} />
   570→            </div>
   571→            <div className="space-y-1.5">
   572→              <Label>County *</Label>
   573→              <Select value={form.county || undefined} onValueChange={(v) => set('county', v)}>
   574→                <SelectTrigger className="w-full"><SelectValue placeholder="Select county" /></SelectTrigger>
   575→                <SelectContent>
   576→                  {KENYA_COUNTIES.map((c) => <SelectItem key={c} value={c}>{c}</SelectItem>)}
   577→                </SelectContent>
   578→              </Select>
   579→            </div>
   580→            <div className="space-y-1.5">
   581→              <Label htmlFor="sva-town">Town / area</Label>
   582→              <Input id="sva-town" placeholder="e.g. Syokimau" value={form.town} onChange={(e) => set('town', e.target.value)} />
   583→            </div>
   584→            <div className="space-y-1.5">
   585→              <Label htmlFor="sva-phone">Phone *</Label>
   586→              <Input id="sva-phone" inputMode="tel" placeholder="e.g. 0712 345 678" value={form.phone} onChange={(e) => set('phone', e.target.value)} />
   587→            </div>
   588→            <div className="space-y-1.5">
   589→              <Label htmlFor="sva-rep">Authorised representative *</Label>
   590→              <Input id="sva-rep" value={form.representative} onChange={(e) => set('representative', e.target.value)} />
   591→              <p className="text-xs text-muted-foreground">For private sellers this is you; the person selling the vehicle(s).</p>
   592→            </div>
   593→          </div>
   594→
   595→          <DocUploader
   596→            idPrefix="sva-new"
   597→            docs={docs}
   598→            uploading={uploading}
   599→            onAdd={handleFiles}
   600→            onRemove={(url) => setDocs((prev) => prev.filter((d) => d.url !== url))}
   601→            title="Identity & ownership documents"
   602→            hint="Attach your government ID (national ID or passport) plus a logbook/ownership document: or an authority-to-sell letter for each vehicle you're listing."
   603→          />
   604→
   605→          <div className="grid gap-1.5">
   606→            <Label htmlFor="sva-notes">Notes for the review team (optional)</Label>
   607→            <Textarea
   608→              id="sva-notes"
   609→              rows={3}
   610→              maxLength={2000}
   611→              value={form.notes}
   612→              onChange={(e) => set('notes', e.target.value)}
   613→              placeholder="Anything that helps the reviewer: e.g. the vehicle(s) you plan to list…"
   614→            />
   615→          </div>
   616→
   617→          <label className="flex items-start gap-2.5 text-sm leading-relaxed" htmlFor="sva-consent">
   618→            <Checkbox id="sva-consent" className="mt-0.5" checked={consent} onCheckedChange={(v) => setConsent(v === true)} />
   619→            <span>I confirm the information and documents are true and I accept the verification terms and privacy notice.</span>
   620→          </label>
   621→
   622→          {apiError && (
   623→            <Alert className="border-destructive/40 bg-destructive/5 text-destructive [&>svg]:text-destructive">
   624→              <AlertTriangle className="h-4 w-4" />
   625→              <AlertTitle>Couldn&apos;t submit the application</AlertTitle>
   626→              <AlertDescription>{apiError}</AlertDescription>
   627→            </Alert>
   628→          )}
   629→
   630→          <p className="rounded-lg bg-muted px-3 py-2 text-xs text-muted-foreground">
   631→            Verification is completed by our review team after document checks; it is never automatic.
   632→            Initial review target: 24 business hours.
   633→          </p>
   634→
   635→          <Button onClick={submit} disabled={submitting} className="min-h-11 px-6">
   636→            {submitting ? <Loader2 className="h-4 w-4 animate-spin" /> : <ShieldCheck className="h-4 w-4" />} Submit for verification
   637→          </Button>
   638→        </div>
   639→      )}
   640→    </div>
   641→  )
   642→}

   663→export function VerificationPanel() {
   664→  const { user, refresh } = useAuth()
   665→  const verified = hasAnyRole(user, 'SELLER_VERIFIED')
   666→  const { app, loading, error, reload } = useSellerApplication(!!user && !verified)
   667→
   668→  const onActionDone = () => {
   669→    reload()
   670→    void refresh()
   671→  }
   672→
   673→  if (!user) return null
   674→
   675→  if (verified) return <VerifiedCard />
   676→
   677→  return (
   678→    <Card>
   679→      <CardHeader className="pb-3">
   680→        <CardTitle className="flex items-center gap-2 text-base">
   681→          <ShieldCheck className="h-5 w-5 text-primary" aria-hidden="true" /> Seller verification
   682→        </CardTitle>
   683→        <CardDescription>
   684→          A one-time manual document review adds the verified trust marker buyers look for.
   685→        </CardDescription>
   686→      </CardHeader>
   687→      <CardContent>
   688→        {loading ? (
   689→          <div className="space-y-3" aria-busy="true">
   690→            <Skeleton className="h-5 w-52" />
   691→            <Skeleton className="h-24 w-full rounded-xl" />
   692→          </div>
   693→        ) : error ? (
   694→          <Alert className="border-destructive/40 bg-destructive/5 text-destructive [&>svg]:text-destructive">
   695→            <AlertTriangle className="h-4 w-4" />
   696→            <AlertTitle>Couldn&apos;t load your verification status</AlertTitle>
   697→            <AlertDescription className="flex flex-wrap items-center gap-3">
   698→              <span>{error}</span>
   699→              <Button type="button" variant="outline" size="sm" className="h-9" onClick={reload}>
   700→                <RefreshCw className="h-4 w-4" /> Try again
   701→              </Button>
   702→            </AlertDescription>
   703→          </Alert>
   704→        ) : !app || app.status === 'DRAFT' ? (
   705→          <NotStarted user={user} previousApp={app} onDone={onActionDone} />
   706→        ) : app.status === 'MORE_INFO_REQUIRED' ? (
   707→          <ActionRequired app={app} onDone={onActionDone} />
   708→        ) : IN_REVIEW_STATUSES.includes(app.status) ? (
   709→          <InProgress app={app} />
   710→        ) : app.status === 'APPROVED' ? (
   711→          <ApprovedPendingRole onRefresh={() => void refresh()} />
   712→        ) : (
   713→          <NotStarted user={user} previousApp={app} onDone={onActionDone} />
   714→        )}
   715→      </CardContent>
   716→    </Card>
   717→  )
   718→}
