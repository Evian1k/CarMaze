   120→function ProgrammeTab({ orgId }: { orgId: string }) {

   122→  const [error, setError] = useState<string | null>(null)
   123→  const [builderOpen, setBuilderOpen] = useState(false)
   124→  const [editTier, setEditTier] = useState<Tier | null>(null)
   125→  const [archiveTier, setArchiveTier] = useState<Tier | null>(null)
   126→  const [redeemFor, setRedeemFor] = useState<Member | null>(null)
   127→
   128→  const load = useCallback(() => {
   129→    apiGet<ManagerData>(`/api/org/memberships?orgId=${orgId}`)
   130→      .then((d) => { setData(d); setError(null) })
   131→      .catch((e) => setError(e instanceof Error ? e.message : 'Could not load the membership programme'))
   132→  }, [orgId])
   133→
   134→  useEffect(() => { load() }, [load])
   135→
   136→  async function toggleTierActive(tier: Tier, active: boolean) {
   137→    try {
   138→      await apiSend(`/api/org/memberships/${tier.id}`, 'PATCH', { active })
   139→      toast.success(active ? `${tier.name} is live again` : `${tier.name} hidden from discovery`)
   140→      load()
   141→    } catch (e) {
   142→      toast.error(e instanceof Error ? e.message : 'Could not update the tier')
   143→    }
   144→  }
   145→
   146→  async function doArchive(tier: Tier) {
   147→    try {
   148→      await apiSend(`/api/org/memberships/${tier.id}`, 'DELETE')
   149→      toast.success(`${tier.name} archived: existing members keep their terms`)
   150→      load()
   151→    } catch (e) {
   152→      toast.error(e instanceof Error ? e.message : 'Could not archive the tier')
   153→    }
   154→  }
   155→
   156→  async function recordRedemption(membershipId: string, benefit: string, refType?: string) {
   157→    try {
   158→      await apiSend('/api/org/memberships/redeem', 'POST', { membershipId, benefit, ...(refType ? { refType } : {}) })
   159→      toast.success('Benefit redemption recorded')
   160→      load()
   161→      return true
   162→    } catch (e) {
   163→      const code = (e as { code?: string }).code
   164→      toast.error(e instanceof Error ? e.message : 'Could not record the redemption', {
   165→        description: code === 'ALREADY_REDEEMED' ? 'This benefit was already redeemed for the current period.' : undefined,
   166→      })
   167→      return false
   168→    }
   169→  }
   170→
   171→  if (error) {
   172→    return (
   173→      <Card>
   174→        <CardContent className="flex flex-col items-center gap-3 py-10 text-center">

   178→          <Button variant="outline" className="min-h-11" onClick={() => { setData(null); load() }}><RefreshCw className="h-4 w-4" /> Try again</Button>

   199→        <StatCard icon={<CalendarClock className="h-5 w-5" />} label="Pending payments" value={String(data.revenue.pendingCount)} hint="Activate after M-Pesa confirmation" />

   204→      </p>
   205→
   206→      {/* Tiers */}
   207→      <section aria-label="Membership tiers" className="space-y-3">
   208→        <div className="flex flex-wrap items-center justify-between gap-2">
   209→          <h3 className="text-lg font-semibold">Membership tiers</h3>
   210→          <Button className="min-h-11" onClick={() => setBuilderOpen(true)}><Plus className="h-4 w-4" /> New tier</Button>
   211→        </div>

   213→          <div className="rounded-xl border border-dashed p-8 text-center">
   214→            <Gift className="mx-auto h-8 w-8 text-muted-foreground" />
   215→            <p className="mt-2 font-medium">No tiers yet</p>
   216→            <p className="mx-auto mt-1 max-w-md text-sm text-muted-foreground">
   217→              Build your first tier: name it, set the price and benefits, and it appears instantly on the public
   218→              Business Memberships page for customers to subscribe to.
   219→            </p>
   220→            <Button variant="outline" className="mt-4 min-h-11" onClick={() => setBuilderOpen(true)}>Create a tier</Button>
   221→          </div>
   222→        ) : (
   223→          <div className="grid gap-4 lg:grid-cols-2">
   224→            {data.tiers.map((t) => (
   225→              <Card key={t.id} className={cn(!t.active && 'opacity-70')}>

   227→                  <div className="flex flex-wrap items-center justify-between gap-2">

   268→                  </div>
   269→                </CardContent>

   278→        <h3 className="text-lg font-semibold">Members</h3>

   282→            <p className="mt-2 font-medium">No members yet</p>

   284→              When customers subscribe to your tiers, they appear here with their billing status and benefit usage.
   285→            </p>
   286→          </div>
   287→        ) : (
   288→          <Card>
   289→            <CardContent className="p-0">
   290→              <div className="overflow-x-auto scrollbar-thin">
   291→                <Table>
   292→                  <TableHeader>

   296→                      <TableHead>Status</TableHead>

   298→                      <TableHead>Recent redemptions</TableHead>

   326→                                  <Stamp className="mr-1 inline h-3 w-3 text-primary" aria-hidden />
   327→                                  {r.benefit} <span className="text-muted-foreground">({r.periodKey})</span>
   328→                                </p>
   329→                              ))}
   330→                              {m.redemptions.length > 2 && (
   331→                                <p className="text-xs text-muted-foreground">+{m.redemptions.length - 2} earlier</p>
   332→                              )}
   333→                            </div>
   334→                          )}
   335→                        </TableCell>
   336→                        <TableCell className="text-right">

   338→                            variant="outline"
   339→                            size="sm"
   340→                            className="min-h-11"

   357→      {/* Dialogs */}

   363→        />
   364→      )}
   365→      {editTier && (
   366→        <TierBuilderDialog
   367→          orgId={orgId}
   368→          tier={editTier}
   369→          onClose={() => setEditTier(null)}
   370→          onSaved={() => { setEditTier(null); load() }}
   371→        />
   372→      )}
   373→      {archiveTier && (
   374→        <AlertDialog open onOpenChange={(o) => { if (!o) setArchiveTier(null) }}>
   375→          <AlertDialogContent>
   376→            <AlertDialogHeader>
   377→              <AlertDialogTitle>Archive “{archiveTier.name}”?</AlertDialogTitle>
   378→              <AlertDialogDescription>
   379→                The tier disappears from public discovery immediately. Existing members keep their
   380→                benefits for the period they paid for. You can re-publish it any time by switching
   381→                the archived tier back on.
   382→              </AlertDialogDescription>
   383→            </AlertDialogHeader>
   384→            <AlertDialogFooter>
   385→              <AlertDialogCancel>Keep tier</AlertDialogCancel>
   386→              <AlertDialogAction
   387→                className="bg-destructive text-white hover:bg-destructive/90"
   388→                onClick={async (e) => { e.preventDefault(); await doArchive(archiveTier); setArchiveTier(null) }}
   389→              >
   390→                Archive tier
   391→              </AlertDialogAction>
   392→            </AlertDialogFooter>
   393→          </AlertDialogContent>
   394→        </AlertDialog>
   395→      )}
   396→      {redeemFor && data.tiers.length > 0 && (
   397→        <RedeemDialog
   398→          member={redeemFor}
   399→          tier={data.tiers.find((t) => t.name === redeemFor.tier) ?? null}

   401→          onSubmit={async (benefit, refType) => {
   402→            const okResult = await recordRedemption(redeemFor.id, benefit, refType)
   403→            if (okResult) setRedeemFor(null)
   404→          }}
   405→        />
   406→      )}
   407→    </div>
   408→  )
   409→}

   441→function TierBuilderDialog({ orgId, tier, onClose, onSaved }: {
   442→  orgId: string
   443→  tier?: Tier
   444→  onClose: () => void
   445→  onSaved: () => void
   446→}) {
   447→  const editing = Boolean(tier)
   448→  const [form, setForm] = useState<BuilderForm>({
   449→    name: tier?.name ?? '',
   450→    description: tier?.description ?? '',
   451→    priceMonthly: tier ? String(tier.priceMonthlyMinor / 100) : '',
   452→    priceYearly: tier?.priceYearlyMinor != null ? String(tier.priceYearlyMinor / 100) : '',
   453→    benefitsText: tier ? tier.benefits.join('\n') : '',
   454→    memberLimit: tier?.memberLimit != null ? String(tier.memberLimit) : '',
   455→    discountCapPct: tier?.discountCapPct != null ? String(tier.discountCapPct) : '',
   456→    branchScope: tier?.branchScope ?? 'ALL',
   457→    branchIds: tier?.branches ?? [],
   458→    terms: tier?.terms ?? '',
   459→  })
   460→  const [branches, setBranches] = useState<BranchRow[] | null>(null)
   461→  const [busy, setBusy] = useState(false)
   462→
   463→  useEffect(() => {
   464→    apiGet<{ branches: BranchRow[] }>(`/api/supply/branches?orgId=${orgId}`)
   465→      .then((d) => setBranches(d.branches))
   466→      .catch(() => setBranches([]))
   467→  }, [orgId])
   468→
   469→  const benefits = useMemo(
   470→    () => form.benefitsText.split('\n').map((s) => s.trim()).filter(Boolean).slice(0, 12),
   471→    [form.benefitsText]
   472→  )
   473→  const monthlyMinor = kesToMinor(form.priceMonthly)
   474→  const yearlyMinor = form.priceYearly.trim() === '' ? null : kesToMinor(form.priceYearly)
   475→  const limitNum = form.memberLimit.trim() === '' ? null : Number(form.memberLimit)
   476→  const capNum = form.discountCapPct.trim() === '' ? null : Number(form.discountCapPct)
   477→  const valid =
   478→    form.name.trim().length >= 2 &&
   479→    monthlyMinor !== null &&
   480→    benefits.length >= 1 &&
   481→    (limitNum === null || (Number.isInteger(limitNum) && limitNum >= 1)) &&
   482→    (capNum === null || (Number.isInteger(capNum) && capNum >= 0 && capNum <= 80)) &&
   483→    (form.branchScope === 'ALL' || form.branchIds.length > 0)
   484→
   485→  async function save() {
   486→    if (monthlyMinor === null) return
   487→    setBusy(true)
   488→    try {
   489→      if (editing && tier) {
   490→        await apiSend(`/api/org/memberships/${tier.id}`, 'PATCH', {
   491→          name: form.name.trim(),
   492→          description: form.description.trim() || null,
   493→          priceMonthlyMinor: monthlyMinor,
   494→          priceYearlyMinor: yearlyMinor,
   495→          benefits,
   496→          memberLimit: limitNum,
   497→          discountCapPct: capNum,
   498→          branchScope: form.branchScope,
   499→          branches: form.branchScope === 'SELECTED' ? form.branchIds : null,
   500→          terms: form.terms.trim() || null,
   501→        })
   502→        toast.success(`${form.name.trim()} updated`)
   503→      } else {
   504→        await apiSend('/api/org/memberships', 'POST', {
   505→          orgId,
   506→          name: form.name.trim(),
   507→          description: form.description.trim() || undefined,
   508→          priceMonthlyMinor: monthlyMinor,
   509→          priceYearlyMinor: yearlyMinor ?? undefined,
   510→          benefits,
   511→          memberLimit: limitNum ?? undefined,
   512→          discountCapPct: capNum ?? undefined,
   513→          branchScope: form.branchScope,
   514→          branches: form.branchScope === 'SELECTED' ? form.branchIds : undefined,
   515→          terms: form.terms.trim() || undefined,
   516→        })
   517→        toast.success(`${form.name.trim()} published: customers can subscribe now`)
   518→      }
   519→      onSaved()
   520→    } catch (e) {
   521→      toast.error(e instanceof Error ? e.message : 'Could not save the tier')
   522→    } finally {
   523→      setBusy(false)
   524→    }
   525→  }
   526→
   527→  return (
   528→    <Dialog open onOpenChange={(o) => { if (!o) onClose() }}>
   529→      <DialogContent className="max-h-[90vh] overflow-y-auto scrollbar-thin sm:max-w-xl">
   530→        <DialogHeader>
   531→          <DialogTitle>{editing ? `Edit ${tier?.name}` : 'New membership tier'}</DialogTitle>
   532→          <DialogDescription>
   533→            {editing
   534→              ? 'Changes apply to new subscribers. Existing members keep the terms they paid for.'
   535→              : 'Publish a tier to the public Business Memberships page. Be honest about limits: they are shown to customers exactly as entered.'}
   536→          </DialogDescription>
   537→        </DialogHeader>
   538→
   539→        <div className="grid gap-4">
   540→          <div className="grid gap-4 sm:grid-cols-2">
   541→            <div className="space-y-1.5">
   542→              <Label htmlFor="tier-name">Tier name</Label>
   543→              <Input id="tier-name" value={form.name} onChange={(e) => setForm({ ...form, name: e.target.value })} placeholder="e.g. Service Club" />
   544→            </div>
   545→            <div className="grid grid-cols-2 gap-3">
   546→              <div className="space-y-1.5">
   547→                <Label htmlFor="tier-monthly">Monthly (KES)</Label>
   548→                <Input id="tier-monthly" inputMode="decimal" value={form.priceMonthly} onChange={(e) => setForm({ ...form, priceMonthly: e.target.value })} placeholder="2500" />
   549→              </div>
   550→              <div className="space-y-1.5">
   551→                <Label htmlFor="tier-yearly">Yearly (KES)</Label>
   552→                <Input id="tier-yearly" inputMode="decimal" value={form.priceYearly} onChange={(e) => setForm({ ...form, priceYearly: e.target.value })} placeholder="optional" />
   553→              </div>
   554→            </div>
   555→          </div>
   556→
   557→          <div className="space-y-1.5">
   558→            <Label htmlFor="tier-desc">Description</Label>
   559→            <Input id="tier-desc" value={form.description} onChange={(e) => setForm({ ...form, description: e.target.value })} placeholder="One line explaining who this tier is for" />
   560→          </div>
   561→
   562→          <div className="space-y-1.5">
   563→            <Label htmlFor="tier-benefits">Benefits: one per line ({benefits.length}/12)</Label>
   564→            <Textarea
   565→              id="tier-benefits"
   566→              rows={4}
   567→              value={form.benefitsText}
   568→              onChange={(e) => setForm({ ...form, benefitsText: e.target.value })}
   569→              placeholder={'2 free oil changes per month\n10% off all parts\nPriority booking'}
   570→            />
   571→            {benefits.length === 0 && <p className="text-xs text-rose-600 dark:text-rose-400">At least one benefit is required.</p>}
   572→          </div>
   573→
   574→          <div className="grid grid-cols-2 gap-4">
   575→            <div className="space-y-1.5">
   576→              <Label htmlFor="tier-limit">Member limit</Label>
   577→              <Input id="tier-limit" inputMode="numeric" value={form.memberLimit} onChange={(e) => setForm({ ...form, memberLimit: e.target.value })} placeholder="unlimited" />
   578→            </div>
   579→            <div className="space-y-1.5">
   580→              <Label htmlFor="tier-cap">Discount cap (%)</Label>
   581→              <Input id="tier-cap" inputMode="numeric" value={form.discountCapPct} onChange={(e) => setForm({ ...form, discountCapPct: e.target.value })} placeholder="e.g. 15" />
   582→            </div>
   583→          </div>
   584→
   585→          <div className="space-y-2">
   586→            <Label>Branch applicability</Label>
   587→            <Select value={form.branchScope} onValueChange={(v) => setForm({ ...form, branchScope: v as 'ALL' | 'SELECTED' })}>
   588→              <SelectTrigger className="w-full"><SelectValue /></SelectTrigger>
   589→              <SelectContent>
   590→                <SelectItem value="ALL">All branches</SelectItem>
   591→                <SelectItem value="SELECTED">Selected branches only</SelectItem>
   592→              </SelectContent>
   593→            </Select>
   594→            {form.branchScope === 'SELECTED' && (
   595→              branches === null ? (
   596→                <Skeleton className="h-20 rounded-lg" />
   597→              ) : branches.length === 0 ? (
   598→                <p className="rounded-md border border-dashed p-2 text-xs text-muted-foreground">
   599→                  No active branches yet: create branches first, or keep the scope on “All branches”.
   600→                </p>
   601→              ) : (
   602→                <div className="max-h-44 space-y-1.5 overflow-y-auto scrollbar-thin rounded-lg border p-2.5">
   603→                  {branches.map((b) => (
   604→                    <label key={b.id} className="flex cursor-pointer items-center gap-2 text-sm">
   605→                      <Checkbox
   606→                        checked={form.branchIds.includes(b.id)}
   607→                        onCheckedChange={(v) =>
   608→                          setForm((f) => ({ ...f, branchIds: v ? [...f.branchIds, b.id] : f.branchIds.filter((id) => id !== b.id) }))
   609→                        }
   610→                      />
   611→                      {b.name} <span className="text-xs text-muted-foreground">{b.code} · {b.town ?? b.county}</span>
   612→                    </label>
   613→                  ))}
   614→                </div>
   615→              )
   616→            )}
   617→          </div>
   618→
   619→          <div className="space-y-1.5">
   620→            <Label htmlFor="tier-terms">Terms (shown prominently to customers)</Label>
   621→            <Textarea
   622→              id="tier-terms"
   623→              rows={3}
   624→              value={form.terms}
   625→              onChange={(e) => setForm({ ...form, terms: e.target.value })}
   626→              placeholder="e.g. Benefits are valid at participating branches only. Discounts cannot be combined with other promotions."
   627→            />
   628→          </div>
   629→        </div>
   630→
   631→        <DialogFooter>
   632→          <Button variant="outline" onClick={onClose}>Cancel</Button>
   633→          <Button className="min-h-11" disabled={!valid || busy} onClick={save}>
   634→            {busy ? 'Saving…' : editing ? 'Save changes' : 'Publish tier'}
   635→          </Button>
   636→        </DialogFooter>
   637→      </DialogContent>
   638→    </Dialog>
   639→  )
   640→}

   644→function RedeemDialog({ member, tier, onClose, onSubmit }: {
   645→  member: Member
   646→  tier: Tier | null
   647→  onClose: () => void
   648→  onSubmit: (benefit: string, refType?: string) => Promise<void>
   649→}) {
   650→  const [benefit, setBenefit] = useState('')
   651→  const [refType, setRefType] = useState('')
   652→  const [busy, setBusy] = useState(false)
   653→
   654→  const periodKey = `${new Date().getFullYear()}-${String(new Date().getMonth() + 1).padStart(2, '0')}`
   655→  const alreadyThisPeriod = new Set(member.redemptions.filter((r) => r.periodKey === periodKey).map((r) => r.benefit))
   656→  const available = tier ? tier.benefits.filter((b) => !alreadyThisPeriod.has(b)) : []
   657→
   658→  return (
   659→    <Dialog open onOpenChange={(o) => { if (!o) onClose() }}>
   660→      <DialogContent className="sm:max-w-md">
   661→        <DialogHeader>
   662→          <DialogTitle>Record redemption: {member.customer.name}</DialogTitle>
   663→          <DialogDescription>
   664→            {member.tier} · {member.refCode}. One redemption per benefit per billing period is
   665→            enforced by the ledger: duplicates are rejected.
   666→          </DialogDescription>
   667→        </DialogHeader>
   668→        <div className="space-y-4">
   669→          {tier === null ? (
   670→            <p className="rounded-lg border border-amber-300 bg-amber-500/10 p-3 text-sm text-amber-800 dark:border-amber-500/40 dark:text-amber-200">
   671→              The tier “{member.tier}” no longer exists in your programme (it may have been archived).
   672→              Redemptions can only be recorded against current tier benefits.
   673→            </p>
   674→          ) : available.length === 0 ? (
   675→            <p className="rounded-lg border border-dashed p-3 text-sm text-muted-foreground">
   676→              All benefits of this tier have already been redeemed for the current period ({periodKey}).
   677→            </p>
   678→          ) : (
   679→            <>
   680→              <div className="space-y-1.5">
   681→                <Label>Benefit</Label>
   682→                <Select value={benefit} onValueChange={setBenefit}>
   683→                  <SelectTrigger className="w-full"><SelectValue placeholder="Select the benefit being used" /></SelectTrigger>
   684→                  <SelectContent>
   685→                    {available.map((b) => <SelectItem key={b} value={b}>{b}</SelectItem>)}
   686→                  </SelectContent>
   687→                </Select>
   688→              </div>
   689→              <div className="space-y-1.5">
   690→                <Label htmlFor="redeem-ref">Reference (optional)</Label>
   691→                <Input id="redeem-ref" value={refType} onChange={(e) => setRefType(e.target.value)} placeholder="e.g. booking ref, invoice ref" />
   692→              </div>
   693→            </>
   694→          )}
   695→        </div>
   696→        <DialogFooter>
   697→          <Button variant="outline" onClick={onClose}>Cancel</Button>
   698→          <Button
   699→            className="min-h-11"
   700→            disabled={benefit === '' || busy}
   701→            onClick={async () => { setBusy(true); try { await onSubmit(benefit, refType.trim() || undefined) } finally { setBusy(false) } }}
   702→          >
   703→            {busy ? 'Recording…' : 'Record redemption'}
   704→          </Button>
   705→        </DialogFooter>
   706→      </DialogContent>
   707→    </Dialog>
   708→  )
   709→}
