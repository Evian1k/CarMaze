   410→export function CreditManager({ orgId }: { orgId: string }) {
   411→  const [facilities, setFacilities] = useState<Facility[] | null>(null)
   412→  const [error, setError] = useState<string | null>(null)
   413→  const [reviewTarget, setReviewTarget] = useState<Facility | null>(null)
   414→  const [chargeTarget, setChargeTarget] = useState<Facility | null>(null)
   415→  const [repayTarget, setRepayTarget] = useState<Facility | null>(null)
   416→  const [suspendTarget, setSuspendTarget] = useState<Facility | null>(null)
   417→  const [closeTarget, setCloseTarget] = useState<Facility | null>(null)
   418→
   419→  const load = useCallback(() => {
   420→    apiGet<{ facilities: Facility[] }>(`/api/org/credit?orgId=${orgId}`)
   421→      .then((d) => { setFacilities(d.facilities); setError(null) })
   422→      .catch((e) => setError(e instanceof Error ? e.message : 'Could not load credit facilities'))
   423→  }, [orgId])
   424→
   425→  useEffect(() => { load() }, [load])
   426→
   427→  async function patch(facility: Facility, body: Record<string, unknown>, successMsg: string) {
   428→    try {
   429→      const d = await apiSend<{ message?: string }>(`/api/org/credit/${facility.id}`, 'PATCH', body)
   430→      toast.success(d.message ?? successMsg)
   431→      load()
   432→      return true
   433→    } catch (e) {
   434→      toast.error(e instanceof Error ? e.message : 'Action failed')
   435→      return false
   436→    }
   437→  }
   438→
   439→  async function postEntry(facility: Facility, body: Record<string, unknown>): Promise<boolean> {
   440→    try {
   441→      const d = await apiSend<{ outstandingMinor: number }>(`/api/org/credit/${facility.id}`, 'POST', body)
   442→      toast.success(`${body.kind === 'CHARGE' ? 'Charge' : 'Repayment'} recorded: outstanding ${money(d.outstandingMinor)}`)
   443→      load()
   444→      return true
   445→    } catch (e) {
   446→      const code = (e as { code?: string }).code
   447→      const message = e instanceof Error ? e.message : 'Could not record the entry'
   448→      toast.error(message, {
   449→        description: code === 'LIMIT_EXCEEDED'
   450→          ? 'The approved limit is enforced by the ledger; the charge was rejected.'
   451→          : code === 'OVER_REPAYMENT'
   452→            ? 'Repayments cannot exceed the outstanding balance.'
   453→            : code === 'OUTSTANDING_BALANCE'
   454→              ? 'Clear the outstanding balance before closing the facility.'
   455→              : undefined,
   456→      })
   457→      return false
   458→    }
   459→  }
   460→
   461→  if (error) {
   462→    return (
   463→      <Card>
   464→        <CardContent className="flex flex-col items-center gap-3 py-10 text-center">
   465→          <CircleAlert className="h-8 w-8 text-muted-foreground" />
   466→          <p className="font-medium">Could not load credit facilities</p>
   467→          <p className="text-sm text-muted-foreground">{error}</p>
   468→          <Button variant="outline" className="min-h-11" onClick={() => { setFacilities(null); load() }}><RefreshCw className="h-4 w-4" /> Try again</Button>
   469→        </CardContent>
   470→      </Card>
   471→    )
   472→  }
   473→
   474→  if (!facilities) {
   475→    return <Skeleton className="h-56 rounded-xl" aria-busy="true" />
   476→  }
   477→
   478→  return (
   479→    <div className="space-y-4">
