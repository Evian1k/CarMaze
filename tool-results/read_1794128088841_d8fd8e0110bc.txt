   109→  const [query, setQuery] = useState('')
   110→  const [statusFilter, setStatusFilter] = useState<(typeof STATUS_FILTERS)[number]>('ALL')
   111→  const [editing, setEditing] = useState<ProductItem | null>(null)
   112→  const [creating, setCreating] = useState(false)
   113→  const [compatFor, setCompatFor] = useState<ProductItem | null>(null)
   114→  const [togglingId, setTogglingId] = useState<string | null>(null)
   115→
   116→  const canWrite = supplierRoleCan(role, 'catalogue.write')
   117→
   118→  // slug → categoryId map harvested from the supplier's own products (the
   119→  // public categories endpoint does not expose IDs; see module header).
   120→  const catIdBySlug = useMemo(() => {
   121→    const map = new Map<string, string>()
   122→    for (const p of products.data?.products ?? []) {
   123→      if (p.category?.slug && p.categoryId) map.set(p.category.slug, p.categoryId)
   124→    }

   126→  }, [products.data])
   127→
   128→  const list = products.data?.products ?? []
   129→  const catalogueBoundElsewhere = list.length > 0 && list.some((p) => p.supplierOrgId !== orgId)
   130→  const filtered = list.filter((p) => {
   131→    if (statusFilter !== 'ALL' && p.status !== statusFilter) return false
   132→    if (!query.trim()) return true
   133→    const q = query.trim().toLowerCase()
   134→    return (
   135→      p.name.toLowerCase().includes(q) ||
   136→      p.sku.toLowerCase().includes(q) ||
   137→      (p.mpn ?? '').toLowerCase().includes(q) ||
   138→      (p.brand ?? '').toLowerCase().includes(q) ||

   168→          onClick={() => setCreating(true)}
   169→          disabled={!canWrite}

   175→      </div>

   177→      {!canWrite && (
   178→        <p className="text-xs text-muted-foreground">

   202→            <button
   203→              key={s}

   207→                'rounded-full border px-3 py-1 text-xs transition-colors',
   208→                statusFilter === s ? 'border-primary bg-primary/10 text-primary' : 'border-border text-muted-foreground hover:text-foreground'

   214→        </div>
   215→      </div>
   216→
   217→      {categories.data && catIdBySlug.size === 0 && (categories.data.categories?.length ?? 0) > 0 && (
   218→        <Alert>
   219→          <Boxes className="h-4 w-4" />
   220→          <AlertTitle>First-product category note</AlertTitle>
   221→          <AlertDescription>
   222→            The platform&apos;s category list currently doesn&apos;t expose the internal IDs new products need (engineering

   230→        <LoadingBlock label="Loading catalogue…" />

   232→        <ModuleNotice title="Catalogue could not load" detail={products.error.message} retry={products.reload} />

   234→        <EmptyState
   235→          icon={<Package className="h-8 w-8" />}
   236→          title={list.length === 0 ? 'No products yet' : 'No products match the filters'}
   237→          detail={list.length === 0
   238→            ? 'Add your first product with pricing, warranty and warehouse stock: published products appear in the supply catalogue for verified garages.'
   239→            : 'Try a different search term or status filter.'}

   257→      {/* Create / edit dialog */}

   263→          onSaved={() => {
   264→            setCreating(false)
   265→            products.reload()
   266→          }}
   267→        />
   268→      )}
   269→      {editing && (
   270→        <ProductFormDialog
   271→          product={editing}
   272→          categories={categories.data?.categories ?? []}
   273→          catIdBySlug={catIdBySlug}
   274→          onClose={() => setEditing(null)}
   275→          onSaved={() => {
   276→            setEditing(null)
   277→            products.reload()
   278→          }}
   279→        />
   280→      )}
   281→
   282→      {/* Compatibility editor */}
   283→      {compatFor && (
   284→        <CompatibilityDialog

   286→          canWrite={canWrite}
   287→          onClose={() => setCompatFor(null)}

   353→        </div>

   355→    </Card>

   361→function ProductFormDialog({ product, categories, catIdBySlug, onClose, onSaved }: {
   362→  product?: ProductItem
   363→  categories: ProductCategory[]
   364→  catIdBySlug: Map<string, string>
   365→  onClose: () => void
   366→  onSaved: () => void
   367→}) {
   368→  const isEdit = Boolean(product)
   369→  const [form, setForm] = useState<ProductForm>(() => (product ? formFromProduct(product) : EMPTY_FORM))
   370→  const [images, setImages] = useState<string[]>(product?.images ?? [])
   371→  const [uploading, setUploading] = useState(false)
   372→  const [saving, setSaving] = useState(false)
   373→  const [categoryError, setCategoryError] = useState<string | null>(null)
   374→
   375→  const set = <K extends keyof ProductForm>(k: K, v: ProductForm[K]) => setForm((f) => ({ ...f, [k]: v }))
   376→
   377→  const unknownCategory = !isEdit && form.categorySlug !== '' && !catIdBySlug.has(form.categorySlug)
   378→
   379→  async function handleFiles(files: FileList | null) {
   380→    if (!files || files.length === 0) return
   381→    const room = 8 - images.length
   382→    if (room <= 0) {
   383→      toast.error('A product can have at most 8 images.')
   384→      return
   385→    }
   386→    setUploading(true)
   387→    try {
   388→      for (const file of Array.from(files).slice(0, room)) {
   389→        const res = await uploadOne(file)
   390→        setImages((prev) => [...prev, res.url])
   391→      }
   392→      toast.success('Image uploaded')
   393→    } catch (e) {
   394→      toast.error(await errMessage(e, 'Upload failed: please try again.'))
   395→    } finally {
   396→      setUploading(false)
   397→    }
   398→  }
   399→
   400→  function validate(): string | null {
   401→    if (form.name.trim().length < 3) return 'Product name must be at least 3 characters.'
   402→    if (!isEdit && (form.sku.trim().length < 2 || form.sku.trim().length > 40)) return 'SKU must be 2–40 characters.'
   403→    if (!form.categorySlug) return 'Choose a product category.'
   404→    if (unknownCategory) return 'That category has no internal ID available yet; pick a category you have used before.'
   405→    if (form.description.trim().length < 10) return 'Description must be at least 10 characters.'
   406→    const unit = kesToMinor(form.unitPrice)
   407→    if (unit === null || unit < 100) return 'Enter a unit price of at least KES 1.'
   408→    if (form.wholesalePrice.trim() && (kesToMinor(form.wholesalePrice) ?? 0) < 0) return 'Wholesale price is invalid.'
   409→    const minQty = form.wholesaleMinQty.trim() ? Number(form.wholesaleMinQty) : 0
   410→    if (!Number.isInteger(minQty) || minQty < 0 || minQty > 10000) return 'Wholesale minimum quantity must be 0–10,000.'
   411→    const tax = form.taxRatePct.trim() ? Number(form.taxRatePct) : 0
   412→    if (!Number.isFinite(tax) || tax < 0 || tax > 50) return 'Tax rate must be 0–50%.'
   413→    const warranty = form.warrantyMonths.trim() ? Number(form.warrantyMonths) : 0
   414→    if (!Number.isInteger(warranty) || warranty < 0 || warranty > 120) return 'Warranty must be 0–120 months.'
   415→    if (images.length > 8) return 'A product can have at most 8 images.'
   416→    if (splitComma(form.tags).length > 12) return 'At most 12 tags.'
   417→    if (splitComma(form.vehicleMakes).length > 30) return 'At most 30 vehicle makes.'
   418→    if (form.provenanceNote.length > 1000) return 'Provenance note must be at most 1000 characters.'
   419→    return null
   420→  }
   421→
   422→  async function save() {
   423→    const problem = validate()
   424→    if (problem) {
   425→      toast.error(problem)
   426→      return
   427→    }
   428→    setSaving(true)
   429→    setCategoryError(null)
   430→    try {
   431→      const categoryId = isEdit ? undefined : catIdBySlug.get(form.categorySlug)
   432→      if (!isEdit && !categoryId) {
   433→        setCategoryError('Category ID unavailable: the platform category list does not expose internal IDs for new categories.')
   434→        setSaving(false)
   435→        return
   436→      }
   437→      if (isEdit && product) {
   438→        await apiSend(`/api/supply/products/${product.id}`, 'PATCH', {
   439→          name: form.name.trim(),
   440→          ...(form.categorySlug ? { categoryId: catIdBySlug.get(form.categorySlug) } : {}),
   441→          mpn: form.mpn.trim() || null,
   442→          oemRef: form.oemRef.trim() || null,
   443→          brand: form.brand.trim() || null,
   444→          description: form.description.trim(),
   445→          condition: form.condition,
   446→          unitPriceMinor: kesToMinor(form.unitPrice)!,
   447→          wholesalePriceMinor: form.wholesalePrice.trim() ? kesToMinor(form.wholesalePrice) : null,
   448→          wholesaleMinQty: form.wholesaleMinQty.trim() ? Number(form.wholesaleMinQty) : 0,
   449→          taxRatePct: form.taxRatePct.trim() ? Number(form.taxRatePct) : 0,
   450→          warrantyMonths: form.warrantyMonths.trim() ? Number(form.warrantyMonths) : 0,
   451→          returnable: form.returnable,
   452→          images,
   453→          tags: splitComma(form.tags),
   454→          vehicleMakes: splitComma(form.vehicleMakes),
   455→          provenanceNote: form.provenanceNote.trim() || undefined,
   456→          status: form.publish ? 'PUBLISHED' : 'DRAFT',
   457→        })
   458→        toast.success('Product updated.')
   459→      } else {
   460→        await apiSend('/api/supply/products', 'POST', {
   461→          name: form.name.trim(),
   462→          sku: form.sku.trim(),
   463→          categoryId,
   464→          mpn: form.mpn.trim() || null,
   465→          oemRef: form.oemRef.trim() || null,
   466→          brand: form.brand.trim() || null,
   467→          description: form.description.trim(),
   468→          condition: form.condition,
   469→          unitPriceMinor: kesToMinor(form.unitPrice)!,
   470→          wholesalePriceMinor: form.wholesalePrice.trim() ? kesToMinor(form.wholesalePrice) : null,
   471→          wholesaleMinQty: form.wholesaleMinQty.trim() ? Number(form.wholesaleMinQty) : 0,
   472→          taxRatePct: form.taxRatePct.trim() ? Number(form.taxRatePct) : 0,
   473→          warrantyMonths: form.warrantyMonths.trim() ? Number(form.warrantyMonths) : 0,
   474→          returnable: form.returnable,
   475→          images,
   476→          tags: splitComma(form.tags),
   477→          vehicleMakes: splitComma(form.vehicleMakes),
   478→          batchTracked: form.batchTracked,
   479→          serialTracked: form.serialTracked,
   480→          provenanceNote: form.provenanceNote.trim() || null,
   481→          publish: form.publish,
   482→        })
   483→        toast.success(form.publish ? 'Product published to the supply catalogue.' : 'Product saved as draft.')
   484→      }
   485→      onSaved()

   720→type CompatForm = { make: string; model: string; yearFrom: string; yearTo: string; engine: string; trim: string; chassis: string; note: string }
   721→
   722→const EMPTY_COMPAT: CompatForm = { make: '', model: '', yearFrom: '', yearTo: '', engine: '', trim: '', chassis: '', note: '' }
   723→
   724→function CompatibilityDialog({ product, canWrite, onClose }: {
   725→  product: ProductItem
   726→  canWrite: boolean
   727→  onClose: () => void
   728→}) {
   729→  const rows = useFetch<{ compatibility: CompatibilityRow[] }>(`/api/supply/products/${product.id}/compatibility`)
   730→  const [form, setForm] = useState<CompatForm>(EMPTY_COMPAT)
   731→  const [saving, setSaving] = useState(false)
   732→  const [deletingId, setDeletingId] = useState<string | null>(null)
   733→
   734→  const set = (k: keyof CompatForm, v: string) => setForm((f) => ({ ...f, [k]: v }))
   735→
   736→  async function add() {
   737→    if (form.make.trim().length < 2) {
   738→      toast.error('Enter the vehicle make (at least 2 characters).')
   739→      return
   740→    }
   741→    const yf = form.yearFrom.trim() ? Number(form.yearFrom) : null
   742→    const yt = form.yearTo.trim() ? Number(form.yearTo) : null
   743→    for (const [label, y] of [['Year from', yf], ['Year to', yt]] as const) {
   744→      if (y !== null && (!Number.isInteger(y) || y < 1950 || y > 2100)) {
   745→        toast.error(`${label} must be a year between 1950 and 2100.`)
   746→        return
   747→      }
   748→    }
   749→    if (yf !== null && yt !== null && yf > yt) {
   750→      toast.error('Year range is invalid: “from” must be ≤ “to”.')
   751→      return
   752→    }
   753→    setSaving(true)
   754→    try {
   755→      await apiSend(`/api/supply/products/${product.id}/compatibility`, 'POST', {
   756→        make: form.make.trim(),
   757→        model: form.model.trim() || null,
   758→        yearFrom: yf,
   759→        yearTo: yt,
   760→        engine: form.engine.trim() || null,
   761→        trim: form.trim.trim() || null,
   762→        chassis: form.chassis.trim() || null,
   763→        note: form.note.trim() || null,
   764→      })
   765→      toast.success('Fitment reference added: it stays “pending admin review” until the platform verifies it.')
   766→      setForm(EMPTY_COMPAT)
   767→      rows.reload()
   768→    } catch (e) {
   769→      toast.error(await errMessage(e, 'Could not add the fitment record.'))
   770→    } finally {
   771→      setSaving(false)
   772→    }
   773→  }
   774→
   775→  async function remove(id: string) {
   776→    setDeletingId(id)
   777→    try {
   778→      await apiSend(`/api/supply/products/${product.id}/compatibility?compatId=${id}`, 'DELETE')
   779→      rows.reload()
   780→    } catch (e) {
   781→      toast.error(await errMessage(e, 'Could not remove the fitment record.'))
   782→    } finally {
   783→      setDeletingId(null)
   784→    }
   785→  }
   786→
   787→  return (
   788→    <Dialog open onOpenChange={(o) => { if (!o) onClose() }}>
   789→      <DialogContent className="max-h-[92vh] overflow-y-auto scrollbar-thin sm:max-w-2xl">
   790→        <DialogHeader>
   791→          <DialogTitle>Vehicle compatibility: {product.name}</DialogTitle>
   792→          <DialogDescription>
   793→            Fitment references you add are marked “pending admin review” until the platform verifies them. CARMAZE never
   794→            invents compatibility: these rows tell garages exactly what you declare the part fits.
   795→          </DialogDescription>
   796→        </DialogHeader>
   797→
   798→        <div className="space-y-2">
   799→          {rows.loading ? (
   800→            <LoadingBlock label="Loading fitment records…" />
   801→          ) : rows.error ? (
   802→            <ModuleNotice title="Fitment records could not load" detail={rows.error.message} retry={rows.reload} />
   803→          ) : (rows.data?.compatibility ?? []).length === 0 ? (
   804→            <EmptyState
   805→              icon={<GitBranch className="h-7 w-7" />}
   806→              title="No fitment records yet"
   807→              detail="Add makes, models and year ranges so garages searching by vehicle can find this part."
   808→            />
   809→          ) : (
   810→            <ul className="max-h-64 space-y-2 overflow-y-auto scrollbar-thin">
   811→              {(rows.data?.compatibility ?? []).map((r) => (
   812→                <li key={r.id} className="flex items-start justify-between gap-3 rounded-lg border border-border px-3 py-2">
   813→                  <div className="min-w-0 text-sm">
   814→                    <p className="font-medium">
   815→                      {r.make}{r.model ? ` ${r.model}` : ''}
   816→                      {r.yearFrom || r.yearTo ? (
   817→                        <span className="font-normal text-muted-foreground">
   818→                          {' '}· {r.yearFrom ?? '…'}–{r.yearTo ?? '…'}
   819→                        </span>
   820→                      ) : null}
   821→                    </p>
   822→                    <p className="truncate text-xs text-muted-foreground">
   823→                      {[r.engine, r.trim, r.chassis, r.note].filter(Boolean).join(' · ') || 'No additional details'}
   824→                    </p>
   825→                    {r.verified ? (
   826→                      <Badge variant="secondary" className="mt-1 gap-1 border-transparent bg-primary/10 px-1 py-0 text-[10px] text-primary">

   831→                        Pending admin review

   845→
   846→        {canWrite && (
   847→          <>
   848→            <Separator />
   849→            <div className="grid gap-3 sm:grid-cols-2">
   850→              <div className="space-y-1.5">
   851→                <Label htmlFor="cf-make">Make *</Label>

   853→                <datalist id="compat-makes">{VEHICLE_MAKES.map((m) => <option key={m} value={m} />)}</datalist>

   860→                <div className="space-y-1.5">
   861→                  <Label htmlFor="cf-yf">Year from</Label>
