   413→// ───────────────────────────── Revision editor ─────────────────────────────

   415→type LineEdit = { itemId: string; productId: string; name: string; qty: string; unit: string; remove: boolean }
   416→type AddLine = { productId: string; qty: string; unit: string }
   417→
   418→function RevisionEditor({ order, busy, act, products, onDone }: {
   419→  order: NonNullable<OrderDetail['order']>
   420→  busy: string | null
   421→  act: (key: string, fn: () => Promise<unknown>) => Promise<unknown>
   422→  products: FetchState<{ products: ProductItem[] }>
   423→  onDone: () => void
   424→}) {
   425→  const [lines, setLines] = useState<LineEdit[]>(() =>
   426→    order.items.map((i) => ({ itemId: i.id, productId: i.productId, name: i.name, qty: String(i.qty), unit: minorToKesInput2(i.unitPriceMinor), remove: false }))
   427→  )
   428→  const [adds, setAdds] = useState<AddLine[]>([])
   429→  const [note, setNote] = useState('')
   430→
   431→  function setLine(idx: number, patch: Partial<LineEdit>) {
   432→    setLines((ls) => ls.map((l, i) => (i === idx ? { ...l, ...patch } : l)))
   433→  }
   434→
   435→  const changes = useMemo(() => {
   436→    const out: { itemId?: string; productId: string; action: 'UPDATE' | 'REMOVE' | 'ADD'; qty?: number; unitPriceMinor?: number }[] = []
   437→    for (const l of lines) {
   438→      const qty = Number(l.qty)
   439→      const unit = kesToMinor(l.unit)
   440→      if (l.remove) {
   441→        out.push({ itemId: l.itemId, productId: l.productId, action: 'REMOVE' })
   442→      } else if (Number.isInteger(qty) && qty >= 1 && unit !== null) {
   443→        // changed only when qty/price differ from the original
   444→        const orig = order.items.find((i) => i.id === l.itemId)
   445→        if (orig && (qty !== orig.qty || unit !== orig.unitPriceMinor)) {
   446→          out.push({ itemId: l.itemId, productId: l.productId, action: 'UPDATE', qty, unitPriceMinor: unit })
   447→        }
   448→      }
   449→    }
   450→    for (const a of adds) {
   451→      const qty = Number(a.qty)
   452→      const unit = kesToMinor(a.unit)
   453→      if (Number.isInteger(qty) && qty >= 1 && unit !== null && unit >= 100) {
   454→        out.push({ productId: a.productId, action: 'ADD', qty, unitPriceMinor: unit })
   455→      }
   456→    }
   457→    return out
   458→  }, [lines, adds, order.items])
   459→
   460→  const newSubtotal = useMemo(() => {
   461→    let sum = 0
   462→    for (const l of lines) {
   463→      if (l.remove) continue
   464→      const qty = Number(l.qty)
   465→      const unit = kesToMinor(l.unit)
   466→      if (Number.isInteger(qty) && qty >= 1 && unit !== null) sum += qty * unit
   467→    }
   468→    for (const a of adds) {
   469→      const qty = Number(a.qty)
   470→      const unit = kesToMinor(a.unit)
   471→      if (Number.isInteger(qty) && qty >= 1 && unit !== null) sum += qty * unit
   472→    }
   473→    return sum
   474→  }, [lines, adds])

   483→          <p className="text-xs text-muted-foreground">
   484→            Adjust quantities or unit prices, remove lines, or add lines from your catalogue. The garage must approve the
   485→            revision before the order is final.
   486→          </p>
   487→        </div>
   488→
   489→        <div className="space-y-2">
   490→          {lines.map((l, idx) => (

   500→                    onChange={(e) => setLine(idx, { qty: e.target.value.replace(/[^\d]/g, '') })} />
   501→                </div>
   502→                <div className="w-28">
   503→                  <Label htmlFor={`rv-unit-${l.itemId}`} className="text-[10px] text-muted-foreground">Unit (KES)</Label>

   506→                </div>

   541→                    <Button size="sm" variant="ghost" className="h-8 text-destructive hover:text-destructive"
   542→                      onClick={() => setAdds((xs) => xs.filter((_, i) => i !== idx))}>
   543→                      Remove
   544→                    </Button>
   545→                  </div>
   546→                </div>
   547→              )
   548→            })}
   549→          </div>
   550→        )}
   551→
   552→        <AddLinePicker products={products} onAdd={(a) => setAdds((xs) => [...xs, a])} existing={new Set([...lines.map((l) => l.productId), ...adds.map((a) => a.productId)])} />

   566→          <Button variant="ghost" onClick={onDone} disabled={busy !== null}>Back</Button>
   567→          <Button
   568→            className="min-h-11"
   569→            disabled={busy !== null || !publishable}
   570→            title={publishable ? 'Send revision to the garage' : 'Adjust at least one line and add a note (min 3 chars).'}

   572→              act('respond', () =>

   697→            </p>
