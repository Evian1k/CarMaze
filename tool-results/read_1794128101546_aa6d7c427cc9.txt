    89→export function BranchManager({ orgId, orgType }: { orgId: string; orgType: string }) {
    90→  const [branches, setBranches] = useState<Branch[] | null>(null)
    91→  const [error, setError] = useState<string | null>(null)
    92→  const [createOpen, setCreateOpen] = useState(false)
    93→  const [editTarget, setEditTarget] = useState<Branch | null>(null)
    94→  const [closeTarget, setCloseTarget] = useState<Branch | null>(null)
    95→
    96→  const load = useCallback(() => {
    97→    apiGet<{ branches: Branch[] }>(`/api/supply/branches?orgId=${orgId}`)
    98→      .then((d) => { setBranches(d.branches); setError(null) })
    99→      .catch((e) => setError(e instanceof Error ? e.message : 'Could not load branches'))
   100→  }, [orgId])
   101→
   102→  useEffect(() => { load() }, [load])
   103→
   104→  async function doClose(branch: Branch) {
   105→    try {
   106→      await apiSend(`/api/supply/branches/${branch.id}`, 'DELETE')
   107→      toast.success(`${branch.name} closed: historical records are preserved`)
   108→      load()
   109→    } catch (e) {
   110→      toast.error(e instanceof Error ? e.message : 'Could not close the branch')
   111→    }
   112→  }
   113→
   114→  return (
   115→    <div className="space-y-4">
   116→      <div className="flex flex-wrap items-center justify-between gap-2">
   117→        <div>
   118→          <h3 className="text-lg font-semibold">
   119→            {orgType === 'DEALER' ? 'Showroom branches' : orgType === 'GARAGE' ? 'Workshop branches' : 'Branches & warehouses'}
   120→          </h3>
   121→          <p className="text-sm text-muted-foreground">Creating, editing and closing branches requires the owner or general manager.</p>
   122→        </div>
   123→        <Button className="min-h-11" onClick={() => setCreateOpen(true)}><Plus className="h-4 w-4" /> New branch</Button>
   124→      </div>
   125→
   126→      {error ? (
   127→        <Card>
   128→          <CardContent className="flex flex-col items-center gap-3 py-10 text-center">
   129→            <CircleAlert className="h-8 w-8 text-muted-foreground" />
   130→            <p className="font-medium">Could not load branches</p>
   131→            <p className="text-sm text-muted-foreground">{error}</p>
   132→            <Button variant="outline" className="min-h-11" onClick={() => { setBranches(null); load() }}><RefreshCw className="h-4 w-4" /> Try again</Button>
   133→          </CardContent>
   134→        </Card>
   135→      ) : !branches ? (
   136→        <div className="grid gap-4 md:grid-cols-2 xl:grid-cols-3" aria-busy="true">
   137→          {[0, 1, 2].map((i) => <Skeleton key={i} className="h-52 rounded-xl" />)}
   138→        </div>
   139→      ) : branches.length === 0 ? (
   140→        <div className="rounded-xl border border-dashed p-8 text-center">
   141→          <Building2 className="mx-auto h-8 w-8 text-muted-foreground" />
   142→          <p className="mt-2 font-medium">No branches yet</p>
   143→          <p className="mx-auto mt-1 max-w-md text-sm text-muted-foreground">
   144→            Add your first location: it becomes the primary branch and anchors inventory, transfers
   145→            and delivery coverage.
   146→          </p>
   147→          <Button variant="outline" className="mt-4 min-h-11" onClick={() => setCreateOpen(true)}>Create the first branch</Button>
   148→        </div>
   149→      ) : (
   150→        <div className="grid gap-4 md:grid-cols-2 xl:grid-cols-3">
   151→          {branches.map((b) => (
   152→            <BranchCard key={b.id} branch={b} onEdit={() => setEditTarget(b)} onClose={() => setCloseTarget(b)} />
   153→          ))}
   154→        </div>
   155→      )}
   156→
   157→      {createOpen && (
   158→        <BranchFormDialog
   159→          orgId={orgId}
   160→          onClose={() => setCreateOpen(false)}
   161→          onSaved={() => { setCreateOpen(false); load() }}
   162→        />
   163→      )}
   164→      {editTarget && (
   165→        <BranchFormDialog
   166→          orgId={orgId}
   167→          branch={editTarget}
   168→          onClose={() => setEditTarget(null)}
   169→          onSaved={() => { setEditTarget(null); load() }}
   170→        />
   171→      )}
   172→      <AlertDialog open={closeTarget !== null} onOpenChange={(o) => { if (!o) setCloseTarget(null) }}>
   173→        <AlertDialogContent>
   174→          <AlertDialogHeader>
   175→            <AlertDialogTitle>Close {closeTarget?.name}?</AlertDialogTitle>
   176→            <AlertDialogDescription>
   177→              The branch stops appearing as an active location. It is never hard-deleted: historical
   178→              stock movements, transfers and orders keep their records. Make another branch primary
   179→              first if this one is primary.
   180→            </AlertDialogDescription>
   181→          </AlertDialogHeader>
   182→          <AlertDialogFooter>
   183→            <AlertDialogCancel>Keep branch</AlertDialogCancel>
   184→            <AlertDialogAction
   185→              className="bg-destructive text-white hover:bg-destructive/90"
   186→              onClick={async (e) => { e.preventDefault(); if (closeTarget) await doClose(closeTarget); setCloseTarget(null) }}
   187→            >
   188→              Close branch
   189→            </AlertDialogAction>
   190→          </AlertDialogFooter>
   191→        </AlertDialogContent>
   192→      </AlertDialog>
   193→    </div>
   194→  )
   195→}

   255→      </CardContent>
   256→    </Card>
   257→  )

   261→  const open = DAYS.filter((d) => hours[d])

   263→    <Collapsible>
   264→      <CollapsibleTrigger className="group inline-flex items-center gap-1 text-sm text-primary hover:underline">

   268→        <ul className="mt-1.5 space-y-0.5 rounded-md border bg-muted/40 p-2 text-xs">
   269→          {DAYS.map((d) => {
   270→            const h = hours[d]
   271→            return (
   272→              <li key={d} className="flex justify-between gap-4">

   275→              </li>
   276→            )
   277→          })}
   278→          {open.length === 0 && <li className="text-muted-foreground">No hours published</li>}
   279→        </ul>
   280→      </CollapsibleContent>
   281→    </Collapsible>
   282→  )

   287→type BranchForm = {
   288→  name: string; code: string; kind: 'BRANCH' | 'WAREHOUSE'
   289→  county: string; town: string; address: string
   290→  lat: string; lng: string; phone: string; managerName: string
   291→  hours: Record<string, { open: string; close: string }>
   292→  deliveryAreas: string
   293→  status: 'ACTIVE' | 'CLOSED'
   294→  isPrimary: boolean
   295→}
   296→
   297→function emptyHours(): BranchForm['hours'] {
   298→  return Object.fromEntries(DAYS.map((d) => [d, { open: '', close: '' }]))
   299→}
   300→
   301→function BranchFormDialog({ orgId, branch, onClose, onSaved }: {
   302→  orgId: string
   303→  branch?: Branch
   304→  onClose: () => void
   305→  onSaved: () => void
   306→}) {
   307→  const editing = Boolean(branch)
   308→  const [form, setForm] = useState<BranchForm>({
   309→    name: branch?.name ?? '',
   310→    code: branch?.code ?? '',
   311→    kind: (branch?.kind as 'BRANCH' | 'WAREHOUSE') ?? 'BRANCH',
   312→    county: branch?.county ?? 'Nairobi',
   313→    town: branch?.town ?? '',
   314→    address: branch?.address ?? '',
   315→    lat: branch?.lat != null ? String(branch.lat) : '',
   316→    lng: branch?.lng != null ? String(branch.lng) : '',
   317→    phone: branch?.phone ?? '',
   318→    managerName: branch?.managerName ?? '',
   319→    hours: (() => {
   320→      const h = emptyHours()
   321→      if (branch?.operatingHours) {
   322→        for (const d of DAYS) {
   323→          const v = branch.operatingHours[d]
   324→          if (v) h[d] = { open: v[0], close: v[1] }
   325→        }
   326→      }
   327→      return h
   328→    })(),
   329→    deliveryAreas: branch?.deliveryAreas.join(', ') ?? '',
   330→    status: (branch?.status as 'ACTIVE' | 'CLOSED') ?? 'ACTIVE',
   331→    isPrimary: branch?.isPrimary ?? false,
   332→  })
   333→  const [busy, setBusy] = useState(false)
   334→
   335→  const latNum = form.lat.trim() === '' ? null : Number(form.lat)
   336→  const lngNum = form.lng.trim() === '' ? null : Number(form.lng)
   337→  const valid =
   338→    form.name.trim().length >= 2 &&
   339→    (editing || form.code.trim().length >= 1) &&
   340→    form.county.trim().length >= 2 &&
   341→    (latNum === null || Number.isFinite(latNum)) &&
   342→    (lngNum === null || Number.isFinite(lngNum))
   343→
   344→  function buildOperatingHours(): Hours | undefined {
   345→    const out: Hours = {}
   346→    let any = false
   347→    for (const d of DAYS) {
   348→      const { open, close } = form.hours[d]
   349→      if (open.trim() && close.trim()) { out[d] = [open.trim(), close.trim()]; any = true }
   350→      else out[d] = null
   351→    }
   352→    return any ? out : undefined
   353→  }
   354→
   355→  async function save() {
   356→    setBusy(true)
   357→    try {
   358→      const hours = buildOperatingHours()
   359→      const areas = form.deliveryAreas.split(',').map((s) => s.trim()).filter(Boolean).slice(0, 20)
   360→      if (editing && branch) {
   361→        await apiSend(`/api/supply/branches/${branch.id}`, 'PATCH', {
   362→          name: form.name.trim(),
   363→          kind: form.kind,
   364→          county: form.county.trim(),
   365→          town: form.town.trim() || null,
   366→          address: form.address.trim() || null,
   367→          lat: form.lat.trim() === '' ? null : Number(form.lat),
   368→          lng: form.lng.trim() === '' ? null : Number(form.lng),
   369→          phone: form.phone.trim() || null,
   370→          managerName: form.managerName.trim() || null,
   371→          operatingHours: hours ?? null,
   372→          deliveryAreas: areas.length > 0 ? areas : null,
   373→          status: form.status,
   374→          isPrimary: form.isPrimary,
   375→        })
   376→        toast.success(`${form.name.trim()} updated`)
   377→      } else {
   378→        await apiSend('/api/supply/branches', 'POST', {
   379→          orgId,
   380→          name: form.name.trim(),
   381→          code: form.code.trim().toUpperCase(),
   382→          kind: form.kind,
   383→          county: form.county.trim(),
   384→          town: form.town.trim() || undefined,
   385→          address: form.address.trim() || undefined,
   386→          lat: form.lat.trim() === '' ? undefined : Number(form.lat),
   387→          lng: form.lng.trim() === '' ? undefined : Number(form.lng),
   388→          phone: form.phone.trim() || undefined,
   389→          managerName: form.managerName.trim() || undefined,
   390→          operatingHours: hours,
   391→          deliveryAreas: areas.length > 0 ? areas : undefined,
   392→        })
   393→        toast.success(`${form.name.trim()} created`)
   394→      }
   395→      onSaved()
   396→    } catch (e) {
   397→      toast.error(e instanceof Error ? e.message : 'Could not save the branch')
   398→    } finally {
   399→      setBusy(false)
   400→    }
   401→  }
   402→
   403→  return (
   404→    <Dialog open onOpenChange={(o) => { if (!o) onClose() }}>
   405→      <DialogContent className="max-h-[90vh] overflow-y-auto scrollbar-thin sm:max-w-xl">
   406→        <DialogHeader>
   407→          <DialogTitle>{editing ? `Edit ${branch?.name}` : 'New branch or warehouse'}</DialogTitle>
   408→          <DialogDescription>
   409→            {editing
   410→              ? 'Changes apply immediately across inventory, transfers and delivery.'
   411→              : 'The first branch of a business becomes its primary automatically.'}
   412→          </DialogDescription>
   413→        </DialogHeader>
   414→
   415→        <div className="grid gap-4">
   416→          <div className="grid gap-4 sm:grid-cols-2">
   417→            <div className="space-y-1.5">
   418→              <Label htmlFor="br-name">Name</Label>
   419→              <Input id="br-name" value={form.name} onChange={(e) => setForm({ ...form, name: e.target.value })} placeholder="e.g. Nairobi CBD workshop" />
   420→            </div>
   421→            <div className="space-y-1.5">
   422→              <Label htmlFor="br-code">Code {!editing && <span className="text-muted-foreground">(short, unique)</span>}</Label>
   423→              <Input
   424→                id="br-code"
   425→                value={form.code}
   426→                onChange={(e) => setForm({ ...form, code: e.target.value.toUpperCase() })}
   427→                disabled={editing}
   428→                maxLength={12}
   429→                placeholder="CBD"
   430→              />
   431→            </div>
   432→          </div>
   433→
   434→          <div className="grid gap-4 sm:grid-cols-3">
   435→            <div className="space-y-1.5">
   436→              <Label>Kind</Label>
   437→              <Select value={form.kind} onValueChange={(v) => setForm({ ...form, kind: v as 'BRANCH' | 'WAREHOUSE' })}>
   438→                <SelectTrigger className="w-full"><SelectValue /></SelectTrigger>
   439→                <SelectContent>
   440→                  <SelectItem value="BRANCH">Branch</SelectItem>
   441→                  <SelectItem value="WAREHOUSE">Warehouse</SelectItem>
   442→                </SelectContent>
   443→              </Select>
   444→            </div>
   445→            <div className="space-y-1.5">
   446→              <Label htmlFor="br-county">County</Label>
   447→              <Select value={form.county} onValueChange={(v) => setForm({ ...form, county: v })}>
   448→                <SelectTrigger id="br-county" className="w-full"><SelectValue /></SelectTrigger>
   449→                <SelectContent className="max-h-64">
   450→                  {KENYA_COUNTIES.map((c) => <SelectItem key={c} value={c}>{c}</SelectItem>)}
   451→                </SelectContent>
   452→              </Select>
   453→            </div>
   454→            <div className="space-y-1.5">
   455→              <Label htmlFor="br-town">Town</Label>
   456→              <Input id="br-town" value={form.town} onChange={(e) => setForm({ ...form, town: e.target.value })} placeholder="e.g. Westlands" />
   457→            </div>
   458→          </div>
   459→
   460→          <div className="space-y-1.5">

   933→    (mode === 'receive' ? receiverName.trim().length >= 2 : true) &&
   934→    (odometerKm === undefined || (Number.isFinite(odometerKm) && odometerKm >= 0))
   935→
   936→  async function upload(files: FileList) {
   937→    setUploading(true)
   938→    try {
   939→      for (const file of Array.from(files).slice(0, 6 - photos.length)) {
   940→        const fd = new FormData()
   941→        fd.append('file', file)
   942→        const d = await apiUpload<{ url: string }>('/api/upload', fd)
   943→        setPhotos((p) => (p.length >= 6 ? p : [...p, d.url]))
   944→      }
   945→    } catch (e) {
   946→      toast.error(e instanceof Error ? e.message : 'Upload failed')
   947→    } finally {
   948→      setUploading(false)
   949→      if (fileRef.current) fileRef.current.value = ''
   950→    }
   951→  }
   952→
   953→  return (
   954→    <Dialog open onOpenChange={(o) => { if (!o) onClose() }}>
   955→      <DialogContent className="max-h-[90vh] overflow-y-auto scrollbar-thin sm:max-w-md">
   956→        <DialogHeader>
   957→          <DialogTitle>
   958→            {mode === 'dispatch' ? `Dispatch ${transfer.refCode}` : `Receive ${transfer.refCode}`}
   959→          </DialogTitle>
   960→          <DialogDescription>
   961→            {mode === 'dispatch'
   962→              ? `Condition report out: ${transfer.fromBranch.name} → ${transfer.toBranch.name}. Odometer and notes become part of the custody history.`
   963→              : `Condition report in at ${transfer.toBranch.name}. On receipt the vehicle's inventory record moves to the destination branch.`}
   964→          </DialogDescription>
   965→        </DialogHeader>
   966→        <div className="space-y-4">
   967→          {mode === 'dispatch' && (
   968→            <div className="space-y-1.5">
   969→              <Label htmlFor="cr-dispatch-ref">Dispatch reference (optional)</Label>
   970→              <Input id="cr-dispatch-ref" value={dispatchRef} onChange={(e) => setDispatchRef(e.target.value)} placeholder={`Default: DSP-${transfer.refCode}`} />
   971→            </div>
   972→          )}
   973→          {mode === 'receive' && (
   974→            <div className="space-y-1.5">
   975→              <Label htmlFor="cr-receiver">Received by (required)</Label>
   976→              <Input id="cr-receiver" value={receiverName} onChange={(e) => setReceiverName(e.target.value)} placeholder="Full name of the receiving staff" />
   977→            </div>
   978→          )}
   979→          <div className="space-y-1.5">
   980→            <Label htmlFor="cr-odometer">Odometer (km, optional)</Label>
   981→            <Input id="cr-odometer" inputMode="numeric" value={odometer} onChange={(e) => setOdometer(e.target.value)} placeholder="e.g. 84500" />
   982→          </div>
   983→          <div className="space-y-1.5">
   984→            <Label htmlFor="cr-notes">Condition notes</Label>
   985→            <Textarea id="cr-notes" rows={3} value={notes} onChange={(e) => setNotes(e.target.value)} placeholder="Scratches, fuel level, missing accessories…" />
   986→          </div>
   987→          <div className="space-y-1.5">
   988→            <Label>Photos ({photos.length}/6)</Label>
   989→            <input
   990→              ref={fileRef}
   991→              type="file"
   992→              accept="image/jpeg,image/png,image/webp"
   993→              multiple
   994→              className="block w-full text-sm text-muted-foreground file:mr-3 file:rounded-md file:border-0 file:bg-muted file:px-3 file:py-2 file:text-sm file:font-medium hover:file:bg-accent"
   995→              onChange={(e) => { if (e.target.files?.length) void upload(e.target.files) }}
   996→              disabled={uploading || photos.length >= 6}
   997→            />
   998→            {uploading && <p className="text-xs text-muted-foreground">Uploading…</p>}
   999→            {photos.length > 0 && (
  1000→              <div className="flex flex-wrap gap-2">
  1001→                {photos.map((url) => (
  1002→                  <span key={url} className="relative">
  1003→                    <img src={url} alt="Condition photo" className="h-14 w-14 rounded-md border object-cover" />
  1004→                    <button
  1005→                      type="button"
  1006→                      aria-label="Remove photo"
  1007→                      className="absolute -right-1.5 -top-1.5 rounded-full border bg-background p-0.5 text-muted-foreground hover:text-destructive"
  1008→                      onClick={() => setPhotos((p) => p.filter((u) => u !== url))}
  1009→                    >
  1010→                      <X className="h-3 w-3" />
  1011→                    </button>
  1012→                  </span>
  1013→                ))}
  1014→              </div>
  1015→            )}
  1016→          </div>
  1017→        </div>
  1018→        <DialogFooter>
  1019→          <Button variant="outline" onClick={onClose}>Cancel</Button>
  1020→          <Button
  1021→            className="min-h-11"
  1022→            disabled={!valid || busy || uploading}
  1023→            onClick={async () => {
  1024→              setBusy(true)
  1025→              try {
  1026→                await onSubmit({
  1027→                  ...(mode === 'dispatch' && dispatchRef.trim() ? { dispatchRef: dispatchRef.trim() } : {}),
  1028→                  ...(mode === 'receive' ? { receivedByName: receiverName.trim() } : {}),
  1029→                  ...(odometerKm !== undefined ? { odometerKm } : {}),
  1030→                  ...(notes.trim() ? { conditionNotes: notes.trim() } : {}),
  1031→                  ...(photos.length > 0 ? { photos } : {}),
  1032→                })
  1033→              } finally { setBusy(false) }
  1034→            }}
  1035→          >
  1036→            {busy ? 'Recording…' : mode === 'dispatch' ? 'Record dispatch' : 'Record receipt'}
  1037→          </Button>
  1038→        </DialogFooter>
  1039→      </DialogContent>
  1040→    </Dialog>
  1041→  )
  1042→}
