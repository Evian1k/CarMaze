   118→/** Current period key as the server writes it, e.g. "2025-11". */
   119→function currentPeriodKey(): string {
   120→  const now = new Date()
   121→  return `${now.getFullYear()}-${String(now.getMonth() + 1).padStart(2, '0')}`
   122→}

   124→// ───────────────────────────── panel ─────────────────────────────
   125→
   126→export function SubscriptionsPanel() {
   127→  const [data, setData] = useState<SubscriptionsData | null>(null)
   128→  const [error, setError] = useState<string | null>(null)
   129→
   130→  const load = useCallback(() => {
   131→    apiGet<SubscriptionsData>('/api/account/subscriptions')
   132→      .then((d) => { setData(d); setError(null) })
   133→      .catch((e) => setError(e instanceof Error ? e.message : 'Could not load your subscriptions'))
   134→  }, [])
   135→
   136→  useEffect(() => { load() }, [load])
   137→
   138→  async function cancelMembership(membershipId: string) {
   139→    try {
   140→      const d = await apiSend<{ message: string }>('/api/memberships/manage', 'PATCH', { membershipId })
   141→      toast.info(d.message)
   142→      load()
   143→    } catch (e) {
   144→      toast.error(e instanceof Error ? e.message : 'Could not cancel the membership')
   145→    }
   146→  }
   147→
   148→  async function acceptCreditTerms(facilityId: string) {
   149→    try {
   150→      const d = await apiSend<{ message?: string }>(`/api/org/credit/${facilityId}`, 'PATCH', { action: 'accept-terms' })
   151→      toast.success(d.message ?? 'Facility activated')
   152→      load()
   153→    } catch (e) {
   154→      toast.error(e instanceof Error ? e.message : 'Could not accept the terms')
   155→    }
   156→  }
   157→
   158→  async function repay(facilityId: string, amountMinor: number) {
   159→    try {
   160→      const d = await apiSend<{ outstandingMinor: number }>(`/api/org/credit/${facilityId}`, 'POST', { kind: 'REPAYMENT', amountMinor })
   161→      toast.success(`Repayment recorded: outstanding balance ${money(d.outstandingMinor)}`)
   162→      load()
   163→    } catch (e) {
   164→      toast.error(e instanceof Error ? e.message : 'Could not record the repayment')
   165→    }
   166→  }
   167→
   168→  if (error) {
   169→    return (
   170→      <Card>
   171→        <CardContent className="flex flex-col items-center gap-3 py-10 text-center">
   172→          <span className="flex h-12 w-12 items-center justify-center rounded-full bg-muted text-muted-foreground"><Wallet className="h-6 w-6" /></span>
   173→          <p className="font-semibold">Subscriptions are unavailable right now</p>
   174→          <p className="text-sm text-muted-foreground">{error}</p>
   175→          <Button variant="outline" className="mt-1 min-h-11" onClick={() => { setData(null); load() }}><RefreshCw className="h-4 w-4" /> Try again</Button>
   176→        </CardContent>
   177→      </Card>
   178→    )
   179→  }
   180→
   181→  if (!data) {
   182→    return (
   183→      <div className="space-y-4" aria-busy="true">
   184→        <div className="grid gap-4 sm:grid-cols-3">{[0, 1, 2].map((i) => <Skeleton key={i} className="h-24 rounded-xl" />)}</div>
   185→        <Skeleton className="h-40 rounded-xl" />
   186→        <Skeleton className="h-40 rounded-xl" />
   187→      </div>
   188→    )
   189→  }
   190→
   191→  const pendingInvoiceCount = data.invoices.filter((i) => i.status === 'PENDING').length
   192→
   193→  return (
   194→    <div className="space-y-6">
   195→      {/* Summary */}
   196→      <div className="grid gap-4 sm:grid-cols-3">
   197→        <StatCard icon={<BadgeCheck className="h-5 w-5" />} label="Active memberships" value={String(data.summary.activeMemberships)} hint="Paid & valid right now" />
   198→        <StatCard
   199→          icon={<CircleAlert className="h-5 w-5" />}
   200→          label="Pending payments"
   201→          value={money(data.summary.pendingPayments)}
   202→          hint={pendingInvoiceCount > 0 ? `${pendingInvoiceCount} unpaid invoice${pendingInvoiceCount === 1 ? '' : 's'}: pay via M-Pesa` : 'Nothing due'}
   203→        />
   204→        <StatCard icon={<Landmark className="h-5 w-5" />} label="Outstanding credit" value={money(data.summary.totalOutstandingCredit)} hint="Across all facilities" />
   205→      </div>
   206→
   207→      <MembershipsSection items={data.memberships} onCancel={cancelMembership} />
   208→
   209→      {data.trustProgrammes.length > 0 && <TrustSection items={data.trustProgrammes} />}
   210→
   211→      {data.creditFacilities.length > 0 && (
   212→        <CreditSection items={data.creditFacilities} onAccept={acceptCreditTerms} onRepay={repay} />
   213→      )}
   214→
   215→      <InvoicesSection invoices={data.invoices} onPaid={load} />
   216→    </div>
   217→  )
   218→}

   357→          </AlertDialogHeader>

   424→function CreditSection({ items, onAccept, onRepay }: {
   425→  items: CreditFacilityItem[]
   426→  onAccept: (facilityId: string) => Promise<void>
   427→  onRepay: (facilityId: string, amountMinor: number) => Promise<void>
   428→}) {
   429→  const [repayTarget, setRepayTarget] = useState<CreditFacilityItem | null>(null)
   430→
   431→  return (
   432→    <section aria-label="Credit facilities" className="space-y-3">
   433→      <h3 className="flex items-center gap-2 text-lg font-semibold"><Landmark className="h-5 w-5 text-primary" /> Payment facilities</h3>
   434→
   435→      <div className="rounded-lg border border-dashed p-3 text-sm text-muted-foreground">
   436→        Credit facilities are offered by businesses you work with. Ask your garage or showroom to send
   437→        an offer; it appears here to accept. Applying never grants credit, and a membership never
   438→        guarantees credit.
   439→      </div>
   440→
   441→      {items.map((f) => {
   442→        const overdue = f.entries.some((e) => e.amountMinor > 0 && !e.paidAt && e.dueDate && new Date(e.dueDate) < new Date())
   443→        const canRepay = (f.status === 'ACTIVE' || f.status === 'SUSPENDED') && f.outstandingMinor > 0
   444→        return (
   445→          <Card key={f.id}>

   453→                    </Badge>

   456→                        <CircleAlert className="h-3 w-3" /> Overdue payment
   457→                      </Badge>
   458→                    )}
   459→                  </CardTitle>
   460→                  <CardDescription className="mt-1">{f.refCode} · {f.business.type === 'GARAGE' ? 'Garage' : 'Showroom'} facility</CardDescription>
   461→                </div>
   462→              </div>
   463→            </CardHeader>
   464→            <CardContent className="space-y-3">
   465→              <div className="grid grid-cols-2 gap-2 sm:grid-cols-4">
   466→                <MoneyStat label="Limit" value={f.limitMinor} />
   467→                <MoneyStat label="Outstanding" value={f.outstandingMinor} strong />
   468→                <MoneyStat label="Available" value={f.availableMinor} />
   469→                <div className="rounded-lg border p-2.5">
   470→                  <p className="text-xs text-muted-foreground">Deposit held</p>
   471→                  <p className="truncate font-semibold tabular-nums">{money(f.depositMinor)}</p>
   472→                </div>
   473→              </div>
   474→
   475→              {f.status === 'OFFERED' && (
   476→                <div className="rounded-lg border border-amber-300 bg-amber-500/10 p-3 dark:border-amber-500/40">
   477→                  <p className="text-sm font-medium text-amber-800 dark:text-amber-300">This business approved a facility for you</p>
   478→                  {f.terms && <p className="mt-1 whitespace-pre-wrap text-sm text-amber-900 dark:text-amber-200">{f.terms}</p>}
   479→                  <p className="mt-2 text-xs text-amber-800/80 dark:text-amber-200/80">
   480→                    Accept the repayment terms to activate it. Nothing is charged until the business records a charge, and you always see it in the ledger below.
   481→                  </p>
   482→                  <Button size="sm" className="mt-2 min-h-11" onClick={() => onAccept(f.id)}>Accept terms</Button>
   483→                </div>
   484→              )}
   485→
   486→              {f.terms && f.status !== 'OFFERED' && (
   487→                <Collapsible>
   488→                  <CollapsibleTrigger className="group inline-flex items-center gap-1 text-sm text-primary hover:underline">
   489→                    Agreed terms <ChevronDown className="h-4 w-4 transition-transform group-data-[state=open]:rotate-180" />
   490→                  </CollapsibleTrigger>
   491→                  <CollapsibleContent>
   492→                    <p className="mt-2 whitespace-pre-wrap rounded-lg border bg-muted/40 p-3 text-sm text-muted-foreground">{f.terms}</p>
   493→                  </CollapsibleContent>
   494→                </Collapsible>
   495→              )}
   496→
   497→              <LedgerTable entries={f.entries.slice(0, 10)} />
   498→
   499→              {canRepay && (
   500→                <div className="flex justify-end">
   501→                  <Button size="sm" className="min-h-11" onClick={() => setRepayTarget(f)}>
   502→                    <Wallet className="h-4 w-4" /> Record a repayment
   503→                  </Button>
   504→                </div>
   505→              )}
   506→            </CardContent>
   507→          </Card>
   508→        )
   509→      })}
   510→
   511→      {/* Repay dialog: honest, ledger-backed */}
   512→      {repayTarget && (
   513→        <RepayDialog
   514→          facility={repayTarget}
   515→          onClose={() => setRepayTarget(null)}
   516→          onSubmit={async (amountMinor) => {
   517→            await onRepay(repayTarget.id, amountMinor)
   518→            setRepayTarget(null)
   519→          }}
   520→        />
   521→      )}
   522→    </section>
   523→  )

   535→function LedgerTable({ entries }: { entries: CreditEntry[] }) {
   536→  if (entries.length === 0) {
   537→    return <p className="rounded-lg border border-dashed p-3 text-sm text-muted-foreground">No ledger entries yet.</p>

   545→            <TableHead className="text-right">Amount</TableHead>

   556→                <TableCell>
   557→                  <span className={cn('font-medium', charge ? 'text-foreground' : 'text-emerald-700 dark:text-emerald-400')}>

   561→                </TableCell>
   562→                <TableCell className={cn('text-right tabular-nums', charge ? '' : 'text-emerald-700 dark:text-emerald-400')}>
   563→                  {charge ? '' : '−'}{money(Math.abs(e.amountMinor))}
   564→                </TableCell>
   565→                <TableCell className="text-right tabular-nums">{money(e.balanceAfterMinor)}</TableCell>
   566→                <TableCell className="text-sm">{e.dueDate ? dateLabel(e.dueDate) : 'Not provided'}</TableCell>
   567→                <TableCell className="text-sm">{e.paidAt ? dateLabel(e.paidAt) : charge ? <span className="text-amber-700 dark:text-amber-400">unpaid</span> : 'Not provided'}</TableCell>
   568→              </TableRow>
   569→            )
   570→          })}
   571→        </TableBody>
   572→      </Table>
   573→    </div>
   574→  )
   575→}

   577→function RepayDialog({ facility, onClose, onSubmit }: {
   578→  facility: CreditFacilityItem
   579→  onClose: () => void
   580→  onSubmit: (amountMinor: number) => Promise<void>
   581→}) {
   582→  const [kes, setKes] = useState('')
   583→  const [busy, setBusy] = useState(false)
   584→  const parsed = Math.round(Number(kes) * 100)
   585→  const valid = Number.isFinite(parsed) && parsed >= 100 && parsed <= facility.outstandingMinor
   586→
   587→  return (
   588→    <Dialog open onOpenChange={(o) => { if (!o) onClose() }}>
   589→      <DialogContent className="sm:max-w-md">
   590→        <DialogHeader>
   591→          <DialogTitle>Record a repayment: {facility.business.name}</DialogTitle>
   592→          <DialogDescription>
   593→            Outstanding balance {money(facility.outstandingMinor)}. Entries are appended to the ledger
   594→            and never edited, so your statement stays verifiable.
   595→          </DialogDescription>
   596→        </DialogHeader>
   597→        <div className="space-y-2">
   598→          <Label htmlFor="repay-amount">Amount (KES)</Label>
   599→          <Input
   600→            id="repay-amount"
   601→            inputMode="decimal"
   602→            value={kes}
   603→            onChange={(e) => setKes(e.target.value)}
   604→            placeholder={`Max ${(facility.outstandingMinor / 100).toLocaleString()}`}
   605→          />
   606→          {!valid && kes !== '' && (
   607→            <p className="text-xs text-rose-600 dark:text-rose-400">Enter an amount between 1 and {(facility.outstandingMinor / 100).toLocaleString()} KES.</p>
   608→          )}
   609→        </div>
   610→        <DialogFooter>
   611→          <Button variant="outline" onClick={onClose}>Cancel</Button>
   612→          <Button
   613→            disabled={!valid || busy}
   614→            className="min-h-11"
   615→            onClick={async () => {
   616→              setBusy(true)
   617→              try { await onSubmit(parsed) } finally { setBusy(false) }
   618→            }}
   619→          >
   620→            {busy ? 'Recording…' : `Record ${valid ? money(parsed) : 'repayment'}`}
   621→          </Button>
   622→        </DialogFooter>
   623→      </DialogContent>
   624→    </Dialog>
   625→  )
   626→}

   686→/** Real M-Pesa STK push for a pending invoice; POST /api/billing/mpesa/stkpush.
   687→ * The callback (not this UI) marks the invoice paid; we simply tell the user
   688→ * honestly what happened and reload. */
   689→function PayInvoiceDialog({ invoice, defaultPhone, onClose, onInitiated }: {
   690→  invoice: InvoiceItem
   691→  defaultPhone: string
   692→  onClose: () => void
   693→  onInitiated: () => void
   694→}) {
   695→  const [phone, setPhone] = useState(defaultPhone)
   696→  const [busy, setBusy] = useState(false)
   697→  const valid = /^(\+?254|0)(1|7)\d{8}$/.test(phone.trim())
   698→
   699→  async function pay() {
   700→    setBusy(true)
   701→    try {
   702→      const d = await apiSend<{ message: string }>('/api/billing/mpesa/stkpush', 'POST', { invoiceId: invoice.id, phone: phone.trim() })
   703→      toast.success(d.message || 'STK push sent: enter your M-Pesa PIN to complete the payment.')
   704→      onInitiated()
   705→    } catch (e) {
   706→      toast.error(e instanceof Error ? e.message : 'Could not start the M-Pesa payment')
   707→    } finally {
   708→      setBusy(false)
   709→    }
   710→  }
   711→
   712→  return (
   713→    <Dialog open onOpenChange={(o) => { if (!o) onClose() }}>
   714→      <DialogContent className="sm:max-w-md">
   715→        <DialogHeader>
   716→          <DialogTitle>Pay {invoice.refCode}: {money(invoice.amountMinor, invoice.currency)}</DialogTitle>
   717→          <DialogDescription>
   718→            We send an M-Pesa STK push to your phone. The payment is only confirmed by the Safaricom
   719→            callback: the invoice stays pending until then.
   720→          </DialogDescription>
   721→        </DialogHeader>
   722→        <div className="space-y-2">
   723→          <Label htmlFor="pay-phone">M-Pesa phone number</Label>
   724→          <Input
   725→            id="pay-phone"
   726→            inputMode="tel"
   727→            value={phone}
   728→            onChange={(e) => setPhone(e.target.value)}
   729→            placeholder="0712 345 678"
   730→          />
   731→        </div>
   732→        <DialogFooter>
   733→          <Button variant="outline" onClick={onClose}>Cancel</Button>
   734→          <Button className="min-h-11" disabled={!valid || busy} onClick={pay}>
   735→            {busy ? 'Sending push…' : 'Send STK push'}
   736→          </Button>
   737→        </DialogFooter>
   738→      </DialogContent>
   739→    </Dialog>
   740→  )
   741→}
