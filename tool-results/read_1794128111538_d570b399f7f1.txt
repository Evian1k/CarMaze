    67→function rideLabel(v: number): string {
    68→  if (v > 0.005) return `Lifted +${v.toFixed(2)}`
    69→  if (v < -0.005) return `Lowered ${v.toFixed(2)}`
    70→  return 'Standard ride height'
    71→}

    74→  const route = useHashRoute()
    75→  const cfgParam = route.query.get('cfg')
    76→  const user = useAuth((s) => s.user)
    77→  const { resolvedTheme } = useTheme()
    78→
    79→  const [webgl, setWebgl] = useState<WebGLState>('checking')

    81→  const [configId, setConfigId] = useState<string | null>(null)
    82→  const [banner, setBanner] = useState<string | null>(null)
    83→  const [saving, setSaving] = useState(false)
    84→  const [shareOpen, setShareOpen] = useState(false)
    85→  const [signinOpen, setSigninOpen] = useState(false)
    86→  const [shareLink, setShareLink] = useState('')
    87→  const [copied, setCopied] = useState(false)
    88→
    89→  // Quote form
    90→  const [quoteOpen, setQuoteOpen] = useState(false)
    91→  const [quoteCounty, setQuoteCounty] = useState('')
    92→  const [quoteTown, setQuoteTown] = useState('')
    93→  const [quoteDesc, setQuoteDesc] = useState('')
    94→  const [quoteVehicle, setQuoteVehicle] = useState('')
    95→  const [descDirty, setDescDirty] = useState(false)
    96→  const [requesting, setRequesting] = useState(false)
    97→
    98→  const controlsRef = useRef<OrbitControlsRef | null>(null)
    99→  const appliedRef = useRef<string | null>(null)
   100→  const configIdRef = useRef<string | null>(null)
   101→  const savedSnapshotRef = useRef<string | null>(null)
   102→
   103→  // ── WebGL capability probe (async so we never setState in the effect body) ──
   104→  useEffect(() => {
   105→    let cancelled = false
   106→    Promise.resolve().then(() => {
   107→      if (cancelled) return
   108→      try {
   109→        const probe = document.createElement('canvas')
   110→        const supported = Boolean(probe.getContext('webgl2') ?? probe.getContext('webgl'))
   111→        setWebgl(supported ? 'ok' : 'unsupported')
   112→      } catch {
   113→        setWebgl('unsupported')
   114→      }
   115→    })
   116→    return () => {
   117→      cancelled = true
   118→    }
   119→  }, [])
   120→
   121→  // ── Restore a shared configuration from ?cfg=<shareId> ──
   122→  useEffect(() => {
   123→    if (!cfgParam || appliedRef.current === cfgParam) return
   124→    appliedRef.current = cfgParam
   125→    let cancelled = false
   126→    apiGet<{ modelId: string; config: CarConfig; createdAt: string; views: number }>(
   127→      `/api/customizer/configs/${encodeURIComponent(cfgParam)}`
   128→    )
   129→      .then((data) => {
   130→        if (cancelled) return
   131→        setConfig(data.config)
   132→        setConfigId(cfgParam)
   133→        configIdRef.current = cfgParam
   134→        savedSnapshotRef.current = JSON.stringify(data.config)
   135→        setShareLink(`${window.location.origin}/#/customizer?cfg=${cfgParam}`)
   136→        setBanner(null)
   137→      })
   138→      .catch(() => {
   139→        if (cancelled) return
   140→        setBanner('This shared configuration is no longer available; showing default setup')
   141→        setConfigId(null)
   142→        configIdRef.current = null
   143→        savedSnapshotRef.current = null
   144→      })
   145→    return () => {
   146→      cancelled = true
   147→    }
   148→  }, [cfgParam])
   149→
   150→  function patch(p: Partial<CarConfig>) {
   151→    setConfig((c) => ({ ...c, ...p }))
   152→  }
   153→
   154→  function resetAll() {
   155→    setConfig(DEFAULT_CONFIG)
   156→    setBanner(null)
   157→    controlsRef.current?.reset()
   158→    toast('Setup reset to the showroom default')
   159→  }
   160→
   161→  /** POST the current config (deduped) and return its share id. */
   162→  async function ensureSaved(current: CarConfig): Promise<string | null> {
   163→    const snapshot = JSON.stringify(current)
   164→    if (configIdRef.current && savedSnapshotRef.current === snapshot) return configIdRef.current
   165→    setSaving(true)
   166→    try {
   167→      const res = await apiSend<{ shareId: string }>('/api/customizer/configs', 'POST', {
   168→        modelId: CUSTOMIZER_MODEL_ID,
   169→        config: current,
   170→      })
   171→      configIdRef.current = res.shareId
   172→      savedSnapshotRef.current = snapshot
   173→      setConfigId(res.shareId)
   174→      setShareLink(`${window.location.origin}/#/customizer?cfg=${res.shareId}`)
   175→      appliedRef.current = res.shareId
   176→      // Make the current URL the share link itself.
   177→      navigate(`customizer?cfg=${res.shareId}`, { replace: true })
   178→      return res.shareId
   179→    } catch (e) {
   180→      toast.error(e instanceof ClientApiError ? e.message : 'Could not save the configuration. Please try again.')
   181→      return null
   182→    } finally {
   183→      setSaving(false)
   184→    }
   185→  }
   186→
   187→  async function handleSaveAndShare() {
   188→    if (!user) {
   189→      setSigninOpen(true)
   190→      return
   191→    }
   192→    const id = await ensureSaved(config)
   193→    if (id) setShareOpen(true)
   194→  }
   195→
   196→  async function handleQuoteClick() {
   197→    if (!user) {
   198→      setSigninOpen(true)
   199→      return
   200→    }
   201→    const id = await ensureSaved(config)
   202→    if (!id) return
   203→    if (!descDirty) setQuoteDesc(describeConfig(config))
   204→    setQuoteOpen(true)
   205→  }

   207→  async function submitQuote() {
   208→    if (!configId) {
   209→      toast.error('Save the configuration first.')
   210→      return
   211→    }
   212→    if (!quoteCounty) {
   213→      toast.error('Select your county so garages can respond.')
   214→      return
   215→    }
   216→    setRequesting(true)
   217→    try {
   218→      const data = await apiSend<QuoteResponse>('/api/service-requests', 'POST', {
   219→        category: 'MODIFICATIONS',
   220→        description: quoteDesc.trim() || describeConfig(config),
   221→        county: quoteCounty,
   222→        town: quoteTown.trim() || undefined,
   223→        configId,
   224→        vehicleDesc: quoteVehicle.trim() || undefined,
   225→      })
   226→      const ref = data.refCode ?? data.request?.refCode ?? data.serviceRequest?.refCode
   227→      toast.success(ref ? `Quote request ${ref} received` : 'Quote request received', {
   228→        description: 'Nearby garages can now quote. Quotes appear in your account requests tab.',
   229→      })
   230→      setQuoteOpen(false)
   231→      setDescDirty(false)
   232→    } catch (e) {
   233→      if (e instanceof ClientApiError && e.status === 401) {
   234→        setQuoteOpen(false)
   235→        setSigninOpen(true)
   236→        return
   237→      }
   238→      toast.error(e instanceof ClientApiError ? e.message : 'Could not submit the request. Please try again.')
   239→    } finally {
   240→      setRequesting(false)
   241→    }
   242→  }

   527→            <Button type="button" variant="secondary" onClick={() => setShareOpen(false)}>

   529→            </Button>
   530→          </DialogFooter>
   531→        </DialogContent>
   532→      </Dialog>

   541→            </DialogDescription>
   542→          </DialogHeader>

   545→              Not now
   546→            </Button>
   547→            <Button
   548→              type="button"
   549→              onClick={() => {
   550→                setSigninOpen(false)
   551→                navigate('signin')
   552→              }}
   553→            >
   554→              Sign in
   555→            </Button>
   556→          </DialogFooter>
   557→        </DialogContent>
   558→      </Dialog>
   559→
   560→      {/* ── Garage quote request dialog ── */}
   561→      <Dialog open={quoteOpen} onOpenChange={setQuoteOpen}>

   568→          </DialogHeader>
   569→
   570→          <div className="space-y-4">
   571→            <div className="flex flex-wrap gap-1.5">

   573→                <span className="inline-block h-3 w-3 rounded-full border border-black/10" style={{ backgroundColor: config.paint }} aria-hidden="true" />
   574→                {paintName(config.paint)} ({config.finish})
   575→              </Badge>

   577→                {RIM_STYLE_OPTIONS.find((r) => r.value === config.rimStyle)?.label} rims · {rimColorName(config.rimColor)}
   578→              </Badge>
   579→              {config.accessory !== 'none' && (
   580→                <Badge variant="outline">{ACCESSORY_OPTIONS.find((a) => a.value === config.accessory)?.label}</Badge>

   582→              <Badge variant="outline">Tint {Math.round(config.glassTint * 100)}%</Badge>

   595→                      {c}

   611→                rows={4}

   615→                  setDescDirty(true)
   616→                }}
   617→                placeholder="Describe the work…"
   618→                maxLength={2000}
   619→              />
   620→            </div>
   621→
   622→            <div className="grid gap-2">
   623→              <Label htmlFor="quote-vehicle">Your vehicle (optional)</Label>
   624→              <Input id="quote-vehicle" value={quoteVehicle} onChange={(e) => setQuoteVehicle(e.target.value)} placeholder="e.g. 2018 Toyota Axio, silver" maxLength={120} />
   625→            </div>
   626→          </div>
   627→
   628→          <DialogFooter className="gap-2">
   629→            <Button type="button" variant="ghost" onClick={() => setQuoteOpen(false)} disabled={requesting}>
   630→              Cancel
   631→            </Button>
   632→            <Button type="button" onClick={submitQuote} disabled={requesting || !quoteCounty} className="gap-2">
   633→              {requesting && <Loader2 className="h-4 w-4 animate-spin" aria-hidden="true" />}
   634→              {requesting ? 'Sending…' : 'Send request'}
   635→            </Button>
   636→          </DialogFooter>
   637→        </DialogContent>
   638→      </Dialog>
   639→    </div>
   640→  )
