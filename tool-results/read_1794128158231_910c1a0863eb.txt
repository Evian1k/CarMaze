   317→type FetchState<T> = { data: T | null; error: FetchError | null; loading: boolean; reload: () => void }
   318→
   319→/** GET hook: all state updates happen in async callbacks (lint-safe). */
   320→function useFetch<T>(url: string | null): FetchState<T> {
   321→  const [completed, setCompleted] = useState<{ key: string; data: T | null; error: FetchError | null } | null>(null)
   322→  const [tick, setTick] = useState(0)
   323→  const key = `${url ?? ''}|${tick}`
   324→
   325→  useEffect(() => {
   326→    if (!url) return
   327→    let cancelled = false
   328→    apiGet<T>(url)
   329→      .then((d) => {
   330→        if (!cancelled) setCompleted({ key, data: d, error: null })
   331→      })
   332→      .catch((e: { status?: number; message?: string }) => {
   333→        if (!cancelled) {
   334→          setCompleted({ key, data: null, error: { status: typeof e?.status === 'number' ? e.status : 0, message: e?.message ?? 'Request failed' } })
   335→        }
   336→      })
   337→    return () => {
   338→      cancelled = true
   339→    }
   340→  }, [url, tick, key])
   341→
   342→  const loading = Boolean(url) && (!completed || completed.key !== key)
   343→  const fresh = completed && url && completed.key.split('|')[0] === url ? completed : null
   344→  const reload = useCallback(() => setTick((t) => t + 1), [])
   345→  return { data: fresh?.data ?? null, error: fresh && fresh.key === key ? fresh.error : null, loading, reload }
   346→}
