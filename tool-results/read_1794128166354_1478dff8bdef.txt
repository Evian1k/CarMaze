   798→// ───────────────────────────── State B: Pending application ─────────────────────────────
   799→
   800→function PendingApplication({ app, onUpdated }: { app: DealerApplication; onUpdated: () => void }) {
   801→  const [docs, setDocs] = useState<DocItem[]>([])
   802→  const [uploading, setUploading] = useState(false)
   803→  const [notes, setNotes] = useState('')
   804→  const [submitting, setSubmitting] = useState(false)
   805→
   806→  const needsInfo = app.status === 'MORE_INFO_REQUIRED'
   807→  const infoEvent = [...app.events].reverse().find((e) => e.type === 'INFO_REQUESTED')
   808→
   809→  async function handleFiles(files: FileList | null) {
   810→    if (!files || files.length === 0) return
   811→    setUploading(true)
   812→    try {
   813→      for (const file of Array.from(files)) {
   814→        const fd = new FormData()
   815→        fd.append('file', file)
   816→        fd.append('kind', 'document')
   817→        const res = await apiUpload<{ url: string; name: string }>('/api/upload', fd)
   818→        setDocs((prev) => [...prev, { name: res.name || file.name, url: res.url }])
   819→      }
   820→      toast.success('Document uploaded')
   821→    } catch (e) {
   822→      toast.error(e instanceof Error ? e.message : 'Upload failed: please try again.')
   823→    } finally {
   824→      setUploading(false)
   825→    }
   826→  }
   827→
   828→  async function resubmit() {
   829→    if (docs.length === 0 && notes.trim().length < 5) {
   830→      toast.error('Add a note or attach the requested documents before resubmitting.')
   831→      return
   832→    }
   833→    setSubmitting(true)
   834→    try {
   835→      await apiSend(`/api/applications/${app.id}`, 'PATCH', {
   836→        action: 'resubmit',
   837→        documents: docs.length > 0 ? docs : undefined,
   838→        notes: notes.trim() || undefined,
   839→      })
   840→      toast.success('Information resubmitted: the review team has been notified.')
   841→      onUpdated()
   842→    } catch (e) {
   843→      toast.error(e instanceof Error ? e.message : 'Could not resubmit.')
   844→    } finally {
   845→      setSubmitting(false)
   846→    }
   847→  }
   848→
   849→  return (
   850→    <div className="mx-auto w-full max-w-3xl px-3 py-8 sm:px-6 sm:py-12">
   851→      <Card>
   852→        <CardHeader>
   853→          <div className="flex flex-wrap items-center justify-between gap-2">
   854→            <CardTitle className="flex items-center gap-2 text-xl">
   855→              <Building2 className="h-5 w-5 text-primary" /> {app.orgName}
   856→            </CardTitle>
   857→            <StatusBadge status={app.status} labels={APPLICATION_STATUS_LABELS} classes={APPLICATION_BADGE_CLASS} />
   858→          </div>
   859→          <CardDescription className="flex flex-wrap items-center gap-x-3 gap-y-1">
   860→            <span className="font-mono text-xs">{app.refCode}</span>
   861→            <span aria-hidden>·</span>
   862→            <span>Submitted {app.submittedAt ? dateLabel(app.submittedAt) : 'Not set'}</span>
   863→            <span aria-hidden>·</span>
   864→            <span className="flex items-center gap-1"><CalendarClock className="h-3.5 w-3.5" /> Initial review target: 24 business hours</span>
   865→          </CardDescription>
   866→        </CardHeader>
   867→        <CardContent className="space-y-5">
   868→          {app.escalated && (
   869→            <Alert className="border-accent/60 bg-accent/10">
   870→              <AlertTriangle className="h-4 w-4" />
   871→              <AlertTitle>Review is taking longer than our target</AlertTitle>
   872→              <AlertDescription>
   873→                Your application has been escalated to senior staff; no action is needed from you. We are sorry for the wait.
   874→              </AlertDescription>
   875→            </Alert>
   876→          )}
   877→
   878→          {needsInfo && (
   879→            <div className="space-y-3 rounded-lg border border-accent/60 bg-accent/10 p-4">
   880→              <div className="flex items-start gap-2">
   881→                <AlertTriangle className="mt-0.5 h-4 w-4 shrink-0" />
   882→                <div className="text-sm">
   883→                  <p className="font-semibold">The review team requested more information</p>
   884→                  {infoEvent?.note && <p className="mt-1 text-muted-foreground">{infoEvent.note}</p>}
   885→                  {app.correctionsDueAt && (
   886→                    <p className="mt-1 text-muted-foreground">Please respond before <strong>{dateLabel(app.correctionsDueAt)}</strong> to keep your application in the queue.</p>
   887→                  )}
   888→                </div>
   889→              </div>
   890→              <Separator />
   891→              <div className="space-y-3">
   892→                <DocUploader
   893→                  idPrefix="rs"
   894→                  docs={docs}
   895→                  uploading={uploading}
   896→                  onAdd={handleFiles}
   897→                  onRemove={(url) => setDocs((prev) => prev.filter((d) => d.url !== url))}
   898→                />
   899→                <div className="grid gap-1.5">
   900→                  <Label htmlFor="rs-notes">Note for the review team</Label>
   901→                  <Textarea id="rs-notes" rows={3} value={notes} onChange={(e) => setNotes(e.target.value)} placeholder="Explain what you corrected or attached…" maxLength={2000} />
   902→                </div>
   903→                <Button onClick={resubmit} disabled={submitting} className="min-h-11">
   904→                  {submitting ? <Loader2 className="h-4 w-4 animate-spin" /> : <Send className="h-4 w-4" />} Resubmit application
   905→                </Button>
   906→              </div>
   907→            </div>
   908→          )}
   909→
   910→          {!needsInfo && app.status !== 'APPROVED' && (
   911→            <p className="text-sm leading-relaxed text-muted-foreground">
   912→              Your application is in the review queue. You will receive an in-app notification as soon as the review team
   913→              updates it: this page updates automatically when you revisit.
   914→            </p>
   915→          )}
   916→          {app.status === 'APPROVED' && (
   917→            <Alert>
   918→              <BadgeCheck className="h-4 w-4 text-primary" />
   919→              <AlertTitle>Application approved</AlertTitle>
   920→              <AlertDescription>
   921→                Your dealership has been verified. If your DealerOS workspace does not appear shortly, sign out and back in
   922→                once: or contact support with your reference {app.refCode}.
   923→              </AlertDescription>
   924→            </Alert>
   925→          )}
   926→
   927→          {/* Event timeline */}
   928→          <div>
   929→            <h3 className="text-sm font-semibold">Application timeline</h3>
   930→            <ol className="mt-3 space-y-0">
   931→              {app.events.map((e, i) => (
   932→                <li key={e.id} className="relative flex gap-3 pb-5 last:pb-0">
   933→                  {i < app.events.length - 1 && <span aria-hidden className="absolute left-[7px] top-4 h-full w-px bg-border" />}
   934→                  <span className={cn('mt-1 h-3.5 w-3.5 shrink-0 rounded-full border-2', ['APPROVED'].includes(e.type) ? 'border-primary bg-primary' : ['INFO_REQUESTED', 'REJECTED', 'ESCALATED'].includes(e.type) ? 'border-accent bg-accent' : 'border-primary/40 bg-background')} />
   935→                  <div className="min-w-0">
   936→                    <p className="text-sm font-medium">{EVENT_LABELS[e.type] ?? e.type}</p>
   937→                    {e.note && <p className="mt-0.5 text-xs leading-relaxed text-muted-foreground">{e.note}</p>}
   938→                    <p className="mt-0.5 text-[11px] text-muted-foreground">
   939→                      {timeAgo(e.createdAt)}{e.actorName ? ` · ${e.actorName}` : ''}
   940→                    </p>
   941→                  </div>
   942→                </li>
   943→              ))}
   944→            </ol>
   945→          </div>
   946→        </CardContent>
   947→      </Card>
   948→    </div>
   949→  )
   950→}
