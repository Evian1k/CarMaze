     4→ * signed-in customers (deliberately separate from the B2B purchase-order

   141→export function ConsumerOrdersTab({ orgId, role, orders, branches }: Props) {
   142→  const [filter, setFilter] = useState<OrderFilter>('ALL')
   143→  const [openId, setOpenId] = useState<string | null>(null)
   144→  const [query, setQuery] = useState('')
   145→
   146→  const canRespond = supplierRoleCan(role, 'order.respond')
   147→
   148→  const list = orders.data?.orders ?? []
   149→  const awaiting = list.filter((o) => o.status === 'SUBMITTED').length
   150→
   151→  const filtered = list.filter((o) => {
   152→    if (filter === 'SUBMITTED' && o.status !== 'SUBMITTED') return false
   153→    if (filter === 'OPEN' && !OPEN_STATUSES.includes(o.status)) return false
   154→    if (filter === 'CLOSED' && !CLOSED_STATUSES.includes(o.status)) return false
   155→    if (query.trim()) {
   156→      const q = query.trim().toLowerCase()
   157→      if (!o.refCode.toLowerCase().includes(q) && !o.buyer.name.toLowerCase().includes(q)) return false
   158→    }
   159→    return true
   160→  })
   161→
   162→  return (
   163→    <div className="space-y-4">
   164→      <div className="flex flex-wrap items-center justify-between gap-3">
   165→        <div>
   166→          <h2 className="text-lg font-semibold tracking-tight">Consumer orders</h2>
   167→          <p className="text-sm text-muted-foreground">
   168→            {awaiting > 0 ? `${awaiting} order${awaiting === 1 ? '' : 's'} awaiting your response.` : 'Parts orders placed directly by customers from your public catalogue.'}
   169→          </p>
   170→        </div>
   171→        <div className="relative w-full sm:w-64">
   172→          <Search className="absolute left-2.5 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground" />
   173→          <Input value={query} onChange={(e) => setQuery(e.target.value)} placeholder="Search ref or customer…" className="pl-8" aria-label="Search consumer orders" />
   174→        </div>
   175→      </div>
   176→
   177→      <div className="flex flex-wrap gap-1.5">
   178→        {(['ALL', 'SUBMITTED', 'OPEN', 'CLOSED'] as OrderFilter[]).map((f) => (
   179→          <button
   180→            key={f}
   181→            type="button"
   182→            onClick={() => setFilter(f)}
   183→            className={cn(
   184→              'rounded-full border px-3 py-1 text-xs transition-colors',
   185→              filter === f ? 'border-primary bg-primary/10 text-primary' : 'border-border text-muted-foreground hover:text-foreground'
   186→            )}
   187→          >
   188→            {f === 'ALL' ? 'All' : f === 'SUBMITTED' ? `Awaiting response${awaiting ? ` (${awaiting})` : ''}` : f === 'OPEN' ? 'Open' : 'Closed'}
   189→          </button>
   190→        ))}
   191→      </div>
   192→
   193→      {orders.loading ? (
   194→        <LoadingBlock label="Loading consumer orders…" />
   195→      ) : orders.error ? (
   196→        <ModuleNotice title="Consumer orders could not load" detail={orders.error.message} retry={orders.reload} />
   197→      ) : filtered.length === 0 ? (
   198→        <EmptyState
   199→          icon={<ClipboardList className="h-8 w-8" />}
   200→          title={list.length === 0 ? 'No consumer orders yet' : 'No orders match the filters'}
   201→          detail={list.length === 0
   202→            ? 'When a signed-in customer orders parts from your published catalogue it lands here for you to accept, dispatch or reject.'
   203→            : 'Try a different search or filter.'}
   204→        />
   205→      ) : (
   206→        <div className="space-y-3">
   207→          {filtered.map((o) => (
   208→            <ConsumerOrderCard key={o.id} order={o} onOpen={() => setOpenId(o.id)} />
   209→          ))}
   210→        </div>
   211→      )}
   212→
   213→      {openId && (
   214→        <ConsumerOrderDetailDialog
   215→          orderId={openId}
   216→          orgId={orgId}
   217→          canRespond={canRespond}
   218→          branches={branches}
   219→          onClose={() => setOpenId(null)}
   220→          onChanged={orders.reload}
   221→        />
   222→      )}
   223→    </div>
   224→  )
   225→}

   258→function ConsumerOrderDetailDialog({ orderId, orgId, canRespond, branches, onClose, onChanged }: {
   259→  orderId: string
   260→  orgId: string
   261→  canRespond: boolean
   262→  branches: FetchState<{ branches: BranchItem[] }>
   263→  onClose: () => void
   264→  onChanged: () => void
   265→}) {
   266→  const detail = useFetch<ConsumerOrderDetail>(`/api/orders/${orderId}`)
   267→  const order = detail.data?.order ?? null
   268→  const [busy, setBusy] = useState<string | null>(null)
   269→
   270→  async function act<T>(key: string, fn: () => Promise<T>): Promise<T | null> {
   271→    setBusy(key)
   272→    try {
   273→      const res = await fn()
   274→      detail.reload()
   275→      onChanged()
   276→      return res
   277→    } catch (e) {
   278→      toast.error(await errMessage(e, 'Action failed: please try again.'))
   279→      return null
   280→    } finally {
   281→      setBusy(null)
   282→    }
   283→  }
   284→
   285→  return (
   286→    <Dialog open onOpenChange={(o) => { if (!o) onClose() }}>
   287→      <DialogContent className="max-h-[94vh] overflow-y-auto scrollbar-thin sm:max-w-3xl">
   288→        {detail.loading || !order ? (
   289→          <div className="space-y-3 py-4" aria-busy="true">
   290→            <Skeleton className="h-7 w-52" />
   291→            <Skeleton className="h-4 w-80" />
   292→            <Skeleton className="h-40 w-full" />
   293→            <Skeleton className="h-40 w-full" />
   294→          </div>
   295→        ) : (
   296→          <>
   297→            <DialogHeader>
   298→              <DialogTitle className="flex flex-wrap items-center gap-2">
   299→                <span className="font-mono">{order.refCode}</span>
   300→                <StatusBadge status={order.status} labels={CO_STATUS_LABELS} classes={CO_BADGE_CLASS} />
   301→                <StatusBadge status={order.paymentStatus} labels={CO_PAYMENT_LABELS} classes="bg-muted text-muted-foreground" />
   302→              </DialogTitle>
   303→              <DialogDescription>
   304→                {order.buyer.name} · {order.county} · {order.deliveryMethod === 'COLLECTION' ? 'customer collects' : 'delivery to customer'}
   305→                {' '}· placed {dateLabel(order.createdAt)}
   306→              </DialogDescription>
   307→            </DialogHeader>
   308→
   309→            {order.status === 'REJECTED' && order.rejectedReason && (
   310→              <Alert variant="destructive">
   311→                <XCircle className="h-4 w-4" />
   312→                <AlertTitle>Rejected</AlertTitle>
   313→                <AlertDescription>{order.rejectedReason}</AlertDescription>
   314→              </Alert>
   315→            )}
   316→
   317→            {/* Actions */}
   318→            {canRespond && order.status === 'SUBMITTED' && (
   319→              <AcceptRejectSection order={order} busy={busy} act={act} branches={branches} />
   320→            )}
   321→            {canRespond && ['ACCEPTED', 'CONFIRMED'].includes(order.status) && (
   322→              <DispatchSection order={order} busy={busy} act={act} />
   323→            )}
   324→            {canRespond && order.status === 'DISPATCHED' && (
   325→              <MarkDeliveredSection order={order} busy={busy} act={act} />
   326→            )}
   327→
   328→            <ItemsTable order={order} />
   329→
   330→            {/* Customer + payment panels */}
   331→            <div className="grid gap-3 sm:grid-cols-2">
   332→              <Card>
   333→                <CardContent className="space-y-2 px-4 py-4 text-sm">
   334→                  <p className="flex items-center gap-2 font-semibold"><MapPin className="h-4 w-4 text-muted-foreground" /> {order.deliveryMethod === 'COLLECTION' ? 'Collection' : 'Delivery'}</p>
   335→                  <dl className="space-y-1 text-xs text-muted-foreground">
   336→                    <div className="flex gap-1"><User className="h-3 w-3 shrink-0" /><span>{order.buyer.name}</span></div>
   337→                    <div className="flex gap-1"><Phone className="h-3 w-3 shrink-0" /><span className="font-mono">{order.phone}</span></div>
   338→                    <div className="flex gap-1"><MapPin className="h-3 w-3 shrink-0" /><span>{order.address}, {order.county}</span></div>
   339→                    {order.fulfillmentBranch && <div>Fulfilment: {order.fulfillmentBranch.name}</div>}
   340→                    {order.courierName && <div>Courier: {order.courierName}</div>}
   341→                    {order.otpHint && (
   342→                      <div className="rounded-md bg-muted/60 px-2 py-1.5">
   343→                        Collection code ends <span className="font-mono font-semibold text-foreground">··{order.otpHint}</span>: the customer received the full code in their notifications. Ask them to read it out at handover.
   344→                      </div>
   345→                    )}
   346→                    {order.notes && <div className="rounded-md bg-muted/60 px-2 py-1.5">Customer note: “{order.notes}”</div>}
   347→                  </dl>
   348→                </CardContent>
   349→              </Card>
   350→              <Card>
   351→                <CardContent className="space-y-2 px-4 py-4 text-sm">
   352→                  <p className="flex items-center gap-2 font-semibold"><ShieldCheck className="h-4 w-4 text-muted-foreground" /> Payment</p>
   353→                  <div className="flex flex-wrap gap-1.5">
   354→                    <StatusBadge status={order.paymentStatus} labels={CO_PAYMENT_LABELS} classes="bg-muted text-muted-foreground" />
   355→                    {order.invoice && (
   356→                      <Badge variant="outline" className="gap-1 text-[10px]">
   357→                        <BadgeCheck className="h-3 w-3" /> Invoice {order.invoice.refCode} · {order.invoice.status}
   358→                      </Badge>
   359→                    )}
   360→                  </div>
   361→                  {order.invoice ? (
   362→                    <p className="text-xs text-muted-foreground">
   363→                      A {money(order.invoice.amountMinor, order.currency)} invoice was issued to the customer
   364→                      {order.invoice.dueDate ? `: due ${dateLabel(order.invoice.dueDate)}` : ''}.
   365→                      {order.invoice.status === 'PAID' ? ' It has been paid.' : ' The customer pays it via M-Pesa from their account; states update only when the provider confirms.'}
   366→                    </p>
   367→                  ) : (
   368→                    <p className="text-xs text-muted-foreground">No invoice yet: one is issued automatically when you accept the order.</p>
   369→                  )}
   370→                </CardContent>
   371→              </Card>
   372→            </div>
   373→
   374→            <Timeline order={order} />
   375→
   376→            <DialogFooter className="justify-start">
   377→              <Button variant="ghost" size="sm" onClick={onClose}>Close</Button>
   378→            </DialogFooter>
   379→          </>
   380→        )}
   381→      </DialogContent>
   382→    </Dialog>
   383→  )
   384→}

   386→// ───────────────────────────── Accept / reject ─────────────────────────────
   387→
   388→function AcceptRejectSection({ order, busy, act, branches }: {
   389→  order: NonNullable<ConsumerOrderDetail['order']>
   390→  busy: string | null
   391→  act: (key: string, fn: () => Promise<unknown>) => Promise<unknown>
   392→  branches: FetchState<{ branches: BranchItem[] }>
   393→}) {
   394→  const [mode, setMode] = useState<'choose' | 'accept' | 'reject'>('choose')
   395→  const [branchId, setBranchId] = useState('')
   396→  const [reason, setReason] = useState('')
   397→
   398→  const activeBranches = (branches.data?.branches ?? []).filter((b) => b.status === 'ACTIVE')
   399→  const selected = activeBranches.find((b) => b.id === branchId) ?? null
   400→
   401→  if (mode === 'choose') {
   402→    return (
   403→      <Card className="border-primary/30">
   404→        <CardContent className="space-y-3 px-4 py-4">
   405→          <div>
   406→            <p className="flex items-center gap-2 text-sm font-semibold"><Inbox className="h-4 w-4 text-primary" /> Respond to this order</p>
   407→            <p className="text-xs text-muted-foreground">
   408→              Accepting takes the goods off your shelf at the fulfilling branch and issues a payable invoice to the customer. Rejecting asks for a reason the customer will see.
   409→            </p>
   410→          </div>
   411→          <div className="flex flex-wrap gap-2">
   412→            <Button className="min-h-11" disabled={busy !== null} onClick={() => setMode('accept')}>
   413→              <CheckCircle2 className="h-4 w-4" /> Accept
   414→            </Button>
   415→            <Button variant="ghost" className="min-h-11 text-destructive hover:text-destructive" disabled={busy !== null} onClick={() => setMode('reject')}>
   416→              <XCircle className="h-4 w-4" /> Reject
   417→            </Button>
   418→          </div>
   419→        </CardContent>
   420→      </Card>
   421→    )
   422→  }
   423→
   424→  if (mode === 'accept') {
   425→    return (
   426→      <Card className="border-primary/30">
   427→        <CardContent className="space-y-3 px-4 py-4">
   428→          <div>
   429→            <p className="flex items-center gap-2 text-sm font-semibold"><PackageCheck className="h-4 w-4 text-primary" /> Accept from a branch</p>
   430→            <p className="text-xs text-muted-foreground">
   431→              Pick the warehouse that holds the stock. Leave it on “best stocked” to let the system choose the branch that can cover every line.
   432→            </p>
   433→          </div>
   434→          <div className="space-y-1.5">
   435→            <Label htmlFor="co-branch">Fulfilling branch / warehouse</Label>
   436→            <Select value={branchId || undefined} onValueChange={setBranchId}>
   437→              <SelectTrigger id="co-branch" aria-label="Fulfilling branch"><SelectValue placeholder="Best stocked branch (auto)" /></SelectTrigger>
   438→              <SelectContent className="max-h-64">
   439→                {activeBranches.map((b) => (
   440→                  <SelectItem key={b.id} value={b.id}>{b.name} ({b.kind === 'WAREHOUSE' ? 'Warehouse' : 'Branch'} · {b.county})</SelectItem>
   441→                ))}
   442→                {activeBranches.length === 0 && <div className="px-3 py-2 text-xs text-muted-foreground">No active branches; add one under Warehouses first.</div>}
   443→              </SelectContent>
   444→            </Select>
   445→          </div>
   446→          <div className="flex flex-wrap justify-end gap-2">
   447→            <Button variant="ghost" onClick={() => { setMode('choose'); setBranchId('') }} disabled={busy !== null}>Back</Button>
   448→            <Button
   449→              className="min-h-11"
   450→              disabled={busy !== null}
   451→              onClick={() =>
   452→                act('accept', () =>
   453→                  apiSend(`/api/orders/${order.id}`, 'POST', {
   454→                    action: 'accept',
   455→                    fulfillmentBranchId: selected ? selected.id : undefined,
   456→                  })
   457→                ).then((res) => {
   458→                  if (res) {
   459→                    const message = (res as { message?: string }).message
   460→                    if (message) toast.success(message)
   461→                    setMode('choose')
   462→                  }
   463→                })
   464→              }
   465→            >
   466→              {busy === 'accept' ? <Loader2 className="h-4 w-4 animate-spin" /> : <CheckCircle2 className="h-4 w-4" />} Accept order
   467→            </Button>
   468→          </div>
   469→        </CardContent>
   470→      </Card>
   471→    )
   472→  }
   473→
   474→  return (
   475→    <Card className="border-destructive/30">
   476→      <CardContent className="space-y-3 px-4 py-4">
   477→        <div>
   478→          <p className="flex items-center gap-2 text-sm font-semibold text-destructive"><XCircle className="h-4 w-4" /> Reject this order</p>
   479→          <p className="text-xs text-muted-foreground">The customer sees your reason. Nothing has been reserved yet, so no stock changes.</p>
