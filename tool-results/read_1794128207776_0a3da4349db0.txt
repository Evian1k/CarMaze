   137→export function ProductDetail({ productId }: { productId: string }) {
   138→  const { user } = useAuth()
   139→  const hydrateSaved = useSavedSupply((s) => s.hydrate)
   140→  const [state, setState] = useState<{ key: string; data: ProductResponse } | null>(null)
   141→  const [errState, setErrState] = useState<{ key: string; message: string } | null>(null)
   142→  const [reloadKey, setReloadKey] = useState(0)
   143→  // Derived loading/error keyed to the current request (no sync setState in effects)
   144→  const fetchKey = `${productId}|${reloadKey}`
   145→  const loading = state?.key !== fetchKey
   146→  const error = errState?.key === fetchKey ? errState.message : null
   147→  const data = state?.data ?? null
   148→
   149→  // Deep links skip the landing: make sure saved bookmarks are hydrated.
   150→  useEffect(() => {
   151→    if (user) void hydrateSaved()
   152→  }, [user, hydrateSaved])
   153→
   154→  useEffect(() => {
   155→    let active = true
   156→    const key = `${productId}|${reloadKey}`
   157→    apiGet<ProductResponse>(`/api/supply/products/${encodeURIComponent(productId)}`)
   158→      .then((d) => { if (active) { setState({ key, data: d }); setErrState(null) } })
   159→      .catch((e: Error) => { if (active) setErrState({ key, message: e.message }) })
   160→    return () => { active = false }
   161→  }, [productId, reloadKey])
   162→
   163→  if (loading) return <ProductDetailSkeleton />
   164→  if (error || !data) {
   165→    return (
   166→      <div className="mx-auto w-full max-w-2xl px-4 py-20 text-center">
   167→        <Package className="mx-auto h-10 w-10 text-muted-foreground" aria-hidden="true" />
   168→        <p className="mt-4 font-medium">Could not load this product</p>
   169→        <p className="mt-1 text-sm text-muted-foreground">{error ?? 'It may no longer be listed.'}</p>
   170→        <div className="mt-6 flex justify-center gap-2">
   171→          <Button variant="outline" className="h-11 px-6" onClick={() => setReloadKey((k) => k + 1)}>Retry</Button>
   172→          <Button className="h-11 px-6" onClick={() => navigate('/supply')}>Back to catalogue</Button>
   173→        </div>
   174→      </div>
   175→    )
   176→  }
   177→
   178→  return <ProductDetailBody key={`${data.product.id}|${reloadKey}`} data={data} isOwner={data.isOwner} signedIn={Boolean(user)} onReviewed={() => setReloadKey((k) => k + 1)} />
   179→}

   238→
   239→      {isOwner && (
   240→        <Alert className="mb-4">
   241→          <Info className="h-4 w-4" aria-hidden="true" />
   242→          <AlertTitle>This is your listing</AlertTitle>
   243→          <AlertDescription>You manage this product from your supplier dashboard.</AlertDescription>
   244→        </Alert>
   245→      )}
   246→
   247→      {/* Top: gallery + purchase panel */}
   248→      <div className="grid gap-6 lg:grid-cols-2">
   249→        {/* Gallery */}
   250→        <div>

   252→            {activeImage && !imgBroken[imgIdx] ? (
   253→              <img src={activeImage} alt={p.name} className="h-full w-full object-cover" onError={() => setImgBroken((m) => ({ ...m, [imgIdx]: true }))} />

   255→              <span className="flex h-full w-full items-center justify-center text-muted-foreground">

   263→              {p.images.map((img, i) => (
   264→                <button
   265→                  key={`${img}-${i}`}
   266→                  onClick={() => setImgIdx(i)}
   267→                  className={`h-16 w-20 shrink-0 overflow-hidden rounded-lg border-2 transition-colors ${i === imgIdx ? 'border-primary' : 'border-transparent hover:border-border'}`}
   268→                  aria-label={`View image ${i + 1} of ${p.name}`}
   269→                  aria-current={i === imgIdx}
   270→                >
   271→                  {img && !imgBroken[i] ? (
   272→                    <img src={img} alt="" className="h-full w-full object-cover" onError={() => setImgBroken((m) => ({ ...m, [i]: true }))} />
   273→                  ) : (
   274→                    <span className="flex h-full items-center justify-center bg-muted text-muted-foreground">
   275→                      <Package className="h-4 w-4" aria-hidden="true" />
   276→                    </span>

   280→            </div>

   300→            {' '}· Ref {p.refCode}
   301→          </p>

   305→              <Stars avg={p.ratingAvg} size="md" />

   321→              <p className="mt-1 text-sm text-muted-foreground">

   326→            <div className="mt-2"><StockStatus qty={p.availableQty} /></div>
   327→
   328→            {/* Qty stepper */}
   329→            <div className="mt-4 flex flex-wrap items-center gap-3">

   408→              <div className="min-w-0 flex-1">
   409→                <div className="flex flex-wrap items-center gap-2">

   423→                  </p>
   424→                )}
   425→                {p.supplier.supplierRating != null && p.supplier.supplierRating > 0 && (
   426→                  <div className="mt-1 flex items-center gap-1.5">

   431→              </div>

   438→          {/* Per-warehouse availability */}

   617→const ORDER_PHONE_RE = /^(\+?254|0)\d{9}$/

   619→type OrderableProduct = { id: string; name: string; unitPriceMinor: number; currency: string; availableQty: number }

   621→function OrderPartDialog({ open, product, initialQty, onClose }: {
   622→  open: boolean
   623→  product: OrderableProduct
   624→  initialQty: number
   625→  onClose: () => void
   626→}) {
   627→  const [qty, setQty] = useState(String(Math.min(Math.max(1, initialQty), 99)))
   628→  const [deliveryMethod, setDeliveryMethod] = useState<'DIRECT' | 'COLLECTION'>('DIRECT')
   629→  const [county, setCounty] = useState('')
   630→  const [address, setAddress] = useState('')
   631→  const [phone, setPhone] = useState('')
   632→  const [notes, setNotes] = useState('')
   633→  const [busy, setBusy] = useState(false)
   634→
   635→  const qtyNum = Number(qty)
   636→  const validQty = Number.isInteger(qtyNum) && qtyNum >= 1 && qtyNum <= 99
   637→  const subtotal = validQty ? product.unitPriceMinor * qtyNum : 0
   638→  const deliveryFee = deliveryMethod === 'COLLECTION' ? 0 : 25_000
   639→  const canSubmit = validQty && county !== '' && address.trim().length >= 5 && ORDER_PHONE_RE.test(phone.trim()) && !busy
   640→
   641→  function submit() {
   642→    if (!canSubmit) return
   643→    setBusy(true)
   644→    apiSend<{ orders: { refCode: string }[]; message: string }>('/api/orders', 'POST', {
   645→      items: [{ productId: product.id, qty: qtyNum }],
   646→      county,
   647→      address: address.trim(),
   648→      phone: phone.trim(),
   649→      notes: notes.trim() || undefined,
   650→      deliveryMethod,
   651→    })
   652→      .then((res) => {
   653→        toast.success(res.message || `Order ${res.orders.map((o) => o.refCode).join(', ')} placed`)
   654→        onClose()
   655→        navigate('/account?tab=orders')
   656→      })
   657→      .catch((e) => {
   658→        toast.error(e instanceof Error ? e.message : 'Could not place the order: please try again.')
   659→      })
   660→      .finally(() => setBusy(false))
   661→  }
   662→
   663→  return (
   664→    <Dialog open={open} onOpenChange={(o) => { if (!o) onClose() }}>
   665→      <DialogContent className="max-h-[94vh] overflow-y-auto scrollbar-thin sm:max-w-md">
   666→        <DialogHeader>
   667→          <DialogTitle>Order this part</DialogTitle>
   668→          <DialogDescription>
   669→            {product.name}: {money(product.unitPriceMinor, product.currency)} per unit. The supplier confirms stock and accepts before you pay.
   670→          </DialogDescription>
   671→        </DialogHeader>
   672→        <div className="space-y-3">
   673→          <div className="grid grid-cols-2 gap-3">
   674→            <div className="space-y-1.5">
   675→              <Label htmlFor="ord-qty">Quantity (1–99)</Label>
   676→              <Input id="ord-qty" inputMode="numeric" value={qty} maxLength={2}
   677→                onChange={(e) => setQty(e.target.value.replace(/[^\d]/g, ''))} />
   678→            </div>
   679→            <div className="space-y-1.5">
   680→              <Label htmlFor="ord-method">Fulfilment</Label>
   681→              <Select value={deliveryMethod} onValueChange={(v) => setDeliveryMethod(v as 'DIRECT' | 'COLLECTION')}>
   682→                <SelectTrigger id="ord-method"><SelectValue /></SelectTrigger>
   683→                <SelectContent>
   684→                  <SelectItem value="DIRECT">Deliver to me (KES 250)</SelectItem>
   685→                  <SelectItem value="COLLECTION">I collect (free)</SelectItem>
   686→                </SelectContent>
   687→              </Select>
   688→            </div>
   689→          </div>
   690→          <div className="space-y-1.5">
   691→            <Label htmlFor="ord-county">County</Label>
   692→            <Select value={county || undefined} onValueChange={setCounty}>
   693→              <SelectTrigger id="ord-county" aria-label="County"><SelectValue placeholder="Choose your county" /></SelectTrigger>
   694→              <SelectContent className="max-h-64">
   695→                {KENYA_COUNTIES.map((c) => <SelectItem key={c} value={c}>{c}</SelectItem>)}
   696→              </SelectContent>
   697→            </Select>
   698→          </div>
   699→          <div className="space-y-1.5">
   700→            <Label htmlFor="ord-address">Delivery address / collection point</Label>
   701→            <Input id="ord-address" value={address} onChange={(e) => setAddress(e.target.value)} placeholder="e.g. Building, street, town" maxLength={300} />
   702→          </div>
   703→          <div className="space-y-1.5">
   704→            <Label htmlFor="ord-phone">Phone (for delivery contact)</Label>
   705→            <Input id="ord-phone" type="tel" value={phone} onChange={(e) => setPhone(e.target.value)} placeholder="0712 345 678" inputMode="tel" maxLength={15} />
   706→          </div>
   707→          <div className="space-y-1.5">
   708→            <Label htmlFor="ord-notes">Notes for the supplier (optional)</Label>
   709→            <Textarea id="ord-notes" rows={2} value={notes} onChange={(e) => setNotes(e.target.value)} maxLength={500}
   710→              placeholder="e.g. Fits a 2015 Hilux: please confirm compatibility." />
   711→          </div>
   712→          <div className="space-y-1 rounded-lg bg-muted/50 px-3 py-2 text-sm">
   713→            <div className="flex justify-between text-muted-foreground"><span>Subtotal</span><span>{money(subtotal, product.currency)}</span></div>
   714→            <div className="flex justify-between text-muted-foreground"><span>Delivery</span><span>{money(deliveryFee, product.currency)}</span></div>
   715→            <div className="flex justify-between font-semibold"><span>Total</span><span>{money(subtotal + (validQty ? deliveryFee : 0), product.currency)}</span></div>
   716→          </div>
   717→        </div>
   718→        <DialogFooter>
   719→          <Button variant="ghost" onClick={onClose} disabled={busy}>Cancel</Button>
   720→          <Button className="min-h-11" onClick={submit} disabled={!canSubmit}>
   721→            {busy ? <Loader2 className="h-4 w-4 animate-spin" /> : <PackageCheck className="h-4 w-4" />} Place order
   722→          </Button>
   723→        </DialogFooter>
   724→      </DialogContent>
   725→    </Dialog>
   726→  )
   727→}

   740→function ReviewsSection({
   741→  productId,
   742→  reviews,
   743→  ratingAvg,
   744→  ratingCount,
   745→  signedIn,
   746→  onReviewed,
   747→}: {
   748→  productId: string
   749→  reviews: DetailReview[]
   750→  ratingAvg: number | null
   751→  ratingCount: number
   752→  signedIn: boolean
   753→  onReviewed: () => void
   754→}) {
   755→  const [rating, setRating] = useState(0)
   756→  const [comment, setComment] = useState('')
   757→  const [busy, setBusy] = useState(false)
   758→
   759→  function submit() {
   760→    if (rating < 1) {
   761→      toast.error('Pick a star rating first')
   762→      return
   763→    }
   764→    setBusy(true)
   765→    apiSend<{ message: string; verifiedPurchase: boolean }>('/api/supply/reviews', 'POST', {
   766→      productId,
   767→      rating,
   768→      comment: comment.trim() || undefined,
   769→    })
   770→      .then((d) => {
   771→        toast.success(d.verifiedPurchase ? 'Review recorded: marked as a verified purchase' : 'Review recorded. Thank you!')
   772→        setRating(0)
   773→        setComment('')
   774→        onReviewed()
   775→      })
   776→      .catch((e) => {
   777→        const status = (e as { status?: number }).status
   778→        if (status === 409) toast.info('You already reviewed this product')
   779→        else toast.error(e instanceof Error ? e.message : 'Could not record review')
   780→      })
   781→      .finally(() => setBusy(false))
   782→  }
   783→
   784→  return (
   785→    <section aria-label="Product reviews">
   786→      <div className="flex flex-wrap items-center gap-2">
   787→        <h2 className="text-lg font-bold">Reviews</h2>
   788→        {ratingCount > 0 && ratingAvg != null && (
   789→          <span className="flex items-center gap-1.5 text-sm text-muted-foreground">
   790→            <Stars avg={ratingAvg} /> {ratingAvg.toFixed(1)} from {ratingCount} review{ratingCount === 1 ? '' : 's'}
   791→          </span>
   792→        )}
   793→      </div>
   794→
   795→      {reviews.length === 0 ? (
   796→        <p className="mt-3 text-sm text-muted-foreground">No reviews yet: be the first to rate this part.</p>
   797→      ) : (
   798→        <ul className="mt-3 space-y-3">
   799→          {reviews.map((r) => (
   800→            <li key={r.id} className="rounded-xl border border-border p-4">
   801→              <div className="flex flex-wrap items-center gap-2">
   802→                <Stars avg={r.rating} />
   803→                <span className="text-sm font-medium">{r.author}</span>
   804→                <span className="text-xs text-muted-foreground">{timeAgo(r.createdAt)}</span>
   805→                {r.verifiedPurchase && (
   806→                  <Badge variant="outline" className="gap-1 border-primary/40 text-primary">
   807→                    <BadgeCheck className="h-3 w-3" aria-hidden="true" /> Verified purchase
   808→                  </Badge>
   809→                )}
   810→              </div>
   811→              {r.comment && <p className="mt-2 text-sm text-foreground/90">{r.comment}</p>}
   812→            </li>
   813→          ))}
   814→        </ul>
   815→      )}
   816→
   817→      {signedIn ? (
   818→        <div className="mt-5 rounded-xl border border-border p-4">
   819→          <p className="text-sm font-semibold">Write a review</p>
   820→          <div className="mt-2 flex items-center gap-1" role="radiogroup" aria-label="Your star rating">
   821→            {[1, 2, 3, 4, 5].map((i) => (
   822→              <button
   823→                key={i}
   824→                role="radio"
   825→                aria-checked={rating === i}
   826→                aria-label={`${i} star${i === 1 ? '' : 's'}`}
   827→                className="rounded p-1 hover:bg-muted"
   828→                onClick={() => setRating(i)}
   829→              >
   830→                <Star className={`h-6 w-6 ${i <= rating ? 'fill-gold text-gold' : 'text-muted-foreground/50'}`} aria-hidden="true" />
   831→              </button>
   832→            ))}
   833→          </div>
   834→          <Textarea
   835→            className="mt-3"
   836→            rows={3}
   837→            value={comment}
   838→            onChange={(e) => setComment(e.target.value)}
   839→            placeholder="How was the part, packaging, delivery? (optional)"
   840→            maxLength={1000}
   841→          />
   842→          <Button className="mt-3 h-11" onClick={submit} disabled={busy || rating < 1}>
   843→            {busy ? 'Submitting…' : 'Submit review'}
   844→          </Button>
   845→        </div>
   846→      ) : (
   847→        <div className="mt-5 flex flex-wrap items-center justify-between gap-3 rounded-xl border border-dashed p-4">
   848→          <p className="text-sm text-muted-foreground">Bought this part? Sign in to leave a review.</p>
   849→          <Button variant="outline" className="h-11" onClick={() => navigate('signin')}>Sign in to review</Button>
   850→        </div>
   851→      )}
   852→    </section>
   853→  )
   854→}
