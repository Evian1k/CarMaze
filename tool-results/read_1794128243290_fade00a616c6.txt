    35→export function CancelSection({ order, busy, setBusy, reload, role }: {
    36→  order: OrderDetailData
    37→  busy: string
    38→  setBusy: (v: string) => void
    39→  reload: () => void
    40→  role: string
    41→}) {
    42→  const [open, setOpen] = useState(false)
    43→  const [reason, setReason] = useState('')
    44→  const mayCancel = canOrder(role)
    45→  const cancelable = CANCELABLE.includes(order.status) && order.paymentStatus !== 'PROTECTED'
    46→  if (!cancelable) return null
    47→
    48→  async function submit() {
    49→    if (reason.trim().length < 3) {
    50→      toast.error('Give a short cancellation reason (min 3 characters).')
    51→      return
    52→    }
    53→    setBusy('cancel')
    54→    try {
    55→      const res = await apiSend<{ message: string }>(`/api/supply/orders/${order.id}/action`, 'POST', { action: 'cancel', reason: reason.trim() })
    56→      toast.success(res.message || 'Order cancelled.')
    57→      setOpen(false)
    58→      reload()
    59→    } catch (e) {
    60→      toast.error(e instanceof ClientApiError ? e.message : 'Could not cancel the order.')
    61→    } finally {
    62→      setBusy('')
    63→    }
    64→  }
    65→
    66→  return (
    67→    <section className="rounded-xl border border-border bg-card/50 p-4">
    68→      <div className="flex flex-wrap items-center justify-between gap-2">
    69→        <div>
    70→          <h3 className="flex items-center gap-1.5 text-sm font-semibold"><Ban className="h-4 w-4 text-muted-foreground" aria-hidden="true" /> Cancel order</h3>
    71→          <p className="text-xs text-muted-foreground">Possible until dispatch{order.paymentStatus === 'PROTECTED' ? '' : ': stock reservations are released and the supplier is notified'}.</p>
    72→        </div>
    73→        {!open && (
    74→          <Button variant="outline" size="sm" disabled={!mayCancel || busy !== ''} onClick={() => setOpen(true)}>
    75→            Cancel this order
    76→          </Button>
    77→        )}
    78→      </div>
    79→      {!mayCancel && <p className="mt-1 text-xs text-muted-foreground">Cancellation needs the owner, GM or procurement role.</p>}
    80→      {open && (
    81→        <div className="mt-3 space-y-2 rounded-lg border border-border bg-background/40 p-3">
    82→          <Label htmlFor="cancel-reason">Reason *</Label>
    83→          <Textarea id="cancel-reason" value={reason} onChange={(e) => setReason(e.target.value)} maxLength={300} className="min-h-16" placeholder="e.g. Customer job rescheduled: needed next week instead" />
    84→          <div className="flex gap-2">
    85→            <Button variant="destructive" size="sm" onClick={submit} disabled={busy === 'cancel'}>
    86→              {busy === 'cancel' ? <BtnSpinner className="mr-1.5" /> : null} Confirm cancellation
    87→            </Button>
    88→            <Button variant="ghost" size="sm" onClick={() => setOpen(false)}>Keep order</Button>
    89→          </div>
    90→        </div>
    91→      )}
    92→    </section>
    93→  )
    94→}

    98→export function InspectionPanel({ order, busy, setBusy, reload, role }: {
    99→  order: OrderDetailData
   100→  busy: string
   101→  setBusy: (v: string) => void
   102→  reload: () => void
   103→  role: string
   104→}) {
   105→  const d = order.delivery
   106→  const active = order.status === 'DELIVERED' && d?.status === 'DELIVERED'
   107→  const mayInspect = canApprove(role) // inspect requires order.approve server-side
   108→  const [result, setResult] = useState<'ACCEPT' | 'PARTIAL' | 'ISSUES'>('ACCEPT')
   109→  const [qtys, setQtys] = useState<Record<string, number>>(() =>
   110→    Object.fromEntries(order.items.map((it) => [it.id, it.qtyAccepted ?? it.qty])))
   111→  const [notes, setNotes] = useState<Record<string, string>>({})
   112→  const [note, setNote] = useState('')
   113→  const [photos, setPhotos] = useState<string[]>([])
   114→
   115→  if (!active) return null
   116→
   117→  async function submit() {
   118→    setBusy('inspect')
   119→    try {
   120→      const res = await apiSend<{ message: string; acceptedTotalMinor: number }>(`/api/supply/orders/${order.id}/action`, 'POST', {
   121→        action: 'inspect',
   122→        result,
   123→        items: order.items.map((it) => ({ itemId: it.id, qtyAccepted: result === 'ACCEPT' ? it.qty : Math.min(qtys[it.id] ?? 0, it.qty), ...(notes[it.id]?.trim() ? { note: notes[it.id].trim() } : {}) })),
   124→        ...(note.trim() ? { note: note.trim() } : {}),
   125→        photos,
   126→      })
   127→      toast.success(res.message || 'Inspection recorded.', { description: `Accepted value: ${money(res.acceptedTotalMinor, order.currency)}` })
   128→      reload()
   129→    } catch (e) {
   130→      toast.error(e instanceof ClientApiError ? e.message : 'Could not record the inspection.')
   131→    } finally {
   132→      setBusy('')
   133→    }
   134→  }
   135→
   136→  return (
   137→    <section className="space-y-3 rounded-xl border border-emerald-500/40 bg-emerald-500/5 p-4">
   138→      <h3 className="flex items-center gap-1.5 text-sm font-semibold">
   139→        <ClipboardCheck className="h-4 w-4 text-emerald-600 dark:text-emerald-400" aria-hidden="true" /> Delivery inspection
   140→      </h3>
   141→      <p className="text-sm text-muted-foreground">
   142→        Check every line against the job sheet. What you accept here is what the supplier gets paid; shortfalls raise an automatic partial refund request.
   143→      </p>
   144→      <ul className="space-y-2">
   145→        {order.items.map((it) => (
   146→          <li key={it.id} className="flex flex-wrap items-center gap-3 rounded-lg border border-border bg-background/40 p-3">
   147→            <Img src={it.image} alt={it.name} className="h-10 w-10 shrink-0 rounded border border-border" />
   148→            <div className="min-w-0 flex-1">
   149→              <p className="truncate text-sm font-medium">{it.name}</p>
   150→              <p className="text-xs text-muted-foreground">Ordered {it.qty} · {money(it.unitPriceMinor, order.currency)}/unit</p>
   151→            </div>
   152→            <div className="flex items-center gap-2">
   153→              <Label className="text-xs text-muted-foreground">Accepted</Label>
   154→              <QtyStepper value={qtys[it.id] ?? 0} min={0} max={it.qty} onChange={(v) => setQtys((s) => ({ ...s, [it.id]: v }))} />
   155→            </div>
   156→            <Input
   157→              value={notes[it.id] ?? ''} onChange={(e) => setNotes((s) => ({ ...s, [it.id]: e.target.value }))}
   158→              placeholder="Line note (optional)" className="h-8 w-full sm:w-48" maxLength={300} aria-label={`Note for ${it.name}`}
   159→            />
   160→          </li>
   161→        ))}
   162→      </ul>
   163→      <div className="grid grid-cols-1 gap-2 sm:grid-cols-3">
   164→        {(['ACCEPT', 'PARTIAL', 'ISSUES'] as const).map((r) => (
   165→          <button
   166→            key={r} type="button" onClick={() => setResult(r)} aria-pressed={result === r}
   167→            className={cn(
   168→              'rounded-lg border p-2.5 text-left text-xs transition-colors',
   169→              result === r ? 'border-primary bg-primary/10 text-foreground' : 'border-border bg-card text-muted-foreground hover:border-primary/40'
   170→            )}
   171→          >
   172→            <span className="block text-sm font-semibold">{r === 'ACCEPT' ? 'Accept all' : r === 'PARTIAL' ? 'Partial acceptance' : 'Issues reported'}</span>
   173→            <span className="mt-0.5 block leading-snug">
   174→              {r === 'ACCEPT' ? 'Everything matches the order.' : r === 'PARTIAL' ? 'Accept the quantities set above.' : 'Record problems: release is blocked until resolved.'}
   175→            </span>
   176→          </button>
   177→        ))}
   178→      </div>
   179→      <Textarea value={note} onChange={(e) => setNote(e.target.value)} maxLength={1000} className="min-h-16" placeholder="Overall inspection note (optional)" aria-label="Inspection note" />
   180→      <PhotoUploader photos={photos} onChange={setPhotos} label="Inspection photos (damages, shortfall, packaging)" />
   181→      {mayInspect ? (
   182→        <Button onClick={submit} disabled={busy === 'inspect'}>
   183→          {busy === 'inspect' ? <BtnSpinner className="mr-1.5" /> : null} Record inspection
   184→        </Button>
   185→      ) : (
   186→        <p className="text-xs text-muted-foreground">Inspection is recorded by the owner or GM.</p>
   187→      )}
   188→    </section>
   189→  )
   190→}

   194→export function PostDeliveryActions({ order, busy, setBusy, reload, role }: {
   195→  order: OrderDetailData
   196→  busy: string
   197→  setBusy: (v: string) => void
   198→  reload: () => void
   199→  role: string
   200→}) {
   201→  const postDelivery = stepIndex(order.status) >= 4 || order.status === 'DISPUTED'
   202→  if (!postDelivery) return null
   203→  const mayRelease = canPay(role)
   204→  const mayReturn = canOrder(role)
   205→  const mayDispute = canDispute(role)
   206→  const releasable = order.status === 'INSPECTED' && order.paymentStatus === 'PROTECTED'
   207→  const accepted = order.acceptedTotalMinor ?? order.totalMinor
   208→
   209→  return (
   210→    <section className="space-y-3 rounded-xl border border-border bg-card/50 p-4">
   211→      <h3 className="flex items-center gap-1.5 text-sm font-semibold">
   212→        <Unlock className="h-4 w-4 text-primary" aria-hidden="true" /> Next steps
   213→      </h3>
   214→
   215→      {order.status === 'INSPECTED' && (
   216→        <div className="space-y-2 rounded-lg border border-teal-500/40 bg-teal-500/5 p-3">
   217→          <p className="text-sm">
   218→            Release <span className="font-bold tabular-nums">{money(accepted, order.currency)}</span> of the protected funds to the supplier.
   219→          </p>
   220→          <p className="text-xs text-muted-foreground">
   221→            Platform commission is deducted from the supplier&apos;s proceeds at this point: you are never charged extra.
   222→          </p>
   223→          <Button disabled={!releasable || !mayRelease || busy === 'release'} onClick={async () => {
   224→            setBusy('release')
   225→            try {
   226→              const res = await apiSend<{ message: string; releasedMinor: number; commissionMinor: number }>(`/api/supply/orders/${order.id}/action`, 'POST', { action: 'release' })
   227→              toast.success(res.message || 'Payment released.', { description: `Released ${money(res.releasedMinor, order.currency)} · commission ${money(res.commissionMinor, order.currency)} recorded separately.` })
   228→              reload()
   229→            } catch (e) {
   230→              toast.error(e instanceof ClientApiError ? e.message : 'Release failed.')
   231→            } finally {
   232→              setBusy('')
   233→            }
   234→          }}>
   235→            {busy === 'release' ? <BtnSpinner className="mr-1.5" /> : <Unlock className="mr-1.5 h-4 w-4" aria-hidden="true" />}
   236→            Request release
   237→          </Button>
   238→          {!mayRelease && <p className="text-xs text-muted-foreground">Release needs the owner, GM, procurement or accountant role.</p>}
   239→          {!releasable && order.paymentStatus !== 'PROTECTED' && (
   240→            <p className="text-xs text-muted-foreground">Funds are not in a protected state; release follows the provider&apos;s confirmation only.</p>
   241→          )}
   242→        </div>
   243→      )}
   244→
   245→      <div className="flex flex-wrap gap-2">
   246→        <DisputeDialog order={order} mayDispute={mayDispute} busy={busy} setBusy={setBusy} reload={reload} />
   247→        <ReturnDialog order={order} mayReturn={mayReturn} busy={busy} setBusy={setBusy} reload={reload} />
   248→        {order.items.some((it) => it.warrantyMonths > 0) && (
   249→          <WarrantyDialog order={order} mayClaim={mayReturn} busy={busy} setBusy={setBusy} reload={reload} />
   250→        )}
   251→      </div>
   252→    </section>
   253→  )
   254→}

   256→function PhotoUploader({ photos, onChange, label, max = 6 }: {
   257→  photos: string[]
   258→  onChange: (urls: string[]) => void
   259→  label: string
   260→  max?: number
   261→}) {
   262→  const [uploading, setUploading] = useState(false)
   263→  async function handleFiles(files: FileList | null) {
   264→    if (!files || files.length === 0) return
   265→    setUploading(true)
   266→    const next = [...photos]
   267→    try {
   268→      for (const f of Array.from(files).slice(0, max - photos.length)) {
   269→        const url = await uploadOne(f)
   270→        next.push(url)
   271→      }
   272→      onChange(next)
   273→      toast.success('Photo attached.')
   274→    } catch (e) {
   275→      toast.error(e instanceof ClientApiError ? e.message : 'Upload failed. JPG, PNG, WebP or PDF up to 8 MB.')
   276→    } finally {
   277→      setUploading(false)
   278→    }
   279→  }
   280→  return (
   281→    <div className="space-y-1.5">
   282→      <Label>{label}</Label>
   283→      {photos.length > 0 && (
   284→        <div className="flex flex-wrap gap-2">
   285→          {photos.map((url, i) => (
   286→            <div key={url} className="group relative">
   287→              <Img src={url} alt={`Attached evidence ${i + 1}`} className="h-14 w-14 rounded-md border border-border" />
   288→              <button
   289→                type="button" aria-label="Remove photo"
   290→                className="absolute -right-1.5 -top-1.5 flex h-5 w-5 items-center justify-center rounded-full border border-border bg-background text-muted-foreground shadow-sm hover:text-destructive"
   291→                onClick={() => onChange(photos.filter((u) => u !== url))}
   292→              >
   293→                <Trash2 className="h-3 w-3" aria-hidden="true" />
   294→              </button>
   295→            </div>
   296→          ))}
   297→        </div>
   298→      )}
   299→      {photos.length < max && (
   300→        <label className="flex cursor-pointer items-center justify-center gap-2 rounded-lg border border-dashed border-border bg-background/40 px-4 py-3 text-sm text-muted-foreground transition-colors hover:border-primary/40 hover:text-foreground">
   301→          {uploading ? <BtnSpinner /> : <Upload className="h-4 w-4" aria-hidden="true" />}
   302→          {uploading ? 'Uploading…' : 'Attach photos'}
   303→          <input
   304→            type="file" accept="image/jpeg,image/png,image/webp" multiple className="hidden" disabled={uploading}
   305→            onChange={(e) => { handleFiles(e.target.files); e.currentTarget.value = '' }} aria-label="Attach photos"
   306→          />
   307→        </label>
   308→      )}
   309→    </div>
   310→  )

   395→function ReturnDialog({ order, mayReturn, busy, setBusy, reload }: {

   407→  const valid = reason.trim().length >= 5 && items.length > 0
   408→
   409→  return (
   410→    <>
   411→      <Button variant="outline" disabled={!mayReturn || returnables.length === 0} onClick={() => setOpen(true)}>
   412→        <Undo2 className="mr-1.5 h-4 w-4" aria-hidden="true" /> Request return

   418→            <DialogDescription>The supplier is notified with the quantities you select.</DialogDescription>
   419→          </DialogHeader>
   420→          <div className="space-y-3">
   421→            {returnables.length === 0 ? (

   438→              <Textarea id="return-reason" value={reason} onChange={(e) => setReason(e.target.value)} maxLength={300} className="min-h-16" />

   444→              setBusy('return')
   445→              try {

   566→type TimelineEntry = {
   567→  id: string
   568→  when: string
   569→  type: string
   570→  actorName: string
   571→  actorSide: string
   572→  note: string | null
   573→  source: 'ORDER' | 'DELIVERY'
   574→}
   575→
   576→export function Timeline({ order }: { order: OrderDetailData }) {
   577→  const entries = useMemo<TimelineEntry[]>(() => {
   578→    const list: TimelineEntry[] = [
   579→      ...order.events.map((e) => ({ id: e.id, when: e.createdAt, type: e.type, actorName: e.actorName, actorSide: e.actorSide, note: e.note, source: 'ORDER' as const })),
   580→      ...(order.delivery?.events ?? []).map((e) => ({ id: `dlv-${e.id}`, when: e.createdAt, type: e.type, actorName: e.actorName, actorSide: e.actorSide, note: e.note, source: 'DELIVERY' as const })),
   581→    ]
   582→    return list.sort((a, b) => new Date(a.when).getTime() - new Date(b.when).getTime())
   583→  }, [order])
   584→
   585→  return (
   586→    <section className="space-y-2 rounded-xl border border-border bg-card/50 p-4">
   587→      <h3 className="text-sm font-semibold">Timeline ({entries.length})</h3>
   588→      <ol className="space-y-3">
   589→        {entries.map((e) => (
   590→          <li key={e.id} className="flex gap-3">
   591→            <span className="mt-1.5 h-2 w-2 shrink-0 rounded-full bg-primary/60" aria-hidden="true" />
   592→            <div className="min-w-0 flex-1">
   593→              <div className="flex flex-wrap items-center gap-1.5">
   594→                <SideChip side={e.actorSide} />
   595→                <span className="text-xs font-semibold">{e.type.replaceAll('_', ' ')}</span>
   596→                <span className="text-xs text-muted-foreground">· {e.actorName} · {e.source === 'DELIVERY' ? 'delivery log' : 'order log'} · <When at={e.when} /></span>
   597→              </div>
   598→              {e.note && <p className="mt-0.5 text-sm text-muted-foreground">{e.note}</p>}
   599→            </div>
   600→          </li>
   601→        ))}
   602→      </ol>
   603→    </section>
   604→  )
   605→}
