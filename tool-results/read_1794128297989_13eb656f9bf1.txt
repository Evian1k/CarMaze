   412→                        <td className="py-2.5 pr-3 text-xs text-muted-foreground">{o.garage.name} → {o.supplier.name}</td>
   413→                        <td className="py-2.5 pr-3 tabular-nums">{money(o.totalMinor)}</td>
   414→                        <td className="py-2.5 pr-3">
   415→                          <Badge variant="outline" className={o.paymentStatus === 'PROTECTED' ? 'border-primary/40 text-primary' : ''}>
   416→                            {o.status} · {o.paymentStatus}
   417→                          </Badge>
   418→                        </td>
   419→                        <td className="py-2.5 pr-3">
   420→                          {o.paymentStatus !== 'PROTECTED' && ['CONFIRMED'].includes(o.status) && (
   421→                            <Button size="sm" variant="outline" onClick={() => { setReconcileDialog(o); setReconcileRef('') }}>Reconcile</Button>
   422→                          )}
   423→                        </td>
   424→                      </tr>
   425→                    ))}
   426→                  </tbody>
   427→                </table>
   428→              </div>
   429→            </CardContent>
   430→          </Card>
   431→
   432→          <Card>
   433→            <CardHeader>
   434→              <CardTitle className="text-base">Refund queue</CardTitle>
   435→              <CardDescription>From partial inspections and dispute resolutions. Processing records the provider reference.</CardDescription>
   436→            </CardHeader>
   437→            <CardContent className="space-y-3">
   438→              {data.refundQueue.length === 0 && <p className="text-sm text-muted-foreground">No refunds awaiting processing.</p>}
   439→              {data.refundQueue.map((r) => (
   440→                <div key={r.id} className="flex flex-wrap items-center justify-between gap-2 rounded-lg border p-3">
   441→                  <div>
   442→                    <p className="text-sm font-medium">{r.refCode} · {money(r.amountMinor)} <Badge variant="outline" className="ml-1">{r.kind}</Badge></p>
   443→                    <p className="text-xs text-muted-foreground">Order {r.order.refCode} ({r.order.paymentStatus}) · {r.reason} · raised {timeAgo(r.createdAt)}</p>
   444→                  </div>
   445→                  <div className="flex gap-2">
   446→                    <Button size="sm" disabled={busy === `r-${r.id}`} onClick={() => {
   447→                      const providerRef = window.prompt('Provider transaction reference (e.g. M-Pesa receipt):')
   448→                      if (providerRef) act(`r-${r.id}`, { action: 'process-refund', refundId: r.id, decision: 'PROCESSED', providerRef })
   449→                    }}>
   450→                      Mark processed
   451→                    </Button>
   452→                    <Button size="sm" variant="ghost" className="text-destructive" disabled={busy === `rr-${r.id}`}
   453→                      onClick={() => act(`rr-${r.id}`, { action: 'process-refund', refundId: r.id, decision: 'REJECTED', note: 'Rejected from admin console' })}>
   454→                      Reject
   455→                    </Button>
   456→                  </div>
   457→                </div>
   458→              ))}
   459→            </CardContent>
   460→          </Card>
   461→        </TabsContent>
   462→      </Tabs>
   463→
   464→      {/* Verify supplier dialog */}
   465→      <Dialog open={Boolean(verifyDialog)} onOpenChange={(o) => !o && setVerifyDialog(null)}>
   466→        <DialogContent>
   467→          <DialogHeader>
   468→            <DialogTitle>Verify {verifyDialog?.name}</DialogTitle>
   469→            <DialogDescription>
   470→              Confirm that registration documents and business-ownership checks are complete. Verified status is public and revocable; it does not certify product authenticity.
   471→            </DialogDescription>
   472→          </DialogHeader>
   473→          <div className="space-y-2">
   474→            <Label htmlFor="verify-note">Review note (audit-logged)</Label>
   475→            <Textarea id="verify-note" value={verifyNote} onChange={(e) => setVerifyNote(e.target.value)} placeholder="Documents checked: reg certificate, KRA PIN, premises…" />
   476→          </div>
   477→          <DialogFooter>
   478→            <Button variant="outline" onClick={() => setVerifyDialog(null)}>Cancel</Button>
   479→            <Button disabled={verifyNote.trim().length < 5 || busy === 'verify'} onClick={() => {
   480→              act('verify', { action: 'verify-supplier', orgId: verifyDialog!.id, note: verifyNote }, () => setVerifyDialog(null))
   481→            }}>
   482→              {busy === 'verify' ? <Loader2 className="mr-1 h-4 w-4 animate-spin" /> : <ShieldCheck className="mr-1 h-4 w-4" />} Confirm verification
   483→            </Button>
   484→          </DialogFooter>
   485→        </DialogContent>
   486→      </Dialog>
   487→
   488→      {/* Resolve dispute dialog */}
   489→      <Dialog open={Boolean(disputeDialog)} onOpenChange={(o) => !o && setDisputeDialog(null)}>
   490→        <DialogContent className="max-w-lg">
   491→          <DialogHeader>
   492→            <DialogTitle>Resolve {disputeDialog?.refCode}</DialogTitle>
   493→            <DialogDescription>All order records, payment confirmations, delivery confirmations, photos and messages are preserved as evidence. Completed transaction records can never be edited by either party.</DialogDescription>
   494→          </DialogHeader>
   495→          <div className="space-y-3">
   496→            <div className="grid grid-cols-2 gap-2">
   497→              {(['REFUND_FULL', 'REFUND_PARTIAL', 'REPLACEMENT', 'REJECTED'] as const).map((r) => (
   498→                <Button key={r} size="sm" variant={disputeResolution === r ? 'default' : 'outline'} onClick={() => setDisputeResolution(r)}>
   499→                  {r.replaceAll('_', ' ')}
   500→                </Button>
   501→              ))}
   502→            </div>
   503→            {disputeResolution === 'REFUND_PARTIAL' && (
   504→              <div className="space-y-1">
   505→                <Label htmlFor="refund-amount">Partial refund amount (KES)</Label>
   506→                <Input id="refund-amount" inputMode="numeric" value={refundAmount} onChange={(e) => setRefundAmount(e.target.value)} placeholder="e.g. 15000" />
   507→              </div>
   508→            )}
   509→            <div className="space-y-1">
   510→              <Label htmlFor="dispute-note">Resolution note</Label>
   511→              <Textarea id="dispute-note" value={disputeNote} onChange={(e) => setDisputeNote(e.target.value)} placeholder="Findings and decision rationale (shared with both parties)" />
   512→            </div>
   513→          </div>
   514→          <DialogFooter>
   515→            <Button variant="outline" onClick={() => setDisputeDialog(null)}>Cancel</Button>
   516→            <Button disabled={disputeNote.trim().length < 5 || busy === 'resolve'} onClick={() => {
   517→              act('resolve', {
   518→                action: 'resolve-dispute',
   519→                disputeId: disputeDialog!.id,
   520→                resolution: disputeResolution,
   521→                note: disputeNote,
   522→                ...(disputeResolution === 'REFUND_PARTIAL' && refundAmount ? { refundAmountMinor: Math.round(Number(refundAmount) * 100) } : {}),
   523→              }, () => setDisputeDialog(null))
   524→            }}>
   525→              Record resolution
   526→            </Button>
   527→          </DialogFooter>
   528→        </DialogContent>
   529→      </Dialog>
   530→
   531→      {/* Reconcile payment dialog */}
   532→      <Dialog open={Boolean(reconcileDialog)} onOpenChange={(o) => !o && setReconcileDialog(null)}>
   533→        <DialogContent>
   534→          <DialogHeader>
   535→            <DialogTitle>Reconcile payment: {reconcileDialog?.refCode}</DialogTitle>
   536→            <DialogDescription>
   537→              Record the real M-Pesa/paybill transaction reference after checking your provider statement. This marks funds as PROTECTED (provider MANUAL_RECONCILED) and lets the supplier dispatch. Funds release only after the garage&apos;s inspection.
   538→            </DialogDescription>
   539→          </DialogHeader>
   540→          <div className="space-y-2">
   541→            <Label htmlFor="rec-ref">Provider reference</Label>
   542→            <Input id="rec-ref" value={reconcileRef} onChange={(e) => setReconcileRef(e.target.value)} placeholder="e.g. SBX12QTXYZ" />
   543→          </div>
   544→          <DialogFooter>
   545→            <Button variant="outline" onClick={() => setReconcileDialog(null)}>Cancel</Button>
   546→            <Button disabled={reconcileRef.trim().length < 3 || busy === 'rec'} onClick={() => {
   547→              act('rec', { action: 'reconcile-payment', orderId: reconcileDialog!.id, providerRef: reconcileRef.trim() }, () => setReconcileDialog(null))
   548→            }}>
   549→              Confirm protected
   550→            </Button>
   551→          </DialogFooter>
   552→        </DialogContent>
   553→      </Dialog>
   554→    </div>
   555→  )
