    21→const schema = z.discriminatedUnion('action', [
    22→  // STEP 2: supplier response
    23→  z.object({
    24→    action: z.literal('respond'),
    25→    decision: z.enum(['ACCEPT', 'REJECT', 'REVISE']),
    26→    note: z.string().max(500).optional().nullable(),
    27→    revision: z.object({
    28→      changes: z.array(z.object({
    29→        itemId: z.string().optional(),
    30→        productId: z.string(),
    31→        action: z.enum(['UPDATE', 'REMOVE', 'ADD']),
    32→        qty: z.number().int().min(1).max(10_000).optional(),
    33→        unitPriceMinor: z.number().int().min(100).optional(),
    34→      })).min(1),
    35→    }).optional(),
    36→  }),
    37→  // garage internal approval gate
    38→  z.object({ action: z.literal('approve'), note: z.string().max(300).optional() }),
    39→  // garage decision on supplier revision
    40→  z.object({ action: z.literal('revision-decision'), decision: z.enum(['APPROVED', 'DECLINED']), revisionId: z.string(), note: z.string().max(300).optional() }),
    41→  // garage cancels before dispatch
    42→  z.object({ action: z.literal('cancel'), reason: z.string().min(3).max(300) }),
    43→  // STEP 3: delivery arrangement
    44→  z.object({
    45→    action: z.literal('arrange'),
    46→    method: z.enum(['DIRECT', 'COLLECTION', 'MEETUP']),
    47→    supplierBranchId: z.string().optional(),
    48→    meetup: z.object({ lat: z.number(), lng: z.number(), address: z.string().min(4).max(200) }).optional(),
    49→    scheduledFor: z.string().optional(),
    50→    windowLabel: z.string().max(60).optional(),
    51→    feeMinor: z.number().int().min(0).max(1_000_000).default(0),
    52→    courierName: z.string().max(80).optional().nullable(),
    53→  }),
    54→  z.object({ action: z.literal('agree-location'), note: z.string().max(300).optional() }),
    55→  // STEP 5: supplier dispatch
    56→  z.object({
    57→    action: z.literal('dispatch'),
    58→    supplierBranchId: z.string().optional(),
    59→    courierName: z.string().max(80).optional().nullable(),
    60→    driverName: z.string().max(80).optional().nullable(),
    61→    driverPhone: z.string().max(20).optional().nullable(),
    62→    vehicleReg: z.string().max(20).optional().nullable(),
    63→    deliveryRef: z.string().max(60).optional().nullable(),
    64→    expectedHours: z.number().int().min(1).max(720).optional(),
    65→    evidence: z.array(z.string().max(300)).max(6).default([]),
    66→  }),
    67→  // STEP 6: handover confirmation (supplier marks handed over; garage verifies OTP/QR)
    68→  z.object({ action: z.literal('handover'), otp: z.string().length(6).optional(), qrToken: z.string().max(20).optional(), captureLocation: z.boolean().default(false), location: z.object({ lat: z.number(), lng: z.number() }).optional(), photos: z.array(z.string().max(300)).max(6).default([]) }),
    69→  z.object({ action: z.literal('handover-confirm') }),
    70→  // STEP 7: inspection
    71→  z.object({
    72→    action: z.literal('inspect'),
    73→    result: z.enum(['ACCEPT', 'PARTIAL', 'ISSUES']),
    74→    items: z.array(z.object({ itemId: z.string(), qtyAccepted: z.number().int().min(0), note: z.string().max(300).optional() })).default([]),
    75→    note: z.string().max(1000).optional(),
    76→    photos: z.array(z.string().max(300)).max(6).default([]),
    77→  }),
    78→  // STEP 8: payment release request (executed against protected funds)
    79→  z.object({ action: z.literal('release'), note: z.string().max(300).optional() }),
    80→  // STEP 9: dispute
    81→  z.object({
    82→    action: z.literal('dispute'),
    83→    reason: z.enum(['MISSING_ITEMS', 'DAMAGED', 'WRONG_ITEM', 'COUNTERFEIT', 'NOT_RECEIVED', 'OTHER']),
    84→    details: z.string().min(10).max(2000),
    85→    evidence: z.array(z.object({ url: z.string().max(300), name: z.string().max(120) })).max(6).default([]),
    86→  }),
    87→  z.object({ action: z.literal('request-return'), reason: z.string().min(5).max(300), items: z.array(z.object({ productId: z.string(), qty: z.number().int().min(1) })).min(1) }),
    88→  z.object({ action: z.literal('warranty'), productId: z.string(), issue: z.string().min(10).max(1000) }),
    89→])

    91→/** POST /api/supply/orders/[id]/action: every state transition of the
    92→ * protected supplier→garage order workflow. Permissions and allowed source
    93→ * states are enforced here for BOTH sides; the UI merely reflects them. */
    94→export const POST = route(async (req: NextRequest, ctx: Ctx) => {
    95→  const { id } = await ctx.params
    96→  const user = await requireUser()
    97→  const body = await parseBody(req, schema)
    98→
    99→  const order = await db.purchaseOrder.findUnique({
   100→    where: { id },
   101→    include: {
   102→      garage: { select: { id: true, name: true } },
   103→      supplier: { select: { id: true, name: true } },
   104→      items: { include: { product: true } }, delivery: true, payment: true,
   105→    },
   106→  })
   107→  if (!order) return fail(404, 'NOT_FOUND', 'Order not found.')
   108→
   109→  // resolve side + membership
   110→  const garageMember = await getMembership(order.garageOrgId, user.id)
   111→  const supplierMember = await getMembership(order.supplierOrgId, user.id)
   112→  const isStaff = user.roles.some((r) => ['FINANCE', 'ADMIN', 'FOUNDER'].includes(r))
   113→  if (!garageMember && !supplierMember && !isStaff) return fail(403, 'FORBIDDEN', 'You are not authorised on this order.')
   114→  const side: 'GARAGE' | 'SUPPLIER' = supplierMember ? 'SUPPLIER' : 'GARAGE'
   115→  // Platform FINANCE/ADMIN/FOUNDER with no membership on either org are legitimate
   116→  // reconciliation actors: they have no OrgMember row, so there is no role to
   117→  // resolve here. roleCan denies an empty role (safe); the isStaff bypass below
   118→  // authorises the FINANCE tools. The previous code dereferenced a null
   119→  // membership and 500'd on every staff action taken outside an org.
   120→  const memberRole: string = side === 'SUPPLIER' ? (supplierMember?.role ?? '') : (garageMember?.role ?? '')
   121→
   122→  // permission matrix check per action
   123→  const actionPerm: Record<string, Parameters<typeof roleCan>[2]> = {
   124→    respond: 'order.respond', approve: 'order.approve', 'revision-decision': 'order.approve',
   125→    cancel: 'order.create', arrange: 'delivery.arrange', 'agree-location': 'delivery.arrange',
   126→    dispatch: 'order.respond', handover: 'order.read', 'handover-confirm': 'order.respond',
   127→    inspect: 'order.approve', release: 'payment.initiate', dispute: 'dispute.open',
   128→    'request-return': 'order.create', warranty: 'order.create',
   129→  }
   130→  const needed = actionPerm[body.action]
   131→  if (needed && !roleCan(side, memberRole, needed) && !isStaff) {
   132→    return fail(403, 'PERMISSION_DENIED', `Your role (${memberRole}) cannot perform "${body.action}" on this order.`)
   133→  }
   134→
   135→  const supplierCtx = { orgId: order.supplierOrgId, side: 'SUPPLIER' as const }
   136→  const garageCtx = { orgId: order.garageOrgId, side: 'GARAGE' as const }
   137→  const actorCtx = side === 'SUPPLIER' ? supplierCtx : garageCtx
   138→
   139→  switch (body.action) {
   140→    // ─────────── STEP 2: supplier response ───────────
   141→    case 'respond': {
   142→      if (side !== 'SUPPLIER') return fail(403, 'PERMISSION_DENIED', 'Only the supplier responds to orders.')
   143→      if (order.status !== 'SUBMITTED') return fail(409, 'INVALID_STATE', `Order is ${order.status}; expected SUBMITTED.`)
   144→      if (body.decision === 'ACCEPT') {
   145→        await db.purchaseOrder.update({ where: { id }, data: { status: 'ACCEPTED', supplierNote: body.note ?? null } })
   146→        await poEvent(id, { type: 'ACCEPTED', actorId: user.id, actorName: user.name, actorSide: 'SUPPLIER', note: body.note ?? 'Supplier accepted the order.' })
   147→        await notifyGarageMembers(order.garageOrgId, id, order.refCode, `Supplier accepted order ${order.refCode}. Arrange delivery and payment.`, 'PO_ACCEPTED')
   148→      } else if (body.decision === 'REJECT') {
   149→        await db.purchaseOrder.update({ where: { id }, data: { status: 'REJECTED', supplierNote: body.note ?? 'Rejected by supplier' } })
   150→        for (const item of order.items) {
   151→          if (item.fulfilBranchId) await releaseReservation(item.productId, item.fulfilBranchId, item.qty, { type: 'PURCHASE_ORDER', id, actorName: user.name })
   152→        }
   153→        await poEvent(id, { type: 'CANCELLED', actorId: user.id, actorName: user.name, actorSide: 'SUPPLIER', note: body.note ?? 'Order rejected by supplier.' })
   154→        await notifyGarageMembers(order.garageOrgId, id, order.refCode, `Supplier rejected order ${order.refCode}. ${body.note ?? ''}`, 'PO_REJECTED')
   155→      } else {
   156→        if (!body.revision) return fail(400, 'VALIDATION_ERROR', 'A revision proposal is required.')
   157→        // recompute totals from proposed changes
   158→        const proposed = new Map<string, { qty: number; unit: number }>()
   159→        for (const it of order.items) proposed.set(it.productId, { qty: it.qty, unit: it.unitPriceMinor })
   160→        for (const ch of body.revision.changes) {
   161→          if (ch.action === 'REMOVE') proposed.delete(ch.productId)
   162→          else proposed.set(ch.productId, { qty: ch.qty ?? proposed.get(ch.productId)?.qty ?? 1, unit: ch.unitPriceMinor ?? proposed.get(ch.productId)?.unit ?? 0 })
   163→        }
   164→        const newSubtotal = [...proposed.values()].reduce((s, v) => s + v.qty * v.unit, 0)
   165→        const revision = await db.pORevision.create({
   166→          data: {
   167→            orderId: id,
   168→            proposedBy: 'SUPPLIER',
   169→            changesJson: JSON.stringify(body.revision.changes),
   170→            newTotalMinor: newSubtotal,
   171→            note: body.note ?? null,
   172→          },
   173→        })
   174→        await db.purchaseOrder.update({ where: { id }, data: { status: 'REVISED', supplierNote: body.note ?? null, revisedBySupplier: true } })
   175→        await poEvent(id, { type: 'REVISED', actorId: user.id, actorName: user.name, actorSide: 'SUPPLIER', note: body.note ?? `Supplier proposed changes (new subtotal ${newSubtotal / 100} KES).` })
   176→        await notifyGarageMembers(order.garageOrgId, id, order.refCode, `Supplier proposed changes on ${order.refCode}. Your approval is required before the order is final.`, 'PO_REVISED')
   177→        return ok({ revisionId: revision.id, message: 'Revision proposed. The garage must approve before the order is final.' })
   178→      }
   179→      return ok({ message: 'Response recorded.' })
   180→    }
   181→
   182→    // ─────────── garage internal approval ───────────
   183→    case 'approve': {
   184→      if (side !== 'GARAGE') return fail(403, 'PERMISSION_DENIED', 'Only the garage approves internally.')
   185→      if (order.status !== 'AWAITING_APPROVAL') return fail(409, 'INVALID_STATE', `Order is ${order.status}; expected AWAITING_APPROVAL.`)
   186→      await db.purchaseOrder.update({ where: { id }, data: { status: 'SUBMITTED' } })
   187→      await poEvent(id, { type: 'APPROVED', actorId: user.id, actorName: user.name, actorSide: 'GARAGE', note: body.note ?? 'Approved by management: sent to supplier.' })
   188→      await notifySupplierMembers(order.supplierOrgId, order.refCode, `${order.garage.name} approved order ${order.refCode}. It now awaits your response.`, 'PO_NEW')
   189→      return ok({ message: 'Order approved and sent to the supplier.' })
   190→    }
   191→
   192→    case 'revision-decision': {
   193→      if (side !== 'GARAGE') return fail(403, 'PERMISSION_DENIED', 'Only the garage decides on revisions.')
   194→      if (order.status !== 'REVISED') return fail(409, 'INVALID_STATE', `Order is ${order.status}; expected REVISED.`)
   195→      const revision = await db.pORevision.findUnique({ where: { id: body.revisionId } })
   196→      if (!revision || revision.orderId !== id) return fail(404, 'NOT_FOUND', 'Revision not found.')
   197→      if (revision.status !== 'PENDING') return fail(409, 'INVALID_STATE', 'Revision already decided.')
   198→
   199→      if (body.decision === 'APPROVED') {
   200→        // apply changes to items & totals; adjust reservations
   201→        const changes = jsonParse<{ action: string; productId: string; itemId?: string; qty?: number; unitPriceMinor?: number }[]>(revision.changesJson, [])
   202→        for (const ch of changes) {
   203→          const item = order.items.find((i) => i.id === ch.itemId || i.productId === ch.productId)
   204→          if (ch.action === 'REMOVE' && item) {
   205→            if (item.fulfilBranchId) await releaseReservation(item.productId, item.fulfilBranchId, item.qty, { type: 'PURCHASE_ORDER', id, actorName: user.name })
   206→            await db.pOItem.delete({ where: { id: item.id } })
   207→          } else if (item && ch.qty && ch.unitPriceMinor) {
   208→            await db.pOItem.update({ where: { id: item.id }, data: { qty: ch.qty, unitPriceMinor: ch.unitPriceMinor, lineTotalMinor: ch.qty * ch.unitPriceMinor } })
   209→          } else if (ch.action === 'ADD' && ch.qty && ch.unitPriceMinor) {
   210→            const product = await db.product.findFirst({ where: { id: ch.productId, supplierOrgId: order.supplierOrgId, status: 'PUBLISHED' } })
   211→            if (!product) return fail(400, 'INVALID_PRODUCT', 'Proposed product is unavailable.')
   212→            await db.pOItem.create({ data: { orderId: id, productId: ch.productId, qty: ch.qty, unitPriceMinor: ch.unitPriceMinor, lineTotalMinor: ch.qty * ch.unitPriceMinor } })
   213→          }
   214→        }
   215→        const items = await db.pOItem.findMany({ where: { orderId: id } })
   216→        const subtotal = items.reduce((s, i) => s + i.lineTotalMinor, 0)
   217→        await db.purchaseOrder.update({ where: { id }, data: { status: 'ACCEPTED', subtotalMinor: subtotal, totalMinor: subtotal + order.taxMinor, garageApprovedRevision: true } })
   218→        await db.pORevision.update({ where: { id: revision.id }, data: { status: 'APPROVED' } })
   219→        await poEvent(id, { type: 'REVISION_DECISION', actorId: user.id, actorName: user.name, actorSide: 'GARAGE', note: body.note ?? 'Garage approved the revised quotation.' })
   220→        await notifySupplierMembers(order.supplierOrgId, order.refCode, `${order.garage.name} approved your revision on ${order.refCode}.`, 'PO_ACCEPTED')
   221→        return ok({ message: 'Revision approved: order finalised at the revised terms.' })
   222→      } else {
   223→        await db.pORevision.update({ where: { id: revision.id }, data: { status: 'DECLINED' } })
   224→        await db.purchaseOrder.update({ where: { id }, data: { status: 'SUBMITTED' } })
   225→        await poEvent(id, { type: 'REVISION_DECISION', actorId: user.id, actorName: user.name, actorSide: 'GARAGE', note: body.note ?? 'Garage declined the revision: original terms stand.' })
   226→        await notifySupplierMembers(order.supplierOrgId, order.refCode, `${order.garage.name} declined your revision on ${order.refCode}. Original terms stand.`, 'PO_REVISED')
   227→        return ok({ message: 'Revision declined. The supplier may respond again on the original terms.' })
   228→      }
   229→    }
   230→
   231→    // ─────────── cancel ───────────
   232→    case 'cancel': {
   233→      if (side !== 'GARAGE') return fail(403, 'PERMISSION_DENIED', 'Only the garage cancels orders.')
   234→      if (!['SUBMITTED', 'AWAITING_APPROVAL', 'ACCEPTED', 'REVISED', 'CONFIRMED'].includes(order.status)) {
   235→        return fail(409, 'INVALID_STATE', `Order cannot be cancelled at ${order.status}.`)
   236→      }
   237→      if (order.paymentStatus === 'PROTECTED') return fail(409, 'PAYMENT_PROTECTED', 'Payment is protected: open a dispute instead.')
   238→      for (const item of order.items) {
   239→        if (item.fulfilBranchId) await releaseReservation(item.productId, item.fulfilBranchId, item.qty, { type: 'PURCHASE_ORDER', id, actorName: user.name })
   240→      }
   241→      await db.purchaseOrder.update({ where: { id }, data: { status: 'CANCELLED', cancelledReason: body.reason } })
   242→      await poEvent(id, { type: 'CANCELLED', actorId: user.id, actorName: user.name, actorSide: 'GARAGE', note: body.reason })
   243→      await notifySupplierMembers(order.supplierOrgId, order.refCode, `${order.garage.name} cancelled order ${order.refCode}. Reason: ${body.reason}`, 'PO_CANCELLED')
   244→      await audit({ actorId: user.id, actorName: user.name, action: 'PO_CANCELLED', targetType: 'PURCHASE_ORDER', targetId: id, metadata: { reason: body.reason } })
   245→      return ok({ message: 'Order cancelled and stock reservations released.' })
   246→    }
   247→
   248→    // ─────────── STEP 3: delivery arrangement ───────────
   249→    case 'arrange': {
   250→      if (order.status !== 'ACCEPTED' && order.status !== 'CONFIRMED' && order.status !== 'REVISED') {
   251→        return fail(409, 'INVALID_STATE', `Delivery is arranged after acceptance (current: ${order.status}).`)
   252→      }
   253→      const method = body.method
   254→      // Money integrity: once payment is moving (PROCESSING/PROTECTED) the
   255→      // delivery fee can no longer change the escrowed total — a late fee
   256→      // bump must not create payout value above what was actually collected.
   257→      const feeFrozen = !!order.payment && ['PROCESSING', 'PROTECTED'].includes(order.payment.status)
   258→      const deliveryFee = feeFrozen ? (order.delivery?.feeMinor ?? order.deliveryFeeMinor) : body.feeMinor
   259→      let supplierBranchId: string | null = order.delivery?.supplierBranchId ?? null
   260→      let meetupJson: string | null = order.delivery?.meetupJson ?? null
   261→      let agreedByGarage = order.delivery?.meetupAgreedByGarage ?? false
   262→      let agreedBySupplier = order.delivery?.meetupAgreedBySupplier ?? false
   263→      let status = 'ARRANGING'
   264→      let message = ''
   265→      let changed = false
   266→
   267→      if (method === 'DIRECT') {
   268→        if (!order.destinationBranchId) return fail(400, 'NO_DESTINATION', 'Choose the garage branch receiving this delivery.')
   269→        status = 'AGREED'
   270→        message = 'Direct delivery to your branch arranged.'
   271→      } else if (method === 'COLLECTION') {
   272→        if (!body.supplierBranchId) return fail(400, 'NO_BRANCH', 'Choose the supplier warehouse for collection.')
   273→        const wb = await db.orgBranch.findFirst({ where: { id: body.supplierBranchId, orgId: order.supplierOrgId, status: 'ACTIVE' } })
   274→        if (!wb) return fail(400, 'NO_BRANCH', 'Warehouse does not belong to the supplier.')
   275→        supplierBranchId = wb.id
   276→        status = 'AGREED'
   277→        message = 'Collection from the supplier warehouse arranged.'
   278→      } else {
   279→        // MEETUP: proposed location requires BOTH parties' agreement; a new
   280→        // proposal resets the counterparty's consent.
   281→        if (!body.meetup) return fail(400, 'NO_LOCATION', 'Propose a handover location on the map.')
   282→        changed = !meetupJson || jsonParse<{ address?: string }>(meetupJson, {}).address !== body.meetup.address
   283→        if (changed) {
   284→          meetupJson = JSON.stringify(body.meetup)
   285→          if (side === 'GARAGE') { agreedByGarage = true; agreedBySupplier = false }
   286→          else { agreedBySupplier = true; agreedByGarage = false }
   287→        }
   288→        message = agreedByGarage && agreedBySupplier ? 'Handover location agreed by both parties.' : 'Location proposed: awaiting the other party\'s agreement.'
   289→        status = agreedByGarage && agreedBySupplier ? 'AGREED' : 'ARRANGING'
   290→      }
   291→
   292→      const scheduledFor = body.scheduledFor ? new Date(body.scheduledFor) : null
   293→      const delivery = await db.delivery.upsert({
   294→        where: { orderId: id },
   295→        create: {
   296→          refCode: refCode('DLV', 6),
   297→          orderId: id, method, status, supplierBranchId, meetupJson,
   298→          meetupAgreedByGarage: agreedByGarage, meetupAgreedBySupplier: agreedBySupplier,
   299→          scheduledFor, windowLabel: body.windowLabel ?? null, feeMinor: deliveryFee,
   300→          events: { create: { type: 'ARRANGED', actorId: user.id, actorName: user.name, actorSide: side, note: `${method} delivery arranged.` } },
   301→        },
   302→        update: {
   303→          method, status, supplierBranchId, meetupJson,
   304→          meetupAgreedByGarage: agreedByGarage, meetupAgreedBySupplier: agreedBySupplier,
   305→          scheduledFor, windowLabel: body.windowLabel !== undefined ? (body.windowLabel ?? null) : (order.delivery?.windowLabel ?? null),
   306→          feeMinor: deliveryFee,
   307→        },
   308→      })
   309→      await db.purchaseOrder.update({
   310→        where: { id },
   311→        data: {
   312→          deliveryMethod: method,
   313→          deliveryFeeMinor: deliveryFee,
   314→          totalMinor: order.subtotalMinor + order.taxMinor + deliveryFee,
   315→          status: status === 'AGREED' ? 'CONFIRMED' : order.status,
   316→        },
   317→      })
   318→      await deliveryEvent(delivery.id, { type: changed ? 'LOCATION_PROPOSED' : 'ARRANGED', actorId: user.id, actorName: user.name, actorSide: side, note: `${method}: ${message}` })
   319→      await poEvent(id, { type: 'NOTE', actorId: user.id, actorName: user.name, actorSide: side, note: `Delivery arrangement: ${message}` })
   320→      await notifyOtherSide(actorCtx, order, `${order.refCode}: ${message}`, 'PO_DELIVERY')
   321→      return ok({ delivery, message })
   322→    }
   323→
   324→    case 'agree-location': {
   325→      const delivery = order.delivery
   326→      if (!delivery || delivery.method !== 'MEETUP') return fail(409, 'INVALID_STATE', 'No meetup location to agree on.')
   327→      if (delivery.status === 'AGREED') return ok({ message: 'Location already agreed.' })
   328→      const updated = await db.delivery.update({
   329→        where: { id: delivery.id },
   330→        data: side === 'GARAGE' ? { meetupAgreedByGarage: true } : { meetupAgreedBySupplier: true },
   331→      })
   332→      const both = updated.meetupAgreedByGarage && updated.meetupAgreedBySupplier
   333→      if (both) {
   334→        await db.delivery.update({ where: { id: delivery.id }, data: { status: 'AGREED' } })
   335→        await db.purchaseOrder.update({ where: { id }, data: { status: 'CONFIRMED' } })
   336→      }
   337→      await deliveryEvent(delivery.id, { type: 'LOCATION_AGREED', actorId: user.id, actorName: user.name, actorSide: side, note: body.note ?? 'Agreed to the handover location.' })
   338→      await notifyOtherSide(actorCtx, order, `${order.refCode}: ${side === 'GARAGE' ? 'Garage' : 'Supplier'} agreed to the handover location${both ? '. Delivery is now locked. Location changes require a new mutual agreement.' : ''}`, 'PO_DELIVERY')
   339→      return ok({ agreed: both, message: both ? 'Both parties agreed. Location is locked.' : 'Your agreement is recorded: awaiting the other party.' })
   340→    }
   341→
   342→    // ─────────── STEP 5: dispatch ───────────
   343→    case 'dispatch': {
   344→      if (side !== 'SUPPLIER') return fail(403, 'PERMISSION_DENIED', 'Only the supplier dispatches.')
   345→      if (order.status !== 'CONFIRMED') return fail(409, 'INVALID_STATE', `Order is ${order.status}; dispatch requires CONFIRMED.`)
   346→      if (order.paymentStatus !== 'PROTECTED') {
   347→        return fail(409, 'PAYMENT_REQUIRED', 'Payment must be protected before dispatch. Funds are only released after the garage accepts the delivery.')
   348→      }
   349→      const delivery = order.delivery
   350→      if (!delivery || delivery.status !== 'AGREED') return fail(409, 'INVALID_STATE', 'Arrange and agree the delivery first.')
   351→      if (body.supplierBranchId) {
   352→        const wb = await db.orgBranch.findFirst({ where: { id: body.supplierBranchId, orgId: order.supplierOrgId, status: 'ACTIVE' } })
   353→        if (!wb) return fail(400, 'NO_BRANCH', 'Warehouse does not belong to your business.')
   354→        await db.delivery.update({ where: { id: delivery.id }, data: { supplierBranchId: wb.id } })
   355→      }
   356→      const otp = sixDigitCode()
   357→      const otpHash = await bcrypt.hash(otp, 10)
   358→      const expectedAt = new Date(Date.now() + (body.expectedHours ?? 24) * 3600 * 1000)
   359→      await db.delivery.update({
   360→        where: { id: delivery.id },
   361→        data: {
   362→          status: 'DISPATCHED',
   363→          courierName: body.courierName ?? delivery.courierName ?? 'Supplier-owned delivery',
   364→          driverName: body.driverName ?? null,
   365→          driverPhone: body.driverPhone ?? null,
   366→          vehicleReg: body.vehicleReg ?? null,
   367→          deliveryRef: body.deliveryRef ?? `DSP-${order.refCode}`,
   368→          dispatchEvidence: body.evidence.length ? JSON.stringify(body.evidence) : delivery.dispatchEvidence,
   369→          dispatchedAt: new Date(),
   370→          expectedAt,
   371→          otpHash,
   372→          otpHint: otp.slice(-2),
   373→          qrToken: shareId(14),
   374→        },
   375→      })
   376→      await db.purchaseOrder.update({ where: { id }, data: { status: 'DISPATCHED' } })
   377→      await deliveryEvent(delivery.id, { type: 'DISPATCHED', actorId: user.id, actorName: user.name, actorSide: 'SUPPLIER', note: `Dispatched${body.driverName ? `: driver ${body.driverName}` : ''}. Expected ${expectedAt.toISOString()}` })
   378→      await poEvent(id, { type: 'DISPATCH', actorId: user.id, actorName: user.name, actorSide: 'SUPPLIER', note: body.deliveryRef ? `Delivery reference ${body.deliveryRef}` : 'Order dispatched.' })
   379→      // OTP delivered through in-app notification (SMS channel is recorded but not configured)
   380→      await notifyGarageMembers(order.garageOrgId, id, order.refCode, `Order ${order.refCode} dispatched. Handover verification code: ${otp} (valid at delivery). Do not share before handover.`, 'PO_DISPATCHED')
   381→      await audit({ actorId: user.id, actorName: user.name, action: 'PO_DISPATCHED', targetType: 'PURCHASE_ORDER', targetId: id })
   382→      return ok({ message: 'Dispatch recorded. The garage received the handover verification code.' })
   383→    }
   384→
   385→    // ─────────── STEP 6: handover ───────────
   386→    case 'handover': {
   387→      const delivery = order.delivery
   388→      if (!delivery || !['DISPATCHED', 'IN_TRANSIT'].includes(delivery.status)) {
   389→        return fail(409, 'INVALID_STATE', 'Handover verification is available after dispatch.')
   390→      }
   391→      if (side !== 'GARAGE') return fail(403, 'PERMISSION_DENIED', 'The receiving party verifies the handover.')
   392→      // Brute-force guard: the 6-digit handover code is the last barrier before
   393→      // inspection/payment. Five wrong attempts per 10 minutes per order.
   394→      const rl = rateLimit(`po-handover:${id}:${user.id}`, 5, 10 * 60 * 1000)
   395→      if (!rl.allowed) return fail(429, 'RATE_LIMITED', `Too many verification attempts. Try again in ${Math.ceil(rl.retryAfterSec / 60)} minutes.`)
   396→      let verified = false
   397→      if (body.otp && delivery.otpHash) verified = await bcrypt.compare(body.otp, delivery.otpHash)
   398→      else if (body.qrToken && delivery.qrToken) verified = body.qrToken === delivery.qrToken
   399→      if (!verified) return fail(403, 'BAD_CODE', 'That verification code does not match this delivery.')
   400→
   401→      const handover = {
   402→        ...(jsonParse<Record<string, unknown>>(delivery.handoverJson ?? '{}', {})),
   403→        garageConfirmedAt: new Date().toISOString(),
   404→        captureLocation: body.captureLocation,
   405→        location: body.captureLocation && body.location ? body.location : undefined,
   406→        photos: body.photos,
   407→      }
   408→      await db.delivery.update({
   409→        where: { id: delivery.id },
   410→        data: { garageConfirmAt: new Date(), handoverJson: JSON.stringify(handover) },
   411→      })
   412→      await finaliseHandoverIfBoth(id)
   413→      await deliveryEvent(delivery.id, { type: 'DELIVERED', actorId: user.id, actorName: user.name, actorSide: 'GARAGE', note: `Handover verified by garage${body.captureLocation && body.location ? ' (location shared)' : ''}.` })
   414→      return ok({ message: 'Handover verified. Record the inspection result to release payment.' })
   415→    }
   416→
   417→    case 'handover-confirm': {
   418→      if (side !== 'SUPPLIER') return fail(403, 'PERMISSION_DENIED', 'Only the supplier confirms handover.')
   419→      const delivery = order.delivery
   420→      if (!delivery || !['DISPATCHED', 'IN_TRANSIT'].includes(delivery.status)) {
   421→        return fail(409, 'INVALID_STATE', 'Handover is available after dispatch.')
   422→      }
   423→      await db.delivery.update({ where: { id: delivery.id }, data: { supplierConfirmAt: new Date() } })
   424→      await finaliseHandoverIfBoth(id)
   425→      await deliveryEvent(delivery.id, { type: 'DELIVERED', actorId: user.id, actorName: user.name, actorSide: 'SUPPLIER', note: 'Supplier confirmed handover.' })
   426→      return ok({ message: 'Handover confirmation recorded.' })
   427→    }
   428→
   429→    // ─────────── STEP 7: inspection ───────────
   430→    case 'inspect': {
   431→      if (side !== 'GARAGE') return fail(403, 'PERMISSION_DENIED', 'Only the receiving garage inspects.')
   432→      const delivery = order.delivery
   433→      if (!delivery || delivery.status !== 'DELIVERED') return fail(409, 'INVALID_STATE', 'Inspection opens after verified handover.')
   434→      const items = await db.pOItem.findMany({ where: { orderId: id } })
   435→      const acceptedMap = new Map(body.items.map((i) => [i.itemId, i]))
   436→
   437→      for (const item of items) {
   438→        const reported = acceptedMap.get(item.id)
   439→        const qtyAccepted = body.result === 'ACCEPT' ? item.qty : Math.min(reported?.qtyAccepted ?? 0, item.qty)
   440→        await db.pOItem.update({
   441→          where: { id: item.id },
   442→          data: { qtyAccepted, qtyDelivered: body.result === 'ACCEPT' ? item.qty : qtyAccepted, note: reported?.note ?? (body.result === 'ACCEPT' ? null : 'Shortfall/damage reported at inspection') },
   443→        })
   444→        if (qtyAccepted > 0 && item.fulfilBranchId) {
   445→          await fulfillStock(item.productId, item.fulfilBranchId, qtyAccepted, { type: 'PURCHASE_ORDER', id, actorName: user.name })
   446→        } else if (qtyAccepted < item.qty && item.fulfilBranchId) {
   447→          await releaseReservation(item.productId, item.fulfilBranchId, item.qty - qtyAccepted, { type: 'PURCHASE_ORDER', id, actorName: user.name })
   448→        }
   449→      }
   450→      const freshItems = await db.pOItem.findMany({ where: { orderId: id } })
   451→      const acceptedSubtotal = freshItems.reduce((s, i) => s + (i.qtyAccepted ?? 0) * i.unitPriceMinor, 0)
   452→      const acceptedTotal = acceptedSubtotal + order.taxMinor + order.deliveryFeeMinor
   453→      const inspectionStatus = body.result === 'ACCEPT' ? 'ACCEPTED' : body.result === 'PARTIAL' ? 'PARTIAL' : 'ISSUES'
   454→      await db.purchaseOrder.update({
   455→        where: { id },
   456→        data: { status: 'INSPECTED', inspectionStatus, acceptedTotalMinor: acceptedTotal },
   457→      })
   458→      // refund due for unaccepted goods
   459→      if (acceptedTotal < order.totalMinor) {
   460→        const existing = await db.refund.findFirst({ where: { orderId: id, status: { in: ['REQUESTED', 'APPROVED'] } } })
   461→        if (!existing) {
   462→          await db.refund.create({
   463→            data: {
   464→              refCode: refCode('RFD', 6),
   465→              orderId: id,
   466→              paymentAuthId: order.payment?.id ?? null,
   467→              amountMinor: order.totalMinor - acceptedTotal,
   468→              kind: 'PARTIAL',
   469→              reason: 'Partial acceptance at inspection',
   470→              requestedById: user.id,
   471→            },
   472→          })
   473→        }
   474→      }
   475→      await poEvent(id, { type: 'INSPECTED', actorId: user.id, actorName: user.name, actorSide: 'GARAGE', note: `${body.note ?? `Inspection result: ${inspectionStatus}.`} Accepted value: ${(acceptedTotal / 100).toLocaleString()} KES. Photos: ${body.photos.length}` })
   476→      await notifySupplierMembers(order.supplierOrgId, order.refCode, `Garage inspected ${order.refCode}: ${inspectionStatus}. Accepted ${(acceptedTotal / 100).toLocaleString()} KES.`, 'PO_INSPECTED')
   477→      return ok({ acceptedTotalMinor: acceptedTotal, inspectionStatus, message: 'Inspection recorded. Payment release can now be requested for the accepted amount.' })
   478→    }
   479→
   480→    // ─────────── STEP 8: payment release ───────────
   481→    case 'release': {
   482→      if (!['GARAGE'].includes(side) && !isStaff) return fail(403, 'PERMISSION_DENIED', 'The receiving garage requests the release.')
   483→      if (order.status !== 'INSPECTED') return fail(409, 'INVALID_STATE', 'Release requires a recorded inspection.')
   484→      const payment = order.payment
   485→      if (!payment || payment.status !== 'PROTECTED') {
   486→        return fail(409, 'NOT_PROTECTED', 'Funds are not in a protected state. Release follows the payment provider\'s protection confirmation only.')
   487→      }
   488→      const accepted = order.acceptedTotalMinor ?? order.totalMinor
   489→      const fees = await getFees()
   490→      const commission = computeCommission(order.subtotalMinor, fees.supplyCommissionPct, fees.minCommissionMinor)
   491→      // Financial integrity: release can never exceed the funds actually
   492→      // protected in escrow (amount protected minus anything already refunded).
   493→      const escrowAvailable = Math.max(0, payment.amountMinor - (payment.refundedAmountMinor ?? 0))
   494→      if (accepted > escrowAvailable) {
   495→        return fail(409, 'RELEASE_EXCEEDS_ESCROW', `Inspection accepted ${(accepted / 100).toLocaleString()} KES but only ${(escrowAvailable / 100).toLocaleString()} KES is protected. Re-run inspection or open a dispute; release cannot exceed protected funds.`)
   496→      }
   497→      await db.paymentAuthorization.update({
   498→        where: { id: payment.id },
   499→        data: {
   500→          status: accepted < order.totalMinor ? 'PARTIALLY_RELEASED' : 'RELEASED',
   501→          releasedAmountMinor: accepted,
   502→          releasedAt: new Date(),
   503→        },
   504→      })
   505→      await db.purchaseOrder.update({
   506→        where: { id },
   507→        data: {
   508→          paymentStatus: accepted < order.totalMinor ? 'PARTIALLY_RELEASED' : 'RELEASED',
   509→          platformFeeMinor: commission,
   510→          commissionRatePct: fees.supplyCommissionPct,
   511→          status: 'COMPLETED',
   512→        },
   513→      })
   514→      if (order.invoiceId) {
   515→        await db.invoice.update({ where: { id: order.invoiceId }, data: { status: 'PAID', paidAt: new Date() } })
   516→      }
   517→      await poEvent(id, { type: 'RELEASED', actorId: user.id, actorName: user.name, actorSide: side, note: `Released ${(accepted / 100).toLocaleString()} KES to supplier. Platform commission: ${(commission / 100).toLocaleString()} KES recorded separately.` })
   518→      // Supplier settlement: money is now OWED to the supplier (PENDING) — it
   519→      // becomes SETTLED only on payout-provider confirmation or an audited
   520→      // FINANCE reconciliation. Never simulated here.
   521→      await createSettlementForRelease({
   522→        supplierOrgId: order.supplierOrgId,
   523→        orderId: id,
   524→        releasedMinor: accepted,
   525→        commissionMinor: commission,
   526→      })
   527→      await audit({
   528→        actorId: user.id, actorName: user.name,
   529→        action: 'PAYMENT_RELEASED', targetType: 'PURCHASE_ORDER', targetId: id,
   530→        metadata: { released: accepted, commission, ref: order.refCode },
   531→      })
   532→      await notifySupplierMembers(order.supplierOrgId, order.refCode, `Payment released for ${order.refCode}: ${(accepted / 100).toLocaleString()} KES (commission ${(commission / 100).toLocaleString()} KES).`, 'PO_RELEASED')
   533→      return ok({ releasedMinor: accepted, commissionMinor: commission, message: 'Payment released. Receipt generated in Billing.' })
   534→    }
   535→
   536→    // ─────────── STEP 9: dispute ───────────
   537→    case 'dispute': {
   538→      const existing = await db.dispute.findFirst({ where: { orderId: id, status: { in: ['OPEN', 'UNDER_REVIEW'] } } })
   539→      if (existing) return fail(409, 'DISPUTE_EXISTS', 'An open dispute already exists for this order.')
   540→      const disputed = order.paymentStatus === 'PROTECTED' || order.paymentStatus === 'PARTIALLY_RELEASED'
   541→      const dispute = await db.dispute.create({
   542→        data: {
   543→          refCode: refCode('DSP', 6),
   544→          orderId: id,
   545→          openedById: user.id,
   546→          openedBySide: side,
   547→          reason: body.reason,
   548→          details: body.details,
   549→          evidence: body.evidence.length ? JSON.stringify(body.evidence) : null,
   550→          amountFrozenMinor: disputed ? (order.acceptedTotalMinor !== null ? order.totalMinor - order.acceptedTotalMinor : order.totalMinor) : null,
   551→          status: 'OPEN',
   552→          events: { create: { type: 'CREATED', actorId: user.id, actorName: user.name, note: `Dispute opened: ${body.reason}` } },
   553→        },
   554→      })
   555→      await db.purchaseOrder.update({ where: { id }, data: { status: 'DISPUTED' } })
   556→      // Hold any owed payout for this order until the dispute resolves.
   557→      await holdSettlementsForOrder(id, `Dispute ${dispute.refCode}: ${body.reason}`)
   558→      await poEvent(id, { type: 'NOTE', actorId: user.id, actorName: user.name, actorSide: side, note: `Dispute ${dispute.refCode} opened (${body.reason}). Amount frozen where supported.` })
   559→      await audit({ actorId: user.id, actorName: user.name, action: 'DISPUTE_OPENED', targetType: 'DISPUTE', targetId: dispute.id, metadata: { order: order.refCode, reason: body.reason } })
   560→      const { staffUserIds } = await import('@/lib/orgs')
   561→      await notifyMany(await staffUserIds('ADMIN', 'FOUNDER', 'SUPPORT'), {
   562→        type: 'DISPUTE_OPENED', title: `Dispute ${dispute.refCode}: ${order.refCode}`, body: `${body.reason}: ${body.details.slice(0, 120)}`, link: '/admin?tab=disputes',
   563→      })
   564→      return ok({ dispute, message: 'Dispute filed. Platform admins will investigate with all order, payment and delivery evidence.' }, { status: 201 })
   565→    }
   566→
   567→    case 'request-return': {
   568→      if (side !== 'GARAGE') return fail(403, 'PERMISSION_DENIED', 'The garage requests returns.')
   569→      const ret = await db.partReturn.create({
   570→        data: {
   571→          refCode: refCode('RTN', 6),
   572→          orderId: id,
   573→          reason: body.reason,
   574→          itemsJson: JSON.stringify(body.items),
   575→        },
   576→      })
   577→      await poEvent(id, { type: 'NOTE', actorId: user.id, actorName: user.name, actorSide: 'GARAGE', note: `Return requested ${ret.refCode}: ${body.reason}` })
   578→      await notifySupplierMembers(order.supplierOrgId, order.refCode, `Return request ${ret.refCode} on ${order.refCode}: ${body.reason}`, 'PO_RETURN')
   579→      return ok({ ret }, { status: 201 })
   580→    }
   581→
   582→    case 'warranty': {
   583→      const product = order.items.find((i) => i.productId === body.productId)?.product
   584→      if (!product) return fail(404, 'NOT_FOUND', 'Product is not part of this order.')
   585→      if (product.warrantyMonths <= 0) return fail(409, 'NO_WARRANTY', 'This product was supplied without a warranty.')
   586→      const claim = await db.warrantyClaim.create({
   587→        data: {
   588→          refCode: refCode('WRT', 6),
   589→          orderId: id, productId: body.productId,
   590→          garageOrgId: order.garageOrgId, supplierOrgId: order.supplierOrgId,
   591→          issue: body.issue,
   592→        },
   593→      })
   594→      await poEvent(id, { type: 'NOTE', actorId: user.id, actorName: user.name, actorSide: 'GARAGE', note: `Warranty claim ${claim.refCode}: ${body.issue.slice(0, 120)}` })
   595→      await notifySupplierMembers(order.supplierOrgId, order.refCode, `Warranty claim ${claim.refCode} on ${order.refCode}.`, 'PO_WARRANTY')
   596→      return ok({ claim }, { status: 201 })
   597→    }
   598→  }
   599→
   600→  return fail(400, 'UNKNOWN_ACTION', 'Unsupported action.')
   601→})

   603→async function finaliseHandoverIfBoth(orderId: string): Promise<void> {
   604→  const delivery = await db.delivery.findUnique({ where: { orderId } })
   605→  if (!delivery) return
   606→  if (delivery.garageConfirmAt && delivery.supplierConfirmAt && delivery.status !== 'DELIVERED') {
   607→    await db.delivery.update({ where: { id: delivery.id }, data: { status: 'DELIVERED', deliveredAt: new Date() } })
   608→    await db.purchaseOrder.update({ where: { id: orderId }, data: { status: 'DELIVERED' } })
   609→  } else if (delivery.status === 'DISPATCHED') {
   610→    await db.delivery.update({ where: { id: delivery.id }, data: { status: 'IN_TRANSIT' } })
   611→    await db.purchaseOrder.update({ where: { id: orderId }, data: { status: 'IN_TRANSIT' } })
   612→  }
   613→}
